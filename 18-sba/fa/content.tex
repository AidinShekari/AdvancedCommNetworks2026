\textbf{مشکل مقیاس‌پذیری:}

\begin{itemize}
\tightlist
\item

\item
  هر واسط مشخصات پروتکل خود را دارد (\lr{\mbox{Diameter}}، \lr{\mbox{GTP-C}} و غیره)
\item
  افزودن یک عنصر شبکه جدید نیازمند تعریف واسط‌های جدید به هر عنصر موجود است
\item
  جفت‌شدگی سخت، تحول را بسیار دشوار می‌کند
\item
  هم‌کنش‌گری سازندگان نیازمند آزمون هر ترکیب واسط است
\end{itemize}

\textbf{مثال:} \lr{\mbox{EPC}} با 6 عنصر هسته \ensuremath{\leftarrow} تا 15 واسط یکتا، هر یک با پروتکل، قالب‌های پیام و رویه‌های خود.

\CNsubsection{1.2}{راه‌حل \lr{\mbox{SBA}}}

\lr{\mbox{5G}} \textbf{معماری خدمات‌محور} را معرفی می‌کند که در آن:

\begin{itemize}
\tightlist
\item
  توابع شبکه \lr{\mbox{(NF)}} \textbf{خدمات} را از طریق یک چارچوب واسط مشترک ارایه می‌دهند
\item
  هر \lr{\mbox{NF}} می‌تواند خدمات هر \lr{\mbox{NF}} دیگر را \textbf{کشف و مصرف} کند
\item
  افزودن یک \lr{\mbox{NF}} جدید فقط نیازمند ثبت خدمات آن است \lr{—} بدون نیاز به واسط‌های جدید
\item
  پشته پروتکل مشترک \lr{(HTTP/\allowbreak{}2 + JSON)} برای تمام ارتباطات صفحه کنترل
\end{itemize}

\CNsubsection{2.2}{مدل‌های تعامل خدمات \lr{\mbox{NF}}}

\lr{\mbox{NF}}ها با استفاده از دو الگو ارتباط برقرار می‌کنند:

\textbf{1. درخواست-پاسخ (همگام)}

\begin{itemize}
\tightlist
\item
  مصرف‌کننده \lr{\mbox{NF}} یک درخواست به تولیدکننده \lr{\mbox{NF}} می‌فرستد
\item
  تولیدکننده پردازش می‌کند و پاسخی بازمی‌گرداند
\item
  مثال: \lr{\mbox{AMF}} از \lr{\mbox{UDM}} برای داده مشترک پرس‌وجو می‌کند
\end{itemize}

\textbf{2. اشتراک-اعلان (ناهمگام)}

\begin{itemize}
\tightlist
\item
  مصرف‌کننده \lr{\mbox{NF}} به رویدادهای تولیدکننده \lr{\mbox{NF}} اشتراک می‌کند
\item
  تولیدکننده هنگام رخ‌دادن رویدادها اعلان می‌فرستد
\item
  مثال: \lr{\mbox{AMF}} به تغییرات داده اشتراک در \lr{\mbox{UDM}} اشتراک می‌کند
\end{itemize}

\CNsubsection{3.1}{\lr{\mbox{AMF}} \lr{—} تابع مدیریت دسترسی و تحرک}

\textbf{هدف:} تمام سیگنالینگ مرتبط با دسترسی و تحرک \lr{\mbox{UE}} را مدیریت می‌کند.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Namf}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Namf\_Communication} \textbar{}} انتقال پیام \lr{N1/\allowbreak{}N2}، سیگنالینگ مختص \lr{\mbox{UE}} \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Namf\_EventExposure} \textbar{}} نمایش رویدادهای تحرک (مکان، دسترسی‌پذیری) \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Namf\_MT} \textbar{}} تحویل پیام با مقصد موبایل \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Namf\_Location} \textbar{}} خدمات مکان \lr{\mbox{UE}} \lr{\textbar{}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  ثبت‌نام و لغو ثبت‌نام \lr{\mbox{UE}}
\item
  مدیریت اتصال (بیکار \ensuremath{\leftrightarrow} متصل)
\item
  مدیریت تحرک (تحویل، به‌روزرسانی نواحی ردیابی)
\item
  مدیریت زمینه امنیتی (رمزنگاری/\allowbreak{}یکپارچگی \lr{\mbox{NAS}})
\item
  رله پیام‌های \lr{\mbox{NAS-SM}} بین \lr{\mbox{UE}} و \lr{\mbox{SMF}}
\end{itemize}

\CNsubsection{3.2}{\lr{\mbox{SMF}} \lr{—} تابع مدیریت نشست}

\textbf{هدف:} نشست‌های \lr{\mbox{PDU}} (داده) بین \lr{\mbox{UE}} و شبکه داده را مدیریت می‌کند.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Nsmf}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Nsmf\_PDUSession} \textbar{}} ایجاد، اصلاح، آزادسازی نشست‌های \lr{\mbox{PDU}} \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nsmf\_EventExposure} \textbar{}} رویدادهای نشست \lr{\mbox{PDU}} (مثلاً تغییر آدرس \lr{\mbox{IP}}) \lr{\textbar{}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  برقراری، اصلاح، آزادسازی نشست \lr{\mbox{PDU}}
\item
  تخصیص و مدیریت آدرس \lr{\mbox{IP}} کاربر
\item
  انتخاب و کنترل \lr{\mbox{UPF}} (از طریق \lr{N4/\allowbreak{}PFCP})
\item
  مدیریت و اجرای جریان \lr{\mbox{QoS}}
\item
  اعلان داده پیوند پایین‌سو
\item
  پیکربندی هدایت و مسیریابی ترافیک
\end{itemize}

\CNsubsection{3.3}{\lr{\mbox{UPF}} \lr{—} تابع صفحه کاربر}

\textbf{هدف:} بسته‌های داده کاربر را بین \lr{\mbox{RAN}} و شبکه‌های داده ارسال می‌کند.

\textbf{\lr{\mbox{\CNwarn{}}} یادداشت:} \lr{\mbox{UPF}} از \lr{\mbox{SBI}} استفاده \textbf{نمی‌کند}. از نقاط مرجع استفاده می‌کند:

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{N3}}}}} \lr{—} بین \lr{\mbox{RAN}} و \lr{\mbox{UPF}} (تونل \lr{\mbox{GTP-U}})
\item
  \lr{\mbox{\textbf{\lr{\mbox{N4}}}}} \lr{—} بین \lr{\mbox{SMF}} و \lr{\mbox{UPF}} (پروتکل \lr{\mbox{PFCP}})
\item
  \lr{\mbox{\textbf{\lr{\mbox{N6}}}}} \lr{—} بین \lr{\mbox{UPF}} و شبکه داده (ترافیک \lr{\mbox{IP}})
\item
  \lr{\mbox{\textbf{\lr{\mbox{N9}}}}} \lr{—} بین \lr{\mbox{UPF}}ها (تونل \lr{\mbox{GTP-U}} بین-\lr{\mbox{UPF}})
\end{itemize}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  مسیریابی و ارسال بسته
\item
  مدیریت \lr{\mbox{QoS}} (علامت‌گذاری، پلیسینگ، شکل‌دهی)
\item
  گزارش‌دهی مصرف ترافیک
\item
  تأیید ترافیک پیوند بالاسو
\item
  رهگیری قانونی (صفحه کاربر)
\item
  بافرینگ بسته‌های پیوند پایین‌سو (\lr{\mbox{UE}} در حالت بیکار)
\end{itemize}

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Nudm}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Nudm\_}\allowbreak{}\CNcode{SubscriberDataManagement} \textbar{}} بازیابی/\allowbreak{}به‌روزرسانی داده اشتراک \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nudm\_}\allowbreak{}\CNcode{UEContextManagement} \textbar{}} ردیابی اینکه کدام \lr{AMF/\allowbreak{}SMF} به \lr{\mbox{UE}} خدمات می‌دهد \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nudm\_}\allowbreak{}\CNcode{UEAuthentication} \textbar{}} تولید بردارهای احراز هویت \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nudm\_EventExposure} \textbar{}} اعلان‌های تغییر داده اشتراک \lr{\textbar{}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}

\CNsubsection{3.6}{\lr{\mbox{PCF}} \lr{—} تابع کنترل سیاست}

\textbf{هدف:} قواعد سیاستی که رفتار شبکه و \lr{\mbox{QoS}} را اداره می‌کنند فراهم می‌کند.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Npcf}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Npcf\_}\allowbreak{}\CNcode{SMPolicyControl} \textbar{}} سیاست‌های سطح نشست (\lr{\mbox{QoS}}، صورت‌حساب، دروازه‌بندی) \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Npcf\_}\allowbreak{}\CNcode{AMPolicyControl} \textbar{}} سیاست‌های دسترسی و تحرک (\lr{\mbox{RFSP}}، ناحیه خدمات) \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Npcf\_}\allowbreak{}\CNcode{PolicyAuthorization} \textbar{}} سیاست با محرک \lr{\mbox{AF}} (مثلاً \lr{\mbox{IMS}} درخواست \lr{\mbox{QoS}} می‌کند) \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Npcf\_}\allowbreak{}\CNcode{BDTPolicyControl} \textbar{}} سیاست‌های انتقال داده پس‌زمینه \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Npcf\_EventExposure} \textbar{}} اعلان‌های رویداد مرتبط با سیاست \lr{\textbar{}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  تصمیمات سیاست \lr{\mbox{QoS}}
\item
  قواعد سیاست صورت‌حساب
\item
  تصمیمات کنترل دسترسی و دروازه‌بندی
\item
  سیاست‌های شبکه‌بندی
\item
  کنترل پایش مصرف
\end{itemize}

\CNsubsection{3.7}{\lr{\mbox{NRF}} \lr{—} تابع مخزن شبکه}

\textbf{هدف:} \textbf{ثبت مرکزی} تمام نمونه‌های \lr{\mbox{NF}} و خدمات آن‌ها \lr{—} ستون فقرات \lr{\mbox{SBA}}.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Nnrf}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Nnrf\_NFManagement} \textbar{}} ثبت‌نام، به‌روزرسانی، لغو ثبت‌نام \lr{\mbox{NF}} \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nnrf\_NFDiscovery} \textbar{}} پرس‌وجوی \lr{\mbox{NF}}های موجود بر اساس نوع، برش، مکان \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nnrf\_AccessToken} \textbar{}} صدور نشانه‌های \lr{\mbox{OAuth2.0}} برای مجوزدهی \lr{\mbox{NF}}-به-\lr{\mbox{NF}} \lr{\textbar{}}

\begin{CNquote}

\lr{\mbox{\CNstar{}}} \textbf{\lr{\mbox{NRF}} توانمندساز حیاتی \lr{\mbox{SBA}} است} \lr{—} بدون آن، \lr{\mbox{NF}}ها نمی‌توانند یکدیگر را کشف کنند و مدل خدمات‌محور به پیکربندی ایستا بازمی‌گردد.

\end{CNquote}

\CNsubsection{3.8}{\lr{\mbox{NSSF}} \lr{—} تابع انتخاب برش شبکه}

\textbf{هدف:} نمونه برش شبکه مناسب را برای \lr{\mbox{UE}} انتخاب می‌کند.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Nnssf}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Nnssf\_NSSelection} \textbar{}} تعیین برش‌های مجاز و مجموعه \lr{\mbox{AMF}} هدف \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nnssf\_}\allowbreak{}\CNcode{NSSAIAvailability} \textbar{}} مدیریت دسترسی‌پذیری \lr{\mbox{NSSAI}} در هر ناحیه ردیابی \lr{\textbar{}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  تعیین \lr{\mbox{NSSAI}} مجاز برای ثبت‌نام \lr{\mbox{UE}}
\item
  انتخاب مجموعه \lr{AMF/\allowbreak{}AMF} هدف بر اساس برش‌های درخواستی
\item
  مدیریت اطلاعات دسترسی‌پذیری \lr{\mbox{NSSAI}} در هر \lr{\mbox{TA}}
\end{itemize}

\CNsubsection{3.9}{\lr{\mbox{NEF}} \lr{—} تابع نمایش شبکه}

\textbf{هدف:} قابلیت‌های شبکه \lr{\mbox{5G}} را به‌صورت ایمن به کاربردهای خارجی نمایش می‌دهد.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Nnef}}}

\textbf{خدمات کلیدی:}\CNbr{}\lr{\textbar{}} خدمت \lr{\textbar{}} توضیح \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}—\textbar{}—\textbar{}}}\CNbr{}\lr{\textbar{} \CNcode{Nnef\_EventExposure} \textbar{}} نمایش رویدادهای شبکه به \lr{\mbox{AF}}ها \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nnef\_PFDManagement} \textbar{}} مدیریت توصیف جریان بسته \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nnef\_}\allowbreak{}\CNcode{TrafficInfluence} \textbar{}} امکان تأثیر \lr{\mbox{AF}} بر مسیریابی ترافیک \lr{\textbar{}}\CNbr{}\lr{\textbar{} \CNcode{Nnef\_}\allowbreak{}\CNcode{ParameterProvision} \textbar{}} تأمین رفتار مورد انتظار \lr{\mbox{UE}} \lr{\textbar{}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  دروازه \lr{\mbox{API}} بین هسته \lr{\mbox{5G}} و دنیای خارجی
\item
  ترجمه اطلاعات داخلی \lr{\mbox{5GC}} به بازنمایی‌های خارجی
\item
  اجرای سیاست‌های امنیتی بر دسترسی خارجی
\item
  محدودسازی نرخ و مدیریت \lr{\mbox{API}}
\end{itemize}

\CNsubsection{3.10}{\lr{\mbox{AF}} \lr{—} تابع کاربرد}

\textbf{هدف:} نمایندگی کاربردهای شخص سوم که با هسته \lr{\mbox{5G}} تعامل دارند.

\textbf{واسط خدمات:} \lr{\mbox{\CNcode{Naf}}}

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  درخواست \lr{\mbox{QoS}} برای جریان‌های کاربرد (مثلاً پخش جریانی ویدیو)
\item
  تأثیر بر مسیریابی ترافیک (مثلاً رایانش لبه)
\item
  اشتراک در رویدادهای شبکه (مثلاً مکان \lr{\mbox{UE}})
\item
  می‌تواند مستقیماً به \lr{\mbox{5GC}} دسترسی یابد (مورد اعتماد) یا از طریق \lr{\mbox{NEF}} (غیرقابل اعتماد)
\end{itemize}

\textbf{مثال‌ها:} \lr{IMS (VoNR)}، پلتفرم‌های پخش جریانی، کاربردهای سازمانی، هماهنگ‌سازهای \lr{\mbox{MEC}}.

\textbf{اکوسیستم سازندگان} & سازندگان مختص مخابرات & سازندگان ابر/\allowbreak{}\lr{\mbox{IT}} می‌توانند مشارکت کنند \\

\CNboxsection{پرسش‌های مرور}

\begin{CNbox}[unbreakable]{q}{پرسش‌های مرور}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  محاسبه کنید برای 8 عنصر \lr{\mbox{EPC}} چند واسط یکتا لازم است. \lr{\mbox{SBA}} چگونه این را حل می‌کند؟
\item
  چرا \lr{HTTP/\allowbreak{}2} بر \lr{HTTP/\allowbreak{}1.1} برای \lr{\mbox{SBI}} انتخاب شد؟
\item
  در حین ثبت‌نام \lr{\mbox{NRF}} چه اطلاعاتی در پروفایل \lr{\mbox{NF}} گنجانده می‌شود؟
\item
  چرا \lr{\mbox{UPF}} به گذرگاه \lr{\mbox{SBI}} متصل نیست؟
\item
  مدل‌های ارتباطی \lr{\mbox{B}} و \lr{\mbox{C}} در \lr{\mbox{SCP}} را مقایسه کنید \lr{—} چه زمانی هر یک را استفاده می‌کنید؟
\item
  \lr{OAuth 2.0} در بستر \lr{\mbox{SBA}} در \lr{\mbox{5G}} چگونه کار می‌کند؟ \lr{\mbox{NRF}} چه نقشی دارد؟
\item
  یک اپراتور می‌خواهد یک \lr{\mbox{NF}} تحلیلی جدید \lr{\mbox{(NWDAF)}} اضافه کند. میزان تلاش را در معماری \lr{\mbox{EPC}} در برابر \lr{\mbox{SBA}} مقایسه کنید.
\item
  چرا \lr{\mbox{NRF}} به‌عنوان «تنها حیاتی‌ترین» مؤلفه \lr{\mbox{SBA}} در نظر گرفته می‌شود؟
\end{enumerate}

\end{CNbox}
