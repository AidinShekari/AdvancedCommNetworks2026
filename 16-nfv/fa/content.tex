\CNsubsection{}{هسته \lr{\mbox{5G = CNF}}ها}

هسته \lr{\mbox{5G}} گروه \lr{3GPP (5GC)} به‌عنوان یک \textbf{معماری خدمات‌محور \lr{\mbox{(SBA)}}} با اصول بومی-ابر طراحی شده است:

\begin{itemize}
\tightlist
\item
  هر تابع شبکه (\lr{\mbox{AMF}}، \lr{\mbox{SMF}}، \lr{\mbox{UPF}}، \lr{\mbox{NRF}} و غیره) یک \lr{\mbox{\textbf{\lr{\mbox{CNF}}}}} است
\item
  روی \lr{\mbox{\textbf{\lr{\mbox{Kubernetes}}}}} مستقر می‌شود
\item
  از طریق \textbf{\lr{HTTP/\allowbreak{}2 + API}های \lr{\mbox{REST}}} ارتباط برقرار می‌کند
\item
  طراحی بدون حالت با مخازن داده خارجی (مثلاً \lr{\mbox{UDSF}})
\item
  مقیاس‌پذیری افقی، ارتقاهای غلتان، خود-ترمیمی
\end{itemize}

\item
  \lr{Horizontal Pod Autoscaler} برای مقیاس‌دهی پویا
\end{itemize}

\CNsection{10}{چالش‌های \lr{\mbox{NFV}}}

\CNchained\CNsubsection{10.1}{عملکرد (پردازش بسته)}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{VM}}ها/\allowbreak{}کانتینرها در مقایسه با پردازش \lr{\mbox{ASIC}} برهنه سربار اضافه می‌کنند
\item
  \textbf{کاهش اثر}: \lr{\mbox{DPDK}}، \lr{\mbox{SR-IOV}}، \lr{\mbox{SmartNIC}}ها، پین‌کردن \lr{\mbox{CPU}}، صفحات بزرگ
\item
  \lr{\mbox{UPF}} و سایر \lr{\mbox{VNF}}های صفحه داده حساس‌ترین هستند
\end{itemize}

\CNsubsection{10.2}{پیچیدگی هماهنگ‌سازی}

\begin{itemize}
\tightlist
\item
  پشته \lr{\mbox{MANO}} برای استقرار و بهره‌برداری پیچیده است
\item
  مسایل هم‌کنش‌گری \lr{\mbox{VNF}} چند-سازنده
\item
  استانداردسازی توصیف‌گر همچنان در حال تحول است (\lr{\mbox{TOSCA}}، \lr{SOL001/\allowbreak{}006})
\item
  عملیات روز-2 (پایش، ترمیم، سیاست‌های مقیاس‌دهی) دشوار است
\end{itemize}

\CNsubsection{10.3}{مدیریت حالت}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{VNF}}ها اغلب به‌عنوان یکپارچه‌های حالت‌دار طراحی می‌شوند (منتقل‌شده از تجهیزات)
\item
  \lr{\mbox{VM}}های حالت‌دار برای مقیاس‌دهی افقی یا مهاجرت دشوار هستند
\item
  رویکرد \lr{\mbox{CNF}}: خارجی‌سازی حالت (\lr{\mbox{Redis}}، \lr{\mbox{etcd}}، پایگاه‌های داده)
\item
  چالش: تأخیر دسترسی به حالت خارجی برای توابع پرتوان‌عملیات
\end{itemize}

\CNsubsection{10.4}{قابلیت اطمینان و دسترس‌پذیری}

\begin{itemize}
\tightlist
\item
  شکست‌های نرم‌افزاری مکررتر از شکست‌های سخت‌افزاری هستند
\item
  نیازمند پایش سلامت مقاوم، خود-ترمیمی، افزونگی
\item
  دسترس‌پذیری در سطح حامل \CNnum{(99.999\%)} نیازمند طراحی دقیق است
\end{itemize}

\CNsubsection{10.5}{امنیت}

\begin{itemize}
\tightlist
\item
  سطح حمله بزرگ‌تر (\lr{\mbox{hypervisor}}، \lr{\mbox{API}}های هماهنگ‌سازی، زیرساخت مشترک)
\item
  جداسازی چند-مستأجری باید در چندین لایه اجرا شود
\item
  آسیب‌پذیری‌های فرار از کانتینر
\end{itemize}

\CNsubsection{10.6}{شکاف مهارتی}

\begin{itemize}
\tightlist
\item
  اپراتورها باید فناوری‌های ابری را بیاموزند (\lr{\mbox{OpenStack}}، \lr{\mbox{K8s}}، \lr{CI/\allowbreak{}CD})
\item
  همگرایی مخابرات + فناوری اطلاعات نیازمند مدل‌های سازمانی جدید است
\end{itemize}

\CNboxsection{نکات کلیدی}

\begin{CNbox}[unbreakable]{sum}{نکات کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{NFV}} توابع شبکه را \textbf{از سخت‌افزار اختصاصی جدا می‌کند} \ensuremath{\leftarrow} نرم‌افزار روی \lr{\mbox{COTS}}
\item
  معماری \lr{\mbox{NFV}} سازمان \lr{\mbox{ETSI}}: \lr{\textbf{\lr{NFVI + VNF + MANO}}}
\item
  \textbf{\lr{\mbox{VNF}}ها} (مبتنی بر \lr{\mbox{VM}}) با \textbf{\lr{\mbox{CNF}}ها} (مبتنی بر کانتینر، بومی-ابر) جایگزین می‌شوند
\item
  \textbf{هسته \lr{\mbox{5G}} از ابتدا به‌عنوان \lr{\mbox{CNF}}} روی \lr{\mbox{Kubernetes}} طراحی شده است
\item
  \lr{\mbox{\textbf{\lr{\mbox{MANO}}}}} کل چرخه عمر را هماهنگ می‌کند: نمونه‌سازی، مقیاس‌دهی، ترمیم، خاتمه
\item
  تحول: فیزیکی \ensuremath{\leftarrow} مجازی \lr{\mbox{(VNF)}} \ensuremath{\leftarrow} \textbf{بومی-ابر \lr{\mbox{(CNF)}}}
\item
  چالش‌های کلیدی: \textbf{عملکرد، پیچیدگی هماهنگ‌سازی و مدیریت حالت}
\item
  \lr{NFV + SDN} با هم \textbf{شبکه‌های مخابراتی کاملاً قابل برنامه‌ریزی و خودکار} را ممکن می‌سازند
\end{enumerate}

\end{CNbox}
