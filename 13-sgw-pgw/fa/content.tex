\end{enumerate}

\CNsection{2}{\lr{\mbox{S-GW}} (دروازه خدمات‌دهنده)}

\CNchained\CNsubsection{2.1}{ارسال صفحه کاربر}

\lr{\mbox{S-GW}} اساساً یک \textbf{سوییچ تونل \lr{\mbox{GTP-U}}} است:

\begin{itemize}
\tightlist
\item
  بسته‌های \lr{\mbox{DL}} را از \lr{\mbox{P-GW}} روی تونل \lr{\mbox{GTP-U}} در \lr{S5/\allowbreak{}S8} دریافت می‌کند
\item
  آن‌ها را به \lr{eNodeB (eNB)} روی تونل \lr{\mbox{GTP-U}} در \lr{\mbox{S1-U}} ارسال می‌کند
\item
  بسته‌های \lr{\mbox{UL}} را از \lr{\mbox{eNB}} روی \lr{\mbox{S1-U}} دریافت و به \lr{\mbox{P-GW}} روی \lr{S5/\allowbreak{}S8} ارسال می‌کند
\end{itemize}

هر حامل \textbf{تونل‌های \lr{\mbox{GTP-U}} جداگانه} روی هر واسط دارد (شناسایی‌شده با \lr{\mbox{TEID}}).

\CNsubsection{2.2}{لنگر تحرک محلی}

در حین \textbf{تحویل \lr{\mbox{X2}}} (مستقیم \lr{\mbox{eNB}}-به-\lr{\mbox{eNB}}):

\begin{itemize}
\tightlist
\item
  \lr{\mbox{S-GW}} تغییر \textbf{نمی‌کند}
\item
  \lr{\mbox{eNB}} مبدأ بسته‌های بافر‌شده/\allowbreak{}در حال عبور را از طریق \lr{\mbox{X2}} به \lr{\mbox{eNB}} مقصد ارسال می‌کند
\item
  پس از تحویل، \lr{\mbox{MME}} یک \lr{\textbf{\lr{Modify Bearer Request}}} با آدرس/\allowbreak{}\lr{\mbox{TEID}} جدید \lr{\mbox{eNB}} به \lr{\mbox{S-GW}} می‌فرستد
\item
  \lr{\mbox{S-GW}} نقطه انتهایی تونل \lr{\mbox{S1-U}} را از \lr{\mbox{eNB}} مبدأ به \lr{\mbox{eNB}} مقصد تغییر می‌دهد (\textbf{تغییر مسیر})
\end{itemize}

این کار از تخریب تونل \lr{\mbox{S5}} به \lr{\mbox{P-GW}} جلوگیری می‌کند \lr{—} فقط بخش \lr{\mbox{S1-U}} تغییر می‌کند.

\CNsubsection{2.3}{بافرینگ داده پیوند پایین‌سو}

وقتی \lr{\mbox{UE}} در \lr{\mbox{\textbf{\lr{\mbox{ECM-IDLE}}}}} است (هیچ تونل \lr{\mbox{S1-U}} وجود ندارد):

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  بسته‌های \lr{\mbox{DL}} از \lr{\mbox{P-GW}} روی تونل \lr{\mbox{S5}} می‌رسند
\item
  \lr{\mbox{S-GW}} نقطه انتهایی \lr{\mbox{S1-U}} ندارد \ensuremath{\leftarrow} \textbf{بسته‌ها را بافر می‌کند}
\item
  \lr{\mbox{S-GW}} یک \lr{\textbf{\lr{Downlink Data Notification}}} به \lr{\mbox{MME}} می‌فرستد (از طریق \lr{\mbox{S11}})
\item
  \lr{\mbox{MME}} صفحه‌بندی را آغاز می‌کند
\item
  پس از پاسخ \lr{\mbox{UE}} (درخواست خدمات)، \lr{\mbox{MME}} یک \lr{Modify Bearer Request} با \lr{\mbox{TEID}} جدید \lr{\mbox{eNB}} می‌فرستد
\item
  \lr{\mbox{S-GW}} بسته‌های بافر‌شده را به \lr{\mbox{eNB}} تخلیه می‌کند
\end{enumerate}

\textbf{محدودیت‌های بافر}: قابل پیکربندی در هر اپراتور؛ معمولاً \CNnum{1-10} بسته در هر حامل. مازاد دور ریخته می‌شود.

\CNsubsection{2.4}{صورت‌حساب در هر \lr{\mbox{UE}}/\allowbreak{}هر حامل}

\lr{\mbox{S-GW}} \textbf{\lr{\mbox{CDR}}ها (رکوردهای داده صورت‌حساب)} را با موارد زیر تولید می‌کند:

\begin{itemize}
\tightlist
\item
  حجم \lr{UL/\allowbreak{}DL} در هر حامل
\item
  مدت
\item
  اطلاعات \lr{\mbox{QoS}} (\lr{\mbox{QCI}}، \lr{\mbox{ARP}}، \lr{GBR/\allowbreak{}MBR})
\item
  علت بسته‌شدن رکورد (تحویل، محدودیت زمان، محدودیت حجم)
\end{itemize}

از طریق واسط \lr{Rf/\allowbreak{}Gz} به \textbf{سیستم صورت‌حساب آفلاین \lr{\mbox{(CDF)}}} فرستاده می‌شود.

\CNsubsection{2.5}{رهگیری قانونی}

\lr{\mbox{S-GW}} می‌تواند ترافیک صفحه کاربر را برای رهگیری قانونی \lr{\mbox{(LI)}} آینه کند:

\begin{itemize}
\tightlist
\item
  واسط \lr{\mbox{X2}} (استاندارد \lr{\mbox{ETSI}}) به تسهیلات پایش اجرای قانون \lr{\mbox{(LEMF)}}
\item
  با حکم قضایی فعال می‌شود، از طریق واسط \lr{\mbox{X1}} از مدیریت \lr{\mbox{LI}} فعال می‌گردد
\item
  بسته‌ها را بدون تأثیر بر سرویس \lr{\mbox{UE}} تکثیر می‌کند
\end{itemize}

\CNsection{3}{\lr{\mbox{P-GW}} (دروازه \lr{\mbox{PDN}})}

\CNchained\CNsubsection{3.1}{اتصال \lr{\mbox{PDN}} و تخصیص آدرس \lr{\mbox{IP}}}

\lr{\mbox{P-GW}} \textbf{دروازه \lr{\mbox{UE}} به اینترنت/\allowbreak{}\lr{\mbox{PDN}}} است \lr{—} آدرس \lr{\mbox{IP}} را به \lr{\mbox{UE}} تخصیص می‌دهد:

\textbf{روش‌های تخصیص \lr{\mbox{IPv4}}:}

\begin{itemize}
\tightlist
\item
  \textbf{ایستا}: پیش‌پیکربندی‌شده در داده اشتراک \lr{\mbox{HSS}}
\item
  \textbf{پویا \lr{\mbox{(DHCPv4)}}}: \lr{\mbox{P-GW}} به‌عنوان کارساز \lr{\mbox{DHCP}} عمل می‌کند یا به \lr{\mbox{DHCP}} خارجی رله می‌کند
\item
  \textbf{پویا (استخر داخلی)}: \lr{\mbox{P-GW}} از استخر آدرس محلی تخصیص می‌دهد (رایج‌ترین)
\end{itemize}

\textbf{تخصیص \lr{\mbox{IPv6}}:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{P-GW}} یک پیشوند /\allowbreak{}\lr{\mbox{64}} از طریق \textbf{اعلام مسیریاب \lr{\mbox{(SLAAC)}}} تخصیص می‌دهد
\item
  \lr{\mbox{UE}} شناسه واسط را تولید می‌کند (\lr{\mbox{EUI-64}} یا افزونه‌های حریم خصوصی)
\item
  پیشوند در \lr{\mbox{PCO}} (گزینه‌های پیکربندی پروتکل) در حین فعال‌سازی حامل تحویل می‌شود
\end{itemize}

\textbf{دو-پشته \lr{\mbox{(IPv4v6)}}:}

\begin{itemize}
\tightlist
\item
  هم آدرس \lr{\mbox{IPv4}} و هم پیشوند \lr{\mbox{IPv6}} به‌طور همزمان تخصیص می‌یابند
\item
  نیازمند نوع \lr{PDN = IPv4v6} در اشتراک است
\end{itemize}

\CNsubsection{3.2}{اجرای سیاست}

\lr{\mbox{P-GW}} سیاست‌های دریافتی از \lr{\mbox{\textbf{\lr{\mbox{PCRF}}}}} را از طریق \textbf{واسط \lr{\mbox{Gx}}} اجرا می‌کند:

\begin{itemize}
\tightlist
\item
  \textbf{قواعد \lr{\mbox{PCC}} (کنترل سیاست و صورت‌حساب):}

  \begin{itemize}
  \tightlist
  \item
    فیلترهای \lr{\mbox{SDF}} (جریان داده خدمات) \lr{—} تطبیق 5-گانه
  \item
    پارامترهای \lr{\mbox{QoS}} (\lr{\mbox{QCI}}، \lr{\mbox{GBR}}، \lr{\mbox{MBR}}، \lr{\mbox{ARP}})
  \item
    کلیدهای صورت‌حساب و گروه‌های نرخ‌گذاری
  \item
    وضعیت دروازه (باز/\allowbreak{}بسته کردن جریان‌های مشخص)
  \end{itemize}
\item
  \textbf{\lr{\mbox{TFT}} (قالب جریان ترافیک):}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\lr{\mbox{TFT}} پیوند پایین‌سو}: در \lr{\mbox{P-GW}} \lr{—} بسته‌های \lr{\mbox{DL}} ورودی را به حامل صحیح نگاشت می‌کند
  \item
    \textbf{\lr{\mbox{TFT}} پیوند بالاسو}: به \lr{\mbox{UE}} فرستاده می‌شود \lr{—} \lr{\mbox{UE}} بسته‌های \lr{\mbox{UL}} را به حامل صحیح نگاشت می‌کند
  \item
    فیلترهای بسته: \lr{\mbox{IP}} مبدأ/\allowbreak{}مقصد، محدوده پورت، پروتکل، \lr{ToS/\allowbreak{}DSCP}
  \end{itemize}
\end{itemize}

\CNsubsection{3.4}{فیلترکردن و مسیریابی بسته}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{DL}}}}}: اینترنت \ensuremath{\leftarrow} \lr{\mbox{P-GW}}: \lr{\mbox{IP}} مقصد را با نشست \lr{\mbox{UE}} تطبیق می‌دهد، فیلتر \lr{\mbox{SDF}} را برای انتخاب حامل اعمال می‌کند، در \lr{\mbox{GTP-U}} به‌سمت \lr{\mbox{S-GW}} کپسوله می‌کند
\item
  \lr{\textbf{\lr{\mbox{UL}}}: UE \ensuremath{\to} P-GW}: \lr{\mbox{GTP-U}} را واکپسوله می‌کند، \lr{\mbox{NAT}} را اعمال می‌کند (در صورت آدرس‌دهی خصوصی)، به اینترنت مسیریابی می‌کند
\item
  \lr{\textbf{\lr{NAT/\allowbreak{}NAPT}}: P-GW} اغلب \lr{\mbox{NAT}} را برای \lr{\mbox{IPv4}} انجام می‌دهد (آدرس‌های خصوصی \lr{\mbox{UE}} \ensuremath{\leftarrow} عمومی)
\item
  \textbf{فشرده‌سازی سرآیند}: در \lr{\mbox{P-GW}} نیست (در لایه \lr{\mbox{PDCP}} در \lr{\mbox{RAN}} انجام می‌شود)
\end{itemize}

\begin{itemize}
\tightlist

\item
  واسط مبتنی بر \lr{\mbox{IP}} (بدون \lr{\mbox{GTP}})
\item
  معادل \lr{\mbox{Gi}} در \lr{\mbox{3G}} (\lr{\mbox{GGSN}} \ensuremath{\leftarrow} اینترنت)
\item

\item

  \item
    هسته \lr{\mbox{IMS}} (\lr{\mbox{APN}} مربوط به \lr{\mbox{ims}})
  \item

  \item
    پلتفرم‌های محتوا/\allowbreak{}\lr{\mbox{CDN}}
  \end{itemize}
\end{itemize}

\CNsubsection{6.2}{واسط \lr{\mbox{Sx}} و پروتکل \lr{\mbox{PFCP}}}

\textbf{\lr{\mbox{PFCP}} (پروتکل کنترل ارسال بسته)} \lr{—} \lr{3GPP TS 29.244}:

\begin{itemize}
\tightlist
\item
  بین مؤلفه‌های \lr{\mbox{CP}} و \lr{\mbox{UP}} اجرا می‌شود (\lr{\mbox{Sxa}} برای \lr{\mbox{S-GW}}، \lr{\mbox{Sxb}} برای \lr{\mbox{P-GW}}، \lr{\mbox{Sxc}} برای \lr{\mbox{TDF}})
\item
  \textbf{برقراری نشست}: \lr{\mbox{CP}} به \lr{\mbox{UP}} می‌گوید قواعد ارسال را ایجاد کند
\item
  \textbf{اصلاح نشست}: \lr{\mbox{CP}} قواعد را به‌روزرسانی می‌کند (مثلاً \lr{\mbox{TEID}} جدید پس از تحویل)
\item
  \textbf{حذف نشست}: \lr{\mbox{CP}} به \lr{\mbox{UP}} دستور می‌دهد وضعیت ارسال را حذف کند
\end{itemize}

\textbf{مفاهیم کلیدی \lr{\mbox{PFCP}}:}\CNbr{}\lr{\textbar{}} عنصر \lr{\textbar{}} هدف \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}———\textbar{}———\textbar{}}}\CNbr{}\lr{\textbar{}} \lr{\mbox{PDR}} (قاعده تشخیص بسته) \lr{\textbar{}} معیارهای تطبیق (\lr{\mbox{TEID}}، \lr{\mbox{IP}}، پورت) \lr{\textbar{}}\CNbr{}\lr{\textbar{}} \lr{\mbox{FAR}} (قاعده اقدام ارسال) \lr{\textbar{}} چه کاری انجام شود (ارسال، بافر، دور ریختن) \lr{\textbar{}}\CNbr{}\lr{\textbar{}} \lr{\mbox{QER}} (قاعده اجرای \lr{\mbox{QoS}}) \lr{\textbar{}} محدودسازی نرخ \lr{(GBR/\allowbreak{}MBR)} \lr{\textbar{}}\CNbr{}\lr{\textbar{}} \lr{\mbox{URR}} (قاعده گزارش‌دهی مصرف) \lr{\textbar{}} اندازه‌گیری حجم/\allowbreak{}زمان برای صورت‌حساب \lr{\textbar{}}\CNbr{}\lr{\textbar{}} \lr{\mbox{BAR}} (قاعده اقدام بافرینگ) \lr{\textbar{}} رفتار بافرینگ برای \lr{\mbox{UE}} بیکار \lr{\textbar{}}

\end{enumerate}
