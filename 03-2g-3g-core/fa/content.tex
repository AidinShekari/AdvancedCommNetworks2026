\CNsection{3.1}{تحول از \lr{\mbox{2G}} به \lr{\mbox{3G}} \lr{—} نیروهای محرک}

\CNchained\CNsubsection{}{چرا به \lr{\mbox{3G}} نیاز داشتیم؟}

گذار از \lr{\mbox{2G (GSM)}} به \lr{3G (UMTS)} توسط تقاضاهای بنیادی بازار و فناوری هدایت شد:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{تقاضای انفجاری داده}: \lr{\mbox{SMS}} و وب‌گردی ابتدایی \lr{\mbox{WAP}} کافی نبود. کاربران ایمیل، وب‌گردی و دانلود فایل را روی دستگاه‌های موبایل می‌خواستند.
\item
  \textbf{خدمات چندرسانه‌ای}: تماس ویدیویی، پخش جریانی صدا/\allowbreak{}ویدیو و محتوای غنی نیازمند پهنای باندی بسیار فراتر از \lr{9.6~kbps GSM} (یا حتی \lr{\mbox{171~kbps}} نظری \lr{\mbox{GPRS}}) بود.
\item
  \textbf{نرخ داده بالاتر}: چشم‌انداز \lr{\mbox{IMT-2000}} اتحادیه مخابرات بین‌المللی \lr{\mbox{(ITU)}} هدف‌گذاری کرده بود:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{144~kbps}} برای تحرک بالا (خودرویی)
  \item
    \lr{\mbox{384~kbps}} برای حالت پیاده
  \item
    \lr{\mbox{2~Mbps}} برای حالت ساکن/\allowbreak{}داخل ساختمان
  \end{itemize}
\item
  \textbf{رومینگ جهانی}: یک استاندارد یکپارچه جهانی (برخلاف \lr{\mbox{2G}} تکه‌تکه: \lr{\mbox{GSM}} در برابر \lr{\mbox{CDMA}} در برابر \lr{\mbox{PDC}}).
\item
  \textbf{کارایی طیفی}: کدهای گسترش \lr{\mbox{WCDMA}} به کاربران بیشتری در هر سلول با تخریب تدریجی امکان می‌دهند.
\end{enumerate}

\CNsubsection{}{چرا دو دامنه؟}

رویکرد تقسیم \lr{CS/\allowbreak{}PS} یک \textbf{تصمیم مهندسی عمل‌گرایانه} بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}

\item
  \textbf{تضمین‌های \lr{\mbox{QoS}}}: سوییچینگ مداری تأخیر قطعی برای صدا فراهم می‌کند (تحویل تضمین‌شده فریم \lr{\mbox{20ms}})

\item

\end{enumerate}

\CNsection{3.5}{واسط \lr{\mbox{Iu}}}

واسط \lr{\mbox{Iu}} مرز حیاتی بین \lr{\mbox{UTRAN}} و شبکه هسته است.

\CNsubsection{}{\lr{\mbox{Iu-CS}} (سوییچینگ مداری)}

\begin{itemize}
\tightlist
\item
  \textbf{انتقال}: \lr{\mbox{ATM}} (\lr{\mbox{AAL2}} برای صفحه کاربر، \lr{\mbox{AAL5}} برای صفحه کنترل)
\item
  \textbf{صفحه کنترل}: \lr{RANAP \ensuremath{\to} SCCP \ensuremath{\to} M3UA \ensuremath{\to} SCTP} (یا \lr{\mbox{MTP3b}})
\item
  \textbf{صفحه کاربر}: پروتکل \lr{\mbox{Iu UP}} (قاب‌بندی برای فریم‌های کدک صوتی \lr{\mbox{AMR}})
\item
  \textbf{هدف}: حمل تماس‌های صوتی و تله‌فونی ویدیویی
\end{itemize}

\CNsubsection{}{\lr{\mbox{Iu-PS}} (سوییچینگ بسته‌ای)}

\begin{itemize}
\tightlist
\item
  \textbf{انتقال}: مبتنی بر \lr{\mbox{IP}} (\lr{\mbox{GTP-U}} برای صفحه کاربر)
\item
  \textbf{صفحه کنترل}: \lr{RANAP \ensuremath{\to} SCCP \ensuremath{\to} M3UA \ensuremath{\to} SCTP}
\item
  \textbf{صفحه کاربر}: \lr{\mbox{GTP-U}} (پروتکل تونل‌سازی \lr{\mbox{GPRS}} - صفحه کاربر)
\item
  \textbf{هدف}: حمل تمام داده بسته‌ای (وب، ایمیل، پخش جریانی)
\end{itemize}

\CNsubsection{}{\lr{\mbox{RANAP}} (بخش کاربرد شبکه دسترسی رادیویی)}

رویه‌های کلیدی روی \lr{\mbox{Iu}}:

\begin{itemize}
\tightlist
\item
  \textbf{تخصیص \lr{\mbox{RAB}} (حامل دسترسی رادیویی)}: حامل سرتاسری را برقرار می‌کند
\item
  \textbf{جابه‌جایی}: نقش \lr{\mbox{RNC}} خدمات‌دهنده را به \lr{\mbox{RNC}} دیگری منتقل می‌کند (تحویل سخت)
\item
  \textbf{صفحه‌بندی}: شبکه هسته \lr{\mbox{UE}} را از طریق \lr{\mbox{UTRAN}} صفحه‌بندی می‌کند
\item
  \textbf{حالت امنیتی}: رمزگذاری و حفاظت یکپارچگی را فعال می‌کند
\item
  \textbf{انتقال مستقیم}: پیام‌های \lr{\mbox{NAS}} را به‌صورت شفاف عبور می‌دهد (مثلاً \lr{\mbox{MM}}، \lr{\mbox{CC}}، \lr{\mbox{SM}})
\end{itemize}

\tightlist
\item

\item

\item
  از هر نوع رسانه‌ای پشتیبانی می‌کند (صدا، ویدیو، بازی، \lr{\mbox{IoT}})
\item
  ترکیب خدمات و مَش‌آپ‌ها را ممکن می‌سازد
\end{itemize}

\textbf{\lr{\mbox{CSFB}} (بازگشت به سوییچینگ مداری)}: شبکه‌های \lr{\mbox{LTE}} اولیه برای تماس‌های صوتی به \lr{3G/\allowbreak{}2G} بازمی‌گشتند. این یک راه‌حل انتقالی بود تا زمانی که پوشش \lr{\mbox{VoLTE}} همه‌جاگیر شود.

\end{CNquote}

\CNsection{3.10}{خلاصه: نکات کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\textbf{\lr{3G (UMTS)}}} رادیو \lr{\mbox{WCDMA}} با کانال‌های اختصاصی را معرفی کرد و هسته تقسیم‌شده \lr{CS/\allowbreak{}PS} نسل \lr{\mbox{2G}} را حفظ کرد
\item
  \lr{\mbox{\textbf{\lr{\mbox{UTRAN}}}}} توابع رادیویی را جدا می‌کند: \lr{\mbox{NodeB}} (لایه فیزیکی) و \lr{\mbox{RNC}} (مدیریت منابع رادیویی)
\item
  \textbf{واسط \lr{\mbox{Iu}}} دسترسی رادیویی را به‌طور تمیز از هسته جدا می‌کند \lr{—} که تحول مستقل را ممکن می‌سازد
\item
  \lr{\textbf{\lr{HSPA (R5/\allowbreak{}R6)}}} هوش را به \lr{\mbox{NodeB}} منتقل کرد (زمان‌بندی سریع، \lr{\mbox{HARQ}}، \lr{\mbox{AMC}}) \lr{—} پیش‌نمایشی از فلسفه \lr{\mbox{LTE}}
\item
  \lr{\mbox{\textbf{\lr{\mbox{IMS}}}}} لایه خدمات تمام-\lr{\mbox{IP}} را فراهم می‌کند که در نهایت \lr{\mbox{SS7}} را برای صدا جایگزین می‌کند
\item
  \textbf{\lr{\mbox{LTE}} همه چیز را به بسته یکپارچه کرد} زیرا \lr{\mbox{IP}} به‌اندازه کافی بالغ شده بود تا \lr{\mbox{QoS}} صدا را تضمین کند، و اقتصاد نگهداری زیرساخت دوگانه غیرقابل توجیه شد
\item
  \textbf{تحول تدریجی بود}: هر نسل محدودیت‌های واقعی نسل قبلی را برطرف کرد
\end{enumerate}
