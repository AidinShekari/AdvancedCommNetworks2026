\CNsection{1}{مفهوم صفحه کاربر}

\textbf{صفحه کاربر} (که \textbf{صفحه داده} یا \textbf{صفحه-\lr{\mbox{U}}} نیز نامیده می‌شود) مسیول حمل داده واقعی کاربر از طریق شبکه موبایل است. این شامل موارد زیر است:

\begin{itemize}
\tightlist
\item
  \textbf{بسته‌های \lr{\mbox{IP}}} (مرور وب، داده کاربرد، انتقال فایل)
\item
  \textbf{داده صدا} (فریم‌های گفتار کدگذاری‌شده \lr{VoLTE/\allowbreak{}VoNR})
\item
  \textbf{جریان‌های ویدیو} (ویدیوی بلادرنگ، محتوای پخش جریانی)
\item
  \textbf{هر بار لایه کاربرد}
\end{itemize}

صفحه کاربر از صفحه کنترل متمایز است \lr{—} در حالی که صفحه کنترل اتصالات را برقرار، مدیریت و تخریب می‌کند، صفحه کاربر همان «لوله‌ای» است که داده پس از برقراری آن اتصالات در آن جریان می‌یابد.

\textbf{ویژگی‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  نیازمندی توان عملیاتی بالا
\item
  حساسیت کم به تأخیر
\item
  ارسال بدون حالت (پس از برقراری تونل‌ها)
\item
  مدیریت بسته آگاه به \lr{\mbox{QoS}}
\item
  شفاف نسبت به سیگنالینگ \lr{—} داده را بدون تفسیر حمل می‌کند
\end{itemize}

\CNsection{2}{پروتکل \lr{\mbox{GTP-U}} (پروتکل تونل‌سازی \lr{\mbox{GPRS}} \lr{—} صفحه کاربر)}

\lr{GTP-U (3GPP TS 29.281)} پروتکل تونل‌سازی اصلی است که برای انتقال داده کاربر بین گره‌های شبکه در تمام نسل‌ها (\lr{\mbox{3G}}، \lr{\mbox{4G}}، \lr{\mbox{5G}}) استفاده می‌شود.

\CNsubsection{2.2}{\lr{\mbox{TEID}} (شناسه نقطه انتهایی تونل)}

\lr{\mbox{\textbf{\lr{\mbox{TEID}}}}} یک مقدار 32 بیتی است که یک نقطه انتهایی تونل \lr{\mbox{GTP}} را در گره دریافت‌کننده به‌طور یکتا شناسایی می‌کند:

\begin{itemize}
\tightlist
\item
  هر جهت یک تونل \lr{\mbox{TEID}} خود را دارد
\item
  \textbf{گره دریافت‌کننده} \lr{\mbox{TEID}} را تخصیص می‌دهد (به فرستنده می‌گوید «هنگام ارسال به من از این \lr{\mbox{TEID}} استفاده کن»)
\item
  \lr{\mbox{TEID = 0}} برای پیام‌های اولیه صفحه کنترل رزرو شده است
\item
  یک گره ممکن است هزاران \lr{\mbox{TEID}} فعال به‌طور همزمان داشته باشد (یکی در هر حامل/\allowbreak{}نشست در هر جهت)
\end{itemize}

\textbf{مثال:}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
gNB tells UPF: "For downlink data to UE-X, use TEID = 0x00A1B2C3 on N3"
UPF tells gNB: "For uplink data from UE-X, use TEID = 0x00D4E5F6 on N3"
\end{Verbatim}
\end{CNplaincode}

\CNsubsection{2.3}{کپسوله‌سازی /\allowbreak{} واکپسوله‌سازی}

در هر گام \lr{\mbox{GTP-U}}، بسته \lr{\mbox{IP}} کاربر درون موارد زیر \textbf{کپسوله} می‌شود:

\CNfigure{m23-gtpu-encap}{کپسوله‌سازی \lr{\mbox{GTP-U}} در هر گام}

\begin{itemize}
\tightlist
\item
  \textbf{کپسوله‌سازی} (گره فرستنده): سرآیند \lr{GTP-U + UDP + IP} بیرونی را پیش اضافه می‌کند
\item
  \textbf{واکپسوله‌سازی} (گره دریافت‌کننده): \lr{\mbox{IP}} بیرونی + \lr{\mbox{UDP}} + سرآیند \lr{\mbox{GTP-U}} را حذف می‌کند، \lr{\mbox{IP}} درونی را ارسال می‌کند
\item
  در گره‌های میانی: ممکن است واکپسوله و با \lr{\mbox{TEID}} جدید دوباره کپسوله کند
\end{itemize}

\CNsubsection{2.4}{شماره‌های ترتیب و سرآیندهای افزونه}

\textbf{شماره‌های ترتیب (16 بیتی):}

\begin{itemize}
\tightlist
\item
  اختیاری اما برای بازترتیبی در نقاط انتهایی تونل استفاده می‌شود
\item
  حیاتی در حین تحویل (تضمین عدم اتلاف/\allowbreak{}بازترتیبی بسته)
\item
  در هر \lr{\mbox{PDU}} در هر تونل افزایش می‌یابد
\end{itemize}

\textbf{سرآیندهای افزونه:}

\begin{itemize}
\tightlist
\item
  از طریق فیلد «نوع سرآیند افزونه بعدی» زنجیره می‌شوند
\item
  هر افزونه: \lr{\CNcode{[Length (1 byte)][Content (variable)][Next Ext Hdr Type (1 byte)]}}
\end{itemize}

  ترافیک غیرمنطبق به \lr{\mbox{DN}} مرکزی ادامه می‌یابد
\item
  \textbf{همان آدرس \lr{\mbox{IP UE}}} برای هر دو مسیر (شفاف نسبت به \lr{\mbox{UE}})
\end{itemize}

\CNboxsection{نکات کلیدی}

\begin{CNbox}[unbreakable]{sum}{نکات کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{GTP-U}}}}} تونل جهانی صفحه کاربر است \lr{—} همان پروتکل از \lr{\mbox{3G}} تا \lr{\mbox{5G}}
\item
  \lr{\mbox{\textbf{\lr{\mbox{TEID}}}}} هر نقطه انتهایی تونل را شناسایی می‌کند \lr{—} توسط گیرنده تخصیص می‌یابد
\item
  \textbf{\lr{\mbox{5G}} صفحه کنترل و صفحه کاربر را جدا می‌کند} \lr{(CUPS) — SMF} از طریق \lr{PFCP/\allowbreak{}N4 UPF} را کنترل می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{SDAP}}}}} لایه جدید \lr{\mbox{5G}} است که جریان‌های \lr{\mbox{QoS}} را به حامل‌های رادیویی نگاشت می‌کند
\item
  \textbf{قواعد \lr{UPF (PDR/\allowbreak{}FAR/\allowbreak{}QER/\allowbreak{}URR/\allowbreak{}BAR)}} پردازش بسته قابل برنامه‌ریزی فراهم می‌کنند
\item
  \textbf{رایانش لبه} به‌طور بومی از طریق طبقه‌بند پیوند بالاسو و نقطه انشعاب پشتیبانی می‌شود
\item
  بهینه‌سازی صفحه کاربر (\lr{\mbox{DC}}، چند-\lr{\mbox{UPF}}، لبه) موارد استفاده متنوع \lr{\mbox{5G}} را ممکن می‌سازد
\end{enumerate}

\end{CNbox}
