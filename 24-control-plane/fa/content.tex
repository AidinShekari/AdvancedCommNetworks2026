\CNsection{1}{مفهوم صفحه کنترل}

\textbf{صفحه کنترل} (صفحه-\lr{\mbox{C}}) تمام \textbf{سیگنالینگ} در شبکه موبایل را مدیریت می‌کند \lr{—} پیام‌هایی که اتصالات را برقرار، مدیریت، اصلاح و تخریب می‌کنند. این صفحه داده کاربر را حمل \textbf{نمی‌کند}.

\textbf{مسیولیت‌های صفحه کنترل:}

\begin{itemize}
\tightlist
\item
  \textbf{احراز هویت و امنیت}: تأیید هویت مشترک، برقراری رمزنگاری
\item
  \textbf{ثبت‌نام و تحرک}: اتصال، جداسازی، به‌روزرسانی نواحی ردیابی، تحویل‌ها
\item
  \textbf{مدیریت نشست}: برقراری/\allowbreak{}اصلاح/\allowbreak{}آزادسازی نشست‌های \lr{\mbox{PDU}} یا حامل‌ها
\item
  \textbf{مذاکره \lr{\mbox{QoS}}}: توافق بر پارامترهای کیفیت برای جریان‌های داده
\item
  \textbf{صفحه‌بندی}: یافتن \lr{\mbox{UE}}های بیکار برای داده/\allowbreak{}تماس‌های ورودی
\item
  \textbf{سیاست و صورت‌حساب}: اعمال قواعد اپراتور بر نشست‌ها
\end{itemize}

\textbf{اصل کلیدی:} صفحه کنترل صفحه کاربر را «برنامه‌ریزی» می‌کند \lr{—} تونل‌ها، قواعد و مسیریابی را برقرار می‌کند که صفحه کاربر سپس برای ارسال داده از آن‌ها استفاده می‌کند.

\CNboxsection{نکات کلیدی}

\begin{CNbox}[unbreakable]{sum}{نکات کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{صفحه کنترل = مغز سیگنالینگ}: تونل‌ها را راه‌اندازی می‌کند، کاربران را احراز هویت می‌کند، تحرک را مدیریت می‌کند
\item
  \textbf{\lr{\mbox{NAS}} سرتاسری است (\lr{\mbox{UE}}\ensuremath{\leftrightarrow}هسته)} \lr{—} شفاف نسبت به \lr{\mbox{RAN}}؛ زمینه امنیتی جدا
\item
  \textbf{\lr{\mbox{RRC}} رادیو را مدیریت می‌کند} \lr{—} راه‌اندازی اتصال، اندازه‌گیری‌ها، فرمان‌های تحویل
\item
  \textbf{\lr{\mbox{5G}} حالت \lr{RRC\_\allowbreak{}INACTIVE} را اضافه می‌کند} برای بازگشت سریع و کارایی \lr{\mbox{IoT}}
\item
  \textbf{\lr{\mbox{NGAP}} جایگزین \lr{\mbox{S1AP}} شد} \lr{—} از شبکه‌بندی، جریان‌های \lr{\mbox{QoS}} و سه حالت \lr{\mbox{RRC}} پشتیبانی می‌کند
\item
  \textbf{\lr{\mbox{GTP-C}} تحول یافت}: \lr{v1 (3G) \ensuremath{\leftarrow} v2 (4G)} \ensuremath{\leftarrow} جایگزین‌شده با \lr{PFCP+HTTP/\allowbreak{}2} در \lr{\mbox{5G}}
\item
  \textbf{\lr{HTTP/\allowbreak{}2 SBI} انقلابی است} \lr{—} ارتباط \lr{\mbox{NF}} بومی-ابر، \lr{\mbox{RESTful}}، خود-مقیاس‌پذیر
\item
  \textbf{تحول سیگنالینگ}: \lr{\mbox{SS7}} (دودویی، \lr{\mbox{TDM}}) \ensuremath{\to} \lr{\mbox{Diameter}} (دودویی، \lr{\mbox{IP}}) \ensuremath{\to} \lr{HTTP/\allowbreak{}2} (\lr{\mbox{JSON}}، بومی-ابر)
\end{enumerate}

\end{CNbox}
