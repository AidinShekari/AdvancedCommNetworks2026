\CNsection{9.3}{توابع شبکه \lr{\mbox{EPC}}}

\CNchained\CNsubsection{9.3.1}{\lr{\mbox{MME}} \lr{—} موجودیت مدیریت تحرک}

\CNchained\CNsubsubsection{هدف}

\lr{\mbox{MME}} \textbf{گره اصلی صفحه کنترل} در \lr{\mbox{EPC}} است. تمام سیگنالینگ بین \lr{\mbox{UE}} و شبکه هسته را مدیریت می‌کند و تحرک، امنیت و برقراری نشست را کنترل می‌کند. \lr{\mbox{MME}} هرگز به داده صفحه کاربر دست نمی‌زند.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{سیگنالینگ \lr{\mbox{NAS}}}: پروتکل‌های لایه غیر-دسترسی (\lr{\mbox{EMM}} و \lr{\mbox{ESM}}) را با \lr{\mbox{UE}} پایان می‌دهد
\item
  \textbf{احراز هویت و امنیت}: \lr{\mbox{EPS-AKA}} را آغاز می‌کند، کلیدها را مشتق می‌کند \lr{(\ensuremath{K_{\mathrm{ASME}}} \ensuremath{\to} \ensuremath{K_{\mathrm{eNB}}})}، الگوریتم‌های امنیت \lr{\mbox{NAS}} را انتخاب می‌کند
\item
  \textbf{مدیریت تحرک}: مکان \lr{\mbox{UE}} را ردیابی می‌کند \lr{\mbox{(TAI)}}، حالت‌های \lr{\mbox{ECM}} را مدیریت می‌کند، سیگنالینگ تحویل را انجام می‌دهد
\item
  \textbf{مدیریت حامل}: حامل‌های \lr{\mbox{EPS}} را از طریق سیگنالینگ \lr{\mbox{S11}} به \lr{\mbox{S-GW}} ایجاد/\allowbreak{}اصلاح/\allowbreak{}حذف می‌کند
\item
  \textbf{صفحه‌بندی}: \lr{\mbox{UE}}ها را در حالت \lr{\mbox{ECM-IDLE}} هنگام رسیدن داده پیوند پایین‌سو صفحه‌بندی می‌کند
\item
  \textbf{انتخاب و استخر \lr{\mbox{MME}}}: چندین \lr{\mbox{MME}} به یک ناحیه استخر خدمات می‌دهند؛ توازن بار از طریق وزن‌دهی \lr{\mbox{S1AP}}
\item
  \textbf{تحرک بین-\lr{\mbox{RAT}}}: تحویل به/\allowbreak{}از \lr{2G/\allowbreak{}3G} را از طریق واسط \lr{\mbox{S3}} به \lr{\mbox{SGSN}} انجام می‌دهد
\item
  \textbf{رهگیری قانونی}: سیگنالینگ برای محرک‌های \lr{\mbox{LI}} فراهم می‌کند
\end{enumerate}

\end{itemize}

\CNsubsection{9.3.2}{\lr{\mbox{HSS}} \lr{—} سرور مشترک خانگی}

\CNchained\CNsubsubsection{هدف}

\lr{\mbox{HSS}} \textbf{پایگاه داده مرکزی مشترکین} برای \lr{\mbox{EPC}} است. این تحول \lr{HLR/\allowbreak{}AuC} از نسل \lr{2G/\allowbreak{}3G} است که تمام اطلاعات دایمی مشترک، اعتبارنامه‌های احراز هویت و پروفایل‌های خدمات را ذخیره می‌کند.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{پایگاه داده مشترکین}: \lr{\mbox{IMSI}}، \lr{\mbox{MSISDN}}، اشتراک‌های خدمات، پیکربندی‌های \lr{\mbox{APN}} را ذخیره می‌کند
\item
  \textbf{تولید بردار احراز هویت}: بردارهای احراز هویت \lr{\mbox{EPS}} (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}}، \lr{\mbox{XRES}}، \lr{\mbox{\ensuremath{K_{\mathrm{ASME}}}}}) را با استفاده از کلید دایمی \lr{\mbox{K}} و شماره ترتیب \lr{\mbox{SQN}} تولید می‌کند
\item
  \textbf{مدیریت مکان}: آدرس \lr{\mbox{MME}} فعلی را برای هر مشترک نگه می‌دارد؛ ثبت‌نام/\allowbreak{}لغو ثبت‌نام را انجام می‌دهد
\item
  \textbf{پروفایل‌های اشتراک}: \lr{\mbox{APN}}های مجاز، پروفایل‌های \lr{\mbox{QoS}} (\lr{\mbox{AMBR}} مشترک‌شده، \lr{\mbox{QCI}} پیش‌فرض)، محدودیت‌های رومینگ را ذخیره می‌کند
\item
  \textbf{ویژگی‌های صورت‌حساب}: پیکربندی صورت‌حساب در هر مشترک
\item
  \textbf{مجوزدهی خدمات}: تعیین می‌کند \lr{\mbox{UE}} مجاز به کدام خدمات است
\end{enumerate}

\CNsubsection{9.3.3}{\lr{\mbox{S-GW}} \lr{—} دروازه خدمات‌دهنده}

\CNchained\CNsubsubsection{هدف}

\lr{\mbox{S-GW}} \textbf{لنگر صفحه کاربر} بین شبکه رادیویی و هسته است. به‌عنوان لنگر تحرک محلی برای تحویل‌های بین-\lr{\mbox{eNB}} عمل می‌کند و تداوم صفحه کاربر را در حین جابه‌جایی \lr{\mbox{UE}} بین ایستگاه‌های پایه تضمین می‌کند.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{ارسال صفحه کاربر}: داده کاربر را بین \lr{\mbox{eNB}} و \lr{\mbox{P-GW}} از طریق تونل‌های \lr{\mbox{GTP-U}} مسیریابی می‌کند
\item
  \textbf{لنگر تحرک محلی}: در حین تحویل \lr{\mbox{X2}}، \lr{\mbox{S-GW}} بدون تغییر می‌ماند؛ فقط نقطه انتهایی \lr{\mbox{GTP-U}} پیوند پایین‌سو به \lr{\mbox{eNB}} مقصد تغییر می‌کند
\item
  \textbf{بافرینگ داده پیوند پایین‌سو}: وقتی \lr{\mbox{UE}} در \lr{\mbox{ECM-IDLE}} است، \lr{\mbox{S-GW}} بسته‌های ورودی را بافر می‌کند و \lr{\mbox{MME}} را برای صفحه‌بندی \lr{\mbox{UE}} فعال می‌کند
\item
  \textbf{صورت‌حساب در هر \lr{\mbox{UE}}}: داده صورت‌حساب (\lr{\mbox{CDR}}ها) را در هر حامل جمع‌آوری و به سیستم صورت‌حساب گزارش می‌کند
\item
  \textbf{رهگیری قانونی}: نقطه کپی صفحه کاربر برای رهگیری قانونی
\item
  \textbf{لنگر بین-\lr{\mbox{RAT}}}: صفحه کاربر را در حین تحویل بین \lr{\mbox{LTE}} و \lr{2G/\allowbreak{}3G} لنگر می‌کند (با \lr{\mbox{SGSN}})
\item
  \textbf{علامت‌گذاری سطح انتقال}: علامت‌گذاری \lr{\mbox{DSCP}} بر اساس \lr{\mbox{QCI}} حامل اعمال می‌کند
\end{enumerate}

\begin{itemize}

\item
  \lr{\mbox{\textbf{\lr{\mbox{GTPv2-C}}}}} (پورت \lr{\mbox{UDP 2123}}): پیام‌های \lr{Create/\allowbreak{}Modify/\allowbreak{}Delete Session} از \lr{\mbox{MME}}
\item
  \lr{\mbox{\textbf{\lr{\mbox{DSCP}}}}}: علامت‌گذاری سرآیند \lr{\mbox{IP}} بیرونی برای \lr{\mbox{QoS}} انتقال
\end{itemize}

\CNsubsubsection{تأثیر شکست}

\begin{itemize}
\tightlist
\item
  \textbf{نشست‌های فعال مختل می‌شوند}: جریان‌های داده در جریان قطع می‌شوند
\item
  \textbf{تغییر مسیر لازم است}: \lr{\mbox{MME}} شکست را تشخیص می‌دهد، \lr{\mbox{S-GW}} جدید انتخاب می‌کند، تونل‌های \lr{\mbox{GTP}} را دوباره برقرار می‌کند
\item
  \textbf{داده بافر‌شده از دست می‌رود}: هر بسته \lr{\mbox{DL}} بافر‌شده برای \lr{\mbox{UE}}های \lr{\mbox{IDLE}} از دست می‌رود
\item
  \textbf{کاهش اثر}: \lr{\mbox{S-GW}} معمولاً با افزونگی \lr{\mbox{N+1}} مستقر می‌شود؛ \lr{\mbox{ICGW}} (کاهش سیگنالینگ حالت بیکار) حالت را کمینه می‌کند
\end{itemize}

\CNsubsection{9.3.4}{\lr{\mbox{P-GW}} \lr{—} دروازه \lr{\mbox{PDN}}}

\CNchained\CNsubsubsection{هدف}

\lr{\mbox{P-GW}} \textbf{نقطه اتصال متقابل} بین \lr{\mbox{EPC}} و شبکه‌های داده بسته‌ای خارجی (\lr{\mbox{PDN}}ها) است. این نقطه لنگر \lr{\mbox{IP}} برای نشست‌های \lr{\mbox{UE}} است \lr{—} آدرس \lr{\mbox{IP}} کاربر تا زمانی که اتصال \lr{\mbox{PDN}} وجود دارد ثابت می‌ماند، بدون توجه به تحرک.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{تخصیص آدرس \lr{\mbox{IP}}}:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{IPv4}}: \lr{\mbox{DHCPv4}} یا تخصیص ایستا از استخرهای پیکربندی‌شده
  \item
    \lr{\mbox{IPv6}}: پیکربندی خودکار آدرس بدون حالت \lr{\mbox{(SLAAC)}} از طریق اعلام مسیریاب؛ تخصیص پیشوند /\allowbreak{}\lr{\mbox{64}}
  \item
    دو-پشته: هر دو به‌طور همزمان
  \end{itemize}
\item
  \textbf{اجرای سیاست \lr{\mbox{(PCEF)}}}:

  \begin{itemize}
  \tightlist
  \item
    قواعد \lr{\mbox{PCC}} دریافتی از \lr{\mbox{PCRF}} را اعمال می‌کند
  \item
    قالب‌های جریان ترافیک \lr{\mbox{(TFT)}}: فیلترهای بسته که جریان‌های \lr{\mbox{IP}} را به حامل‌ها نگاشت می‌کنند
  \item
    فیلترهای جریان داده خدمات \lr{\mbox{(SDF)}}: تطبیق 5-گانه برای \lr{\mbox{QoS}}/\allowbreak{}صورت‌حساب
  \end{itemize}
\item
  \textbf{صورت‌حساب}:

  \begin{itemize}
  \tightlist
  \item
    آفلاین (واسط \lr{\mbox{Gz}} \ensuremath{\leftarrow} تولید \lr{\mbox{CDR}})
  \item
    آنلاین (واسط \lr{\mbox{Gy}} \ensuremath{\leftarrow} کنترل اعتبار بلادرنگ با \lr{\mbox{OCS}})
  \end{itemize}
\item
  \textbf{واسط \lr{\mbox{SGi}}}: اتصال به شبکه‌های خارجی (اینترنت، \lr{\mbox{IMS}}، \lr{\mbox{VPN}} سازمانی)
\item
  \lr{\textbf{\lr{NAT/\allowbreak{}NAPT}}}: ترجمه آدرس شبکه اختیاری برای \lr{\mbox{IPv4}}
\item
  \textbf{لنگرگذاری \lr{\mbox{GTP}}}: تونل‌های \lr{\mbox{GTP}} در \lr{S5/\allowbreak{}S8} از \lr{\mbox{S-GW}} را پایان می‌دهد
\end{enumerate}

\CNsubsection{9.3.5}{\lr{\mbox{PCRF}} \lr{—} تابع قواعد سیاست و صورت‌حساب}

\CNchained\CNsubsubsection{هدف}

\lr{\mbox{PCRF}} \textbf{نقطه تصمیم سیاست} در معماری \lr{\mbox{PCC}} است. تصمیمات بلادرنگ درباره مجوزدهی \lr{\mbox{QoS}}، دروازه‌بندی و قواعد صورت‌حساب بر اساس پروفایل مشترک، شرایط شبکه و نیازمندی‌های کاربرد می‌گیرد.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{تولید قاعده \lr{\mbox{PCC}}}: قواعد کنترل سیاست و صورت‌حساب شامل موارد زیر ایجاد می‌کند:

  \begin{itemize}
  \tightlist
  \item
    فیلترهای \lr{\mbox{SDF}} (کدام ترافیک)
  \item
    پارامترهای \lr{\mbox{QoS}} (\lr{\mbox{QCI}}، \lr{\mbox{GBR}}، \lr{\mbox{MBR}})
  \item
    کلیدها و روش‌های صورت‌حساب
  \item
    وضعیت دروازه (باز/\allowbreak{}بسته)
  \end{itemize}
\item
  \textbf{مجوزدهی \lr{\mbox{QoS}}}: \lr{\mbox{QoS}} درخواستی را در برابر اشتراک و سیاست شبکه اعتبارسنجی می‌کند
\item
  \textbf{سیاست پویا}: تغییرات سیاست بلادرنگ (مثلاً وقتی تماس \lr{\mbox{VoLTE}} شروع می‌شود \ensuremath{\leftarrow} نصب قاعده حامل \lr{\mbox{GBR}})
\item
  \textbf{پردازش واسط \lr{\mbox{Rx}}}: اطلاعات نشست را از \lr{AF/\allowbreak{}IMS} (توصیف رسانه، پهنای باند) دریافت و به قواعد \lr{\mbox{PCC}} ترجمه می‌کند
\item
  \textbf{محدودیت‌های هزینه}: با \lr{\mbox{OCS}} برای تصمیمات سیاست بر اساس اعتبار باقیمانده ارتباط برقرار می‌کند
\item
  \textbf{سیاست مبتنی بر اشتراک}: قواعد بر اساس لایه مشترک اعمال می‌کند (مثلاً طرح‌های طلایی/\allowbreak{}نقره‌ای/\allowbreak{}برنزی)
\end{enumerate}

\CNsubsection{9.3.6}{\lr{\mbox{PCEF}} \lr{—} تابع اجرای سیاست و صورت‌حساب}

\CNchained\CNsubsubsection{هدف}

\lr{\mbox{PCEF}} \textbf{نقطه اجرای سیاست} است که هم‌مکان با \lr{\mbox{P-GW}} قرار دارد. قواعد \lr{\mbox{PCC}} دریافتی از \lr{\mbox{PCRF}} را با اعمال فیلترکردن بسته، اجرای \lr{\mbox{QoS}} و اقدامات صورت‌حساب روی هر بسته صفحه کاربر اجرا می‌کند.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{اجرای فیلتر \lr{\mbox{SDF}}}: بسته‌های \lr{\mbox{IP}} ورودی/\allowbreak{}خروجی را با فیلترهای جریان داده خدمات تطبیق می‌دهد (5-گانه: \lr{\mbox{IP}} مبدأ، \lr{\mbox{IP}} مقصد، پورت مبدأ، پورت مقصد، پروتکل)
\item
  \textbf{کنترل دروازه}: دروازه را برای \lr{\mbox{SDF}}های مشخص باز یا بسته می‌کند (مسدودکردن/\allowbreak{}اجازه ترافیک)
\item
  \textbf{اجرای \lr{\mbox{QoS}}}: محدودسازی نرخ در هر جریان را اعمال می‌کند:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{GBR}} (نرخ بیت تضمین‌شده): نرخ حداقلی تضمین‌شده
  \item
    \lr{\mbox{MBR}} (حداکثر نرخ بیت): سقف سخت در هر حامل
  \item
    \lr{\mbox{AMBR}} (حداکثر نرخ بیت تجمیعی): سقف در سراسر تمام حامل‌های غیر-\lr{\mbox{GBR}}
  \end{itemize}
\item
  \textbf{اتصال حامل}: فیلترهای \lr{\mbox{SDF}} را به حامل‌های \lr{\mbox{EPS}} مناسب نگاشت می‌کند
\item
  \textbf{اجرای صورت‌حساب}:

  \begin{itemize}
  \tightlist
  \item
    کلیدهای صورت‌حساب را به ترافیک اعمال می‌کند
  \item
    اندازه‌گیری حجم/\allowbreak{}زمان در هر گروه نرخ‌گذاری
  \item
    صورت‌حساب آنلاین: مصرف را گزارش می‌دهد، محدودیت‌های اعتبار را اجرا می‌کند
  \item
    صورت‌حساب آفلاین: \lr{\mbox{CDR}}ها را تولید می‌کند
  \end{itemize}
\item
  \textbf{پایش مصرف}: آستانه‌های حجم را به \lr{\mbox{PCRF}} برای تصمیمات سیاست گزارش می‌دهد
\end{enumerate}

\CNsection{9.4}{حامل‌های \lr{\mbox{EPC}}}

\CNchained\CNsubsection{}{مفهوم حامل}

یک حامل \lr{\mbox{EPS}} یک \textbf{کانال منطقی سرتاسری} بین \lr{\mbox{UE}} و \lr{\mbox{P-GW}} با ویژگی‌های \lr{\mbox{QoS}} مشخص است. هر حامل توسط سه مؤلفه تعریف می‌شود:

\CNdisplay{\text{\rl{حامل}} = \{\,\mathrm{TFT} + \text{\rl{پارامترهای QoS}} + \text{\rl{تونل(های) GTP}}\,\}}

\begin{itemize}
\tightlist
\item
  \textbf{\lr{\mbox{TFT}} (قالب جریان ترافیک)}: فیلترهای بسته (5-گانه) که طبقه‌بندی می‌کنند کدام بسته‌های \lr{\mbox{IP}} به این حامل تعلق دارند
\item
  \textbf{پارامترهای \lr{\mbox{QoS}}}: \lr{\mbox{QCI}}، \lr{\mbox{ARP}}، \lr{GBR/\allowbreak{}MBR} (برای حامل‌های \lr{\mbox{GBR}})، \lr{\mbox{AMBR}} (تجمیعی)
\item
  \textbf{تونل \lr{\mbox{GTP}}}: شناسایی‌شده با \lr{\mbox{TEID}} (شناسه نقطه انتهایی تونل) در هر گام:

  \begin{itemize}
  \tightlist
  \item
    \lr{S1-U: eNB \ensuremath{\leftrightarrow} S-GW}
  \item
    \lr{S5/\allowbreak{}S8: S-GW \ensuremath{\leftrightarrow} P-GW}
  \end{itemize}
\end{itemize}

  \textbf{همیشه فعال}: برای کل مدت اتصال \lr{\mbox{PDN}} وجود دارد
\item
  \textbf{غیر-\lr{\mbox{GBR}}}: بدون نرخ بیت تضمین‌شده؛ بهترین تلاش با سقف \lr{\mbox{AMBR}}
\item
  \textbf{\lr{\mbox{TFT}} تطبیق-همه}: تمام ترافیکی را حمل می‌کند که با \lr{\mbox{TFT}} هیچ حامل اختصاصی مطابقت ندارد
\item
  \textbf{\lr{\mbox{QoS}} از اشتراک}: مقادیر \lr{\mbox{QCI}} و \lr{\mbox{AMBR}} از پروفایل اشتراک \lr{\mbox{HSS}} می‌آیند
\item
  \textbf{یکی در هر اتصال \lr{\mbox{PDN}}}: هر \lr{\mbox{APN}} دقیقاً یک حامل پیش‌فرض می‌گیرد (\lr{\mbox{EBI}} = شناسه حامل \lr{\mbox{EPS}} تخصیص‌یافته)
\end{itemize}

\CNsubsection{9.4.2}{حامل اختصاصی}

\begin{itemize}
\tightlist
\item
  \textbf{در صورت نیاز ایجاد می‌شود}: با محرک درخواست خدمات (قاعده \lr{\mbox{PCRF}}) یا درخواست \lr{\mbox{UE}}/\allowbreak{}شبکه
\item
  \textbf{\lr{\mbox{QoS}} مشخص}: می‌تواند \lr{\mbox{GBR}} (تضمین‌شده) یا غیر-\lr{\mbox{GBR}} (بهترین تلاش اولویت‌دار) باشد
\item
  \textbf{\lr{\mbox{TFT}} مشخص}: فیلترهای بسته ترافیک منطبق را به این حامل هدایت می‌کنند
\item
  \textbf{مرتبط با حامل پیش‌فرض}: همان اتصال \lr{\mbox{PDN}} را به اشتراک می‌گذارد (همان آدرس \lr{\mbox{IP}})
\item
  \textbf{چرخه عمر}: از طریق تخصیص منابع حامل \ensuremath{\leftarrow} تصمیم \lr{PCRF \ensuremath{\to} P-GW} فعال‌سازی حامل اختصاصی را آغاز می‌کند
\item
  \textbf{محرک‌های نمونه}: شروع تماس \lr{VoLTE (QCI 1)}، پخش جریانی ویدیو \lr{\mbox{(QCI 4)}}، بازی \lr{\mbox{(QCI 3)}}
\end{itemize}

\CNsubsection{9.5.2}{مشتق‌سازی \lr{\mbox{\ensuremath{K_{\mathrm{ASME}}}}}}

\lr{\mbox{\textbf{\lr{\mbox{\ensuremath{K_{\mathrm{ASME}}}}}}}} (کلید برای موجودیت مدیریت امنیت دسترسی) کلید اصلی برای تمام امنیت \lr{\mbox{EPC}} است:

\CNdisplay{K_{\mathrm{ASME}} = \mathrm{KDF}\left(\mathrm{CK} \,\|\, \mathrm{IK},\ \CNmtext{SN-ID},\ \mathrm{SQN} \oplus \mathrm{AK}\right)}

که در آن:

\begin{itemize}
\tightlist
\item
  \lr{\mbox{KDF}} = تابع استخراج کلید مبتنی بر \lr{HMAC-SHA-256}
\item
  \lr{\mbox{CK}} = کلید رمز = \lr{f3(K, RAND)}
\item
  \lr{\mbox{IK}} = کلید یکپارچگی = \lr{f4(K, RAND)}
\item
  \lr{\mbox{SN-ID}} = شناسه شبکه خدمت‌دهنده \lr{(MCC \ensuremath{\|} MNC)}
\item
  \lr{\mbox{SQN}} = شماره توالی
\item
  \lr{\mbox{AK}} = کلید گمنامی = \lr{f5(K, RAND)}
\end{itemize}

\CNchained\CNsubsection{9.6.1}{اتصال \lr{\mbox{PDN}} (برقراری حامل پیش‌فرض)}

اتصال \lr{\mbox{PDN}} \textbf{حامل پیش‌فرض \lr{\mbox{EPS}}} را ایجاد می‌کند و یک آدرس \lr{\mbox{IP}} به \lr{\mbox{UE}} تخصیص می‌دهد.

\CNsubsubsection{جریان رویه}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\textbf{\lr{\mbox{UE \ensuremath{\to} MME}}}: ESM PDN Connectivity Request} (\lr{\mbox{APN}}، نوع \lr{\mbox{PDN}}: \lr{IPv4/\allowbreak{}IPv6/\allowbreak{}IPv4v6})
\item
  \lr{\textbf{\lr{MME \ensuremath{\to} S-GW}}: Create Session Request} (\lr{\mbox{IMSI}}، \lr{\mbox{QoS}} حامل از پروفایل \lr{\mbox{HSS}}، \lr{\mbox{APN}})
\item
  \lr{\textbf{\lr{S-GW \ensuremath{\to} P-GW}}: Create Session Request} (با \lr{\mbox{F-TEID}} مربوط به \lr{\mbox{S-GW}} ارسال می‌شود)
\item
  \lr{\mbox{\textbf{\lr{\mbox{P-GW}}}}}: آدرس \lr{\mbox{IP}} را تخصیص می‌دهد، با \lr{\mbox{PCRF}} تماس می‌گیرد (\lr{CCR-Initial} روی \lr{\mbox{Gx}})، قواعد \lr{\mbox{PCC}} پیش‌فرض را نصب می‌کند
\item
  \lr{\textbf{\lr{P-GW \ensuremath{\to} S-GW}}: Create Session Response} (آدرس \lr{\mbox{IP}}، زمینه حامل، \lr{\mbox{TFT}})
\item
  \lr{\textbf{\lr{S-GW \ensuremath{\to} MME}}: Create Session Response} (\lr{\mbox{F-TEID}} مربوط به \lr{\mbox{S-GW}}، زمینه حامل)
\item
  \lr{\textbf{\lr{MME \ensuremath{\to} eNB}}: Initial Context Setup Request} (زمینه حامل، \lr{\mbox{\ensuremath{K_{\mathrm{eNB}}}}}، \lr{\mbox{NAS}}: \lr{Attach Accept + Activate Default Bearer})
\item
  \lr{\textbf{\lr{\mbox{eNB \ensuremath{\to} UE}}}: RRC Connection Reconfiguration} (برقراری \lr{\mbox{DRB}}) + پیام‌های \lr{\mbox{NAS}}
\item
  \lr{\textbf{\lr{UE \ensuremath{\to} eNB \ensuremath{\to} MME}}: Attach Complete /\allowbreak{} Activate Default EPS Bearer Context Accept}
\item
  \lr{\textbf{\lr{MME \ensuremath{\to} S-GW}}: Modify Bearer Request} (\lr{\mbox{F-TEID}} مربوط به \lr{\mbox{eNB}} برای پیوند پایین‌سو)
\end{enumerate}

\textbf{نتیجه}: حامل سرتاسری برقرار می‌شود: \lr{UE \ensuremath{\leftrightarrow} eNB \ensuremath{\leftrightarrow} S-GW \ensuremath{\leftrightarrow} P-GW} \ensuremath{\leftrightarrow} اینترنت

\CNsubsection{9.6.2}{فعال‌سازی حامل اختصاصی}

حامل‌های اختصاصی \textbf{با شروع شبکه} فعال می‌شوند (با محرک قاعده \lr{\mbox{PCRF}} یا سیاست اپراتور):

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\textbf{\lr{PCRF \ensuremath{\to} P-GW}}: Re-Auth-Request (RAR)} با قاعده \lr{\mbox{PCC}} (\lr{\mbox{QCI}}، \lr{\mbox{GBR}}، فیلترهای \lr{\mbox{TFT}})
\item
  \lr{\textbf{\lr{P-GW \ensuremath{\to} S-GW}}: Create Bearer Request} (\lr{\mbox{EBI}}، \lr{\mbox{TFT}}، \lr{\mbox{QoS}}، \lr{\mbox{F-TEID}} مربوط به \lr{\mbox{S5}})
\item
  \lr{\textbf{\lr{S-GW \ensuremath{\to} MME}}: Create Bearer Request} (با \lr{\mbox{F-TEID}} مربوط به \lr{\mbox{S1-U}} ارسال می‌شود)
\item
  \lr{\textbf{\lr{MME \ensuremath{\to} eNB \ensuremath{\to} UE}}: Bearer Setup Request (NAS: Activate Dedicated Bearer Context Request)}
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}}: می‌پذیرد، \lr{\mbox{TFT}} را برای طبقه‌بندی پیوند بالاسو ذخیره می‌کند
\item
  \lr{\textbf{\lr{UE \ensuremath{\to} eNB \ensuremath{\to} MME}}: Activate Dedicated EPS Bearer Context Accept}
\item
  \lr{\textbf{\lr{MME \ensuremath{\to} S-GW \ensuremath{\to} P-GW}}: Create Bearer Response} (اکنون تمام \lr{\mbox{TEID}}ها متصل شده‌اند)
\end{enumerate}

\CNsubsection{9.6.3}{اصلاح حامل}

اصلاح حامل \lr{\mbox{QoS}} یا \lr{\mbox{TFT}} یک حامل موجود را تغییر می‌دهد:

\begin{itemize}
\tightlist
\item
  \textbf{با شروع شبکه}: \lr{\mbox{PCRF}} قاعده \lr{\mbox{PCC}} را به‌روزرسانی می‌کند \ensuremath{\leftarrow} \lr{\mbox{P-GW}} یک \lr{Update Bearer Request} می‌فرستد
\item
  \textbf{با شروع \lr{\mbox{UE}}}: \lr{\mbox{UE}} یک \lr{Bearer Resource Modification Request} به \lr{\mbox{MME}} می‌فرستد
\item
  \textbf{مثال‌ها}: ارتقای کیفیت ویدیو (افزایش \lr{\mbox{GBR}})، افزودن فیلتر \lr{\mbox{SDF}} جدید، تغییر وضعیت دروازه
\end{itemize}

\CNsubsection{9.6.4}{غیرفعال‌سازی حامل}

\begin{itemize}
\tightlist
\item
  \textbf{با شروع شبکه}: \lr{\mbox{PCRF}} قاعده \lr{\mbox{PCC}} را حذف می‌کند \ensuremath{\leftarrow} \lr{\mbox{P-GW}} یک \lr{Delete Bearer Request} می‌فرستد
\item
  \textbf{با شروع \lr{\mbox{UE}}}: \lr{\mbox{UE}} یک \lr{Deactivate EPS Bearer Context Request} می‌فرستد
\item
  \textbf{غیرفعال‌سازی حامل پیش‌فرض}: تمام حامل‌های اختصاصی مرتبط را حذف + آدرس \lr{\mbox{IP}} را آزاد می‌کند
\item
  \textbf{قطع اتصال \lr{\mbox{PDN}}}: زمانی فعال می‌شود که آخرین حامل پیش‌فرض غیرفعال شود
\end{itemize}

\CNsubsection{9.7.2}{به‌روزرسانی ناحیه ردیابی \lr{\mbox{(TAU)}}}

\lr{\mbox{TAU}} شبکه را از مکان تقریبی \lr{\mbox{UE}} در حالت \lr{\mbox{ECM-IDLE}} مطلع نگه می‌دارد:

\textbf{زمانی که \lr{\mbox{TAU}} فعال می‌شود:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} وارد ناحیه ردیابی \lr{\mbox{(TA)}} جدیدی می‌شود که در فهرست \lr{\mbox{TAI}} آن نیست
\item
  تایمر \lr{\mbox{TAU}} دوره‌ای منقضی می‌شود (قابل پیکربندی، معمولاً 54 دقیقه)
\item
  پس از فعال‌سازی \lr{\mbox{ISR}} (کاهش سیگنالینگ حالت بیکار)
\end{itemize}

\textbf{رویه \lr{\mbox{TAU}}:}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{UE \ensuremath{\to} MME}}: \lr{Tracking Area Update Request} (\lr{\mbox{GUTI}} قدیمی، \lr{\mbox{TAI}} بازدیدشده)
\item
  \lr{\mbox{MME}}: هویت را تأیید می‌کند، به‌صورت اختیاری احراز هویت می‌کند
\item
  \lr{MME \ensuremath{\to} HSS}: \lr{Update Location} (اگر \lr{\mbox{MME}} جدید باشد)
\item
  \lr{MME \ensuremath{\to} MME} قدیمی: انتقال زمینه (اگر \lr{\mbox{MME}} تغییر کرده باشد)
\item
  \lr{\mbox{MME \ensuremath{\to} UE}}: \lr{TAU Accept} (\lr{\mbox{GUTI}} جدید، فهرست \lr{\mbox{TAI}} جدید)
\item
  \lr{MME \ensuremath{\to} S-GW}: \lr{Modify Bearer} (اگر \lr{\mbox{S-GW}} تغییر کرده باشد)
\end{enumerate}

\textbf{فهرست \lr{\mbox{TAI}}}: \lr{\mbox{MME}} یک فهرست از \lr{\mbox{TA}}ها تخصیص می‌دهد تا فراوانی \lr{\mbox{TAU}} را کاهش دهد (مبادله توازن بار در برابر هزینه صفحه‌بندی)

\CNsubsection{9.7.3}{تحویل \lr{\mbox{X2}} (درون-\lr{\mbox{LTE}}، بدون تغییر \lr{\mbox{MME}})}

\textbf{سریع‌ترین تحویل} در \lr{\mbox{LTE}} \lr{—} سیگنالینگ مستقیم \lr{\mbox{eNB}}-به-\lr{\mbox{eNB}} بدون دخالت \lr{\mbox{MME}} برای ارسال داده:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{\lr{\mbox{eNB}} مبدأ}: گزارش‌های اندازه‌گیری نشان می‌دهند سلول مقصد بهتر است
\item
  \textbf{مبدأ \ensuremath{\leftarrow} \lr{\mbox{eNB}} مقصد} \lr{\mbox{(X2)}}: \lr{Handover Request} (زمینه \lr{\mbox{UE}}، حامل‌ها، \lr{\mbox{QoS}})
\item
  \textbf{\lr{\mbox{eNB}} مقصد}: \lr{\mbox{UE}} را می‌پذیرد، منابع را تخصیص می‌دهد
\item
  \textbf{مقصد \ensuremath{\leftarrow} \lr{\mbox{eNB}} مبدأ} \lr{\mbox{(X2)}}: \lr{Handover Request Acknowledge} (آدرس ارسال \lr{\mbox{DL}})
\item
  \textbf{\lr{\mbox{eNB}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{UE}}}: \lr{RRC Connection Reconfiguration} (فرمان تحویل)
\item
  \textbf{\lr{\mbox{eNB}} مبدأ}: شروع ارسال داده بافر‌شده + داده \lr{\mbox{DL}} جدید به \lr{\mbox{eNB}} مقصد از طریق \lr{\mbox{X2}}
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}}: از مبدأ جدا می‌شود، با سلول مقصد همگام می‌شود
\item
  \textbf{\lr{\mbox{UE \ensuremath{\to} eNB}} مقصد}: \lr{RRC Connection Reconfiguration Complete}
\item
  \textbf{\lr{\mbox{eNB}} مقصد \ensuremath{\leftarrow} \lr{\mbox{MME}}}: \lr{Path Switch Request} (\lr{\mbox{ECGI}} جدید، \lr{\mbox{TEID}}های جدید)
\item
  \lr{\textbf{\lr{MME \ensuremath{\to} S-GW}}: Modify Bearer Request} (آدرس \lr{\mbox{eNB}} مقصد)
\item
  \lr{\mbox{\textbf{\lr{\mbox{S-GW}}}}}: مسیر \lr{\mbox{DL}} را به \lr{\mbox{eNB}} مقصد تغییر می‌دهد (نشانگر پایان به مبدأ)
\item
  \textbf{\lr{MME \ensuremath{\to} eNB} مقصد}: \lr{Path Switch Request Acknowledge}
\item
  \textbf{مقصد \ensuremath{\leftarrow} \lr{\mbox{eNB}} مبدأ}: \lr{UE Context Release} (پاک‌سازی)
\end{enumerate}

\begin{CNinsight}

\textbf{نکته کلیدی}: \lr{\mbox{S-GW}} بدون تغییر است (لنگر تحرک محلی). فقط نقطه انتهایی \lr{\mbox{S1-U}} جابه‌جا می‌شود.

\end{CNinsight}

\CNsubsection{9.7.4}{تحویل \lr{\mbox{S1}} (با تغییر \lr{MME/\allowbreak{}S-GW})}

زمانی استفاده می‌شود که \lr{\mbox{X2}} در دسترس نیست یا جابه‌جایی \lr{MME/\allowbreak{}S-GW} لازم است:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{\lr{\mbox{eNB}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{MME}} مبدأ}: \lr{Handover Required} (شناسه مقصد، علت)
\item
  \textbf{\lr{\mbox{MME}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{MME}} مقصد}: \lr{Forward Relocation Request} (زمینه \lr{\mbox{UE}}) [اگر \lr{\mbox{MME}} تغییر کند]
\item
  \textbf{\lr{\mbox{MME}} مقصد \ensuremath{\leftarrow} \lr{\mbox{eNB}} مقصد}: \lr{Handover Request} (برقراری حامل)
\item
  \textbf{\lr{\mbox{eNB}} مقصد \ensuremath{\leftarrow} \lr{\mbox{MME}} مقصد}: \lr{Handover Request Acknowledge}
\item
  \textbf{\lr{\mbox{MME}} مقصد \ensuremath{\leftarrow} \lr{\mbox{MME}} مبدأ}: \lr{Forward Relocation Response}
\item
  \textbf{\lr{\mbox{MME}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{eNB}} مبدأ}: \lr{Handover Command}
\item
  \textbf{\lr{\mbox{eNB}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{UE}}}: \lr{RRC Connection Reconfiguration}
\item
  \textbf{\lr{\mbox{UE \ensuremath{\to} eNB}} مقصد}: \lr{Handover Confirm}
\item
  \textbf{\lr{\mbox{eNB}} مقصد \ensuremath{\leftarrow} \lr{\mbox{MME}} مقصد}: \lr{Handover Notify}
\item
  \textbf{\lr{\mbox{MME}} مقصد \ensuremath{\leftarrow} \lr{\mbox{S-GW}}}: \lr{Modify Bearer /\allowbreak{} Create Session} (اگر \lr{\mbox{S-GW}} تغییر کند)
\end{enumerate}

\textbf{ارسال غیرمستقیم داده}: داده \lr{\mbox{DL}} در حین شکاف تحویل از \lr{\mbox{eNB}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{S-GW}} مبدأ \ensuremath{\leftarrow} \lr{\mbox{S-GW}} مقصد \ensuremath{\leftarrow} \lr{\mbox{eNB}} مقصد ارسال می‌شود.

  \textbf{\lr{\mbox{EPC}} سوییچینگ مداری را حذف می‌کند}: تمام خدمات (شامل صدا) روی حامل‌های \lr{\mbox{IP}}
\item
  \textbf{تقسیم کنترل/\allowbreak{}صفحه کاربر}: مقیاس‌پذیری مستقل و تحول به‌سمت \lr{\mbox{5G}} را ممکن می‌سازد
\item
  \textbf{\lr{\mbox{QoS}} مبتنی بر حامل}: کیفیت قطعی برای خدمات بلادرنگ \lr{(QCI 1-4 = GBR)}
\item
  \textbf{امنیت از پایه}: احراز هویت متقابل \lr{(EPS-AKA)}، مشتق‌سازی کلید لایه‌ای، یکپارچگی اجباری
\item
  \textbf{تحرک بدون اختلال}: لنگرگذاری \lr{\mbox{GTP}} در \lr{\mbox{S-GW}} (محلی) و \lr{\mbox{P-GW}} (جهانی) نشست‌های \lr{\mbox{IP}} را حفظ می‌کند
\item
  \textbf{معماری سیاست}: \lr{PCRF/\allowbreak{}PCEF} رفتار ترافیک بلادرنگ و آگاه به مشترک را ممکن می‌سازند
\item
  \textbf{تاب‌آوری از طریق استخرسازی}: استخرهای \lr{\mbox{MME}}، استخرهای \lr{\mbox{S-GW}} و افزونگی \lr{\mbox{HSS}} نقاط شکست واحد را کمینه می‌کنند
\end{enumerate}
