\CNsection{1}{چرا زمان‌بندی اهمیت دارد}

در \lr{\mbox{LTE}}، کانال رادیویی یک \textbf{منبع مشترک} است. چندین کاربر برای منابع محدود زمان-فرکانس روی همان حامل رقابت می‌کنند. زمان‌بند، هوش مرکزی در \lr{\mbox{eNodeB}} است که تصمیم می‌گیرد:

\begin{itemize}
\tightlist
\item
  \textbf{چه کسی} اجازه ارسال/\allowbreak{}دریافت دارد
\item
  \textbf{چه زمانی} (کدام زیرفریم)
\item
  \textbf{کجا} (کدام بلوک‌های منبع)
\item
  \textbf{چگونه} (کدام طرح مدولاسیون و کدگذاری)
\end{itemize}

بدون زمان‌بندی، کاربران با هم برخورد می‌کردند، منابع هدر می‌رفت و تضمین \lr{\mbox{QoS}} غیرممکن بود. زمان‌بند باید توازن برقرار کند بین:

\begin{itemize}
\tightlist
\item
  \textbf{بیشینه‌سازی توان عملیاتی} \lr{—} استفاده کارآمد از منابع
\item
  \textbf{انصاف} \lr{—} گرسنه نگذاشتن کاربران لبه سلول
\item
  \textbf{تأخیر} \lr{—} برآورده‌کردن بودجه تأخیر برای خدمات بلادرنگ
\item
  \lr{\mbox{\textbf{\lr{\mbox{QoS}}}}} \lr{—} احترام به حامل‌های \lr{\mbox{GBR}} (نرخ بیت تضمین‌شده)
\end{itemize}

\CNsection{4}{زمان‌بندی پیوند پایین‌سو}

\CNchained\CNsubsection{}{مرور}

زمان‌بند \lr{\mbox{eNodeB}} تمام تخصیص‌های منابع پیوند پایین‌سو را کنترل می‌کند. \lr{\mbox{UE}}ها را از تخصیص‌های خود از طریق پیام‌های \textbf{\lr{\mbox{DCI}} (اطلاعات کنترل پیوند پایین‌سو)} که روی \textbf{\lr{\mbox{PDCCH}} (کانال کنترل فیزیکی پیوند پایین‌سو)} حمل می‌شوند مطلع می‌کند.

\CNsubsection{}{محتوای \lr{\mbox{DCI}}}

پیام \lr{\mbox{DCI}} به \lr{\mbox{UE}} می‌گوید:

\begin{itemize}
\tightlist
\item
  کدام \lr{\mbox{RB}}ها تخصیص یافته‌اند (تخصیص منابع)
\item
  از کدام \lr{\mbox{MCS}} برای رمزگشایی استفاده کند
\item
  شماره فرآیند \lr{\mbox{HARQ}} و نسخه افزونگی
\item
  فرمان‌های کنترل توان
\item
  پارامترهای مرتبط با \lr{\mbox{MIMO}} (پیش‌کدگذاری، لایه‌ها)
\end{itemize}

\CNsubsection{}{انتخاب \lr{\mbox{MCS}}}

زمان‌بند \lr{\mbox{MCS}} را بر اساس موارد زیر انتخاب می‌کند:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{UE}} مقدار \lr{\mbox{\textbf{\lr{\mbox{CQI}}}}} (نشانگر کیفیت کانال) را گزارش می‌دهد
\item
  زمان‌بند \lr{\mbox{CQI}} را به شاخص \lr{\mbox{MCS}} مناسب نگاشت می‌کند
\item
  تنظیمات حلقه بیرونی را بر اساس تاریخچه \lr{ACK/\allowbreak{}NACK} اعمال می‌کند
\item
  \lr{\mbox{DCI}} را با \lr{\mbox{MCS}} انتخاب‌شده کدگذاری می‌کند تا \lr{\mbox{UE}} بتواند \lr{\mbox{PDSCH}} را رمزگشایی کند
\end{enumerate}

\CNsection{5}{زمان‌بندی پیوند بالاسو}

\CNchained\CNsubsection{}{تفاوت‌های کلیدی با پیوند پایین‌سو}

برخلاف پیوند پایین‌سو، زمان‌بندی پیوند بالاسو باید موارد زیر را در نظر بگیرد:

\begin{itemize}
\tightlist
\item
  \textbf{محدودیت \lr{\mbox{SC-FDMA}}:} \lr{\mbox{UE}}ها باید \lr{\mbox{RB}}های \textbf{پیوسته} تخصیص یابند (خاصیت تک-حامله)
\item
  \textbf{محدودیت‌های توان \lr{\mbox{UE}}:} فاصله از \lr{\mbox{eNodeB}} بر نرخ قابل دستیابی تأثیر می‌گذارد
\item
  \textbf{آگاهی از بافر:} \lr{\mbox{eNodeB}} به‌طور مستقیم وضعیت بافر \lr{\mbox{UE}} را نمی‌داند
\end{itemize}

\CNsubsection{}{اعطای زمان‌بندی}

\lr{\mbox{eNodeB}} یک \textbf{اعطای پیوند بالاسو} از طریق \lr{\mbox{DCI}} (قالب \lr{0/\allowbreak{}4}) روی \lr{\mbox{PDCCH}} می‌فرستد که شامل موارد زیر است:

\begin{itemize}
\tightlist
\item
  تخصیص \lr{\mbox{RB}} (پیوسته)
\item
  شاخص \lr{\mbox{MCS}}
\item
  فرمان کنترل توان
\item
  اطلاعات \lr{\mbox{HARQ}}
\end{itemize}

\CNsubsection{}{گزارش حاشیه توان \lr{\mbox{(PHR)}}}

\lr{\mbox{PHR}} به \lr{\mbox{eNodeB}} می‌گوید \lr{\mbox{UE}} چقدر \textbf{توان ارسال اضافی} در دسترس دارد:

\CNdisplay{PH = P_{MAX} - P_{PUSCH}}

\begin{itemize}
\tightlist
\item
  اگر \lr{\mbox{PH \textgreater{} 0}}: \lr{\mbox{UE}} می‌تواند \lr{\mbox{MCS}} بالاتر یا \lr{\mbox{RB}}های بیشتر را پشتیبانی کند
\item
  اگر \lr{\mbox{PH \ensuremath{\le} 0}}: \lr{\mbox{UE}} محدود به توان است، زمان‌بند باید تخصیص را کاهش دهد
\end{itemize}

\CNsection{6}{نشانگر کیفیت کانال \lr{\mbox{(CQI)}}}

\CNchained\CNsubsection{}{\lr{\mbox{CQI}} چه چیزی را نمایندگی می‌کند}

\lr{\mbox{CQI}} یک \textbf{شاخص 4 بیتی} \lr{\mbox{(1–15)}} است که توسط \lr{\mbox{UE}} گزارش می‌شود و بالاترین \lr{\mbox{MCS}} قابل دریافت با نرخ خطای بلوک انتقال \lr{(BLER) \ensuremath{\le} \textbf{10\%}} را نشان می‌دهد.

\lr{\mbox{CQI = 0}} به معنای «خارج از محدوده» است (کانال برای هر ارسالی بسیار ضعیف است).

\CNsection{7}{تطبیق پیوند}

\CNchained\CNsubsection{}{مدولاسیون و کدگذاری تطبیقی \lr{\mbox{(AMC)}}}

\lr{\mbox{AMC}} به‌صورت پویا مرتبه مدولاسیون و نرخ کد را برای تطبیق با شرایط لحظه‌ای کانال تنظیم می‌کند:

\begin{itemize}
\tightlist
\item
  \textbf{کانال خوب} \ensuremath{\leftarrow} مدولاسیون مرتبه بالاتر + نرخ کد بالاتر \ensuremath{\leftarrow} بیت بیشتر در هر \lr{\mbox{RE}}
\item
  \textbf{کانال ضعیف} \ensuremath{\leftarrow} مدولاسیون مرتبه پایین‌تر + نرخ کد پایین‌تر \ensuremath{\leftarrow} بیت کمتر در هر \lr{\mbox{RE}} اما مقاوم
\end{itemize}

\CNsubsection{}{تطبیق پیوند حلقه بیرونی \lr{\mbox{(OLLA)}}}

حلقه داخلی \lr{(CQI \ensuremath{\to} MCS)} ممکن است به دلایل زیر نادقیق باشد:

\begin{itemize}
\tightlist
\item
  تأخیر گزارش‌دهی \lr{\mbox{CQI}}
\item
  خطاهای اندازه‌گیری \lr{\mbox{UE}}
\item
  تغییرپذیری تداخل
\end{itemize}

\lr{\mbox{\textbf{\lr{\mbox{OLLA}}} CQI}} مؤثر را با یک آفست تنظیم می‌کند:

\CNdisplay{MCS_{effective} = f(CQI + \Delta_{OLLA})}

آفست بر اساس بازخورد \lr{\mbox{HARQ}} به‌روزرسانی می‌شود:

\begin{itemize}
\tightlist
\item
  \textbf{\lr{\mbox{ACK}} دریافت شد:} \lr{$\Delta_{\mathrm{OLLA}} \mathrel{+}= \mathrm{step\_up}$} (مثلاً \lr{\mbox{+0.1~dB}}) \ensuremath{\leftarrow} امتحان \lr{\mbox{MCS}} بالاتر
\item
  \textbf{\lr{\mbox{NACK}} دریافت شد:} \lr{$\Delta_{\mathrm{OLLA}} \mathrel{-}= \mathrm{step\_down}$} (مثلاً \lr{\mbox{-1.0~dB}}) \ensuremath{\leftarrow} بازگشت به \lr{\mbox{MCS}} پایین‌تر
\end{itemize}

هدف: حفظ \lr{\textbf{\lr{BLER 10\%}}} در اولین ارسال (نسبت \lr{$\mathrm{step\_down}/\mathrm{step\_up}$} حدود \lr{$9{:}1$}).

\CNsubsection{}{\lr{Proportional Fair (PF)}}

\textbf{اصل:} بیشینه‌سازی نسبت نرخ لحظه‌ای به توان عملیاتی میانگین.

\CNdisplay{\CNmtext{User}_{selected} = \arg\max_k \frac{R_k^{inst}(t)}{\bar{R}_k(t)}}

که در آن:

\begin{itemize}
\tightlist
\item
  \lr{$R_k^{inst}(t)$} = نرخ قابل دستیابی لحظه‌ای برای کاربر \lr{\mbox{k}} (از \lr{\mbox{CQI}})
\item
  \lr{$\bar{R}_k(t)$} = میانگین متحرک نمایی توان عملیاتی کاربر \lr{\mbox{k}}
\end{itemize}

\Needspace{16\baselineskip}

به‌روزرسانی میانگین:\CNdisplay{\bar{R}_k(t+1) = (1-\frac{1}{t_c})\bar{R}_k(t) + \frac{1}{t_c} \cdot r_k(t)}

که در آن \lr{$t_c$} پنجره میانگین‌گیری است (معمولاً \lr{100–1000 TTI}).

\CNboxsection{نکات کلیدی}

\begin{CNbox}[unbreakable]{sum}{نکات کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{شبکه منابع} یک ماتریس زمان-فرکانس است \lr{—} زمان‌بندی در دانه‌بندی \lr{\mbox{RB}} در هر \lr{\mbox{TTI 1~ms}} عمل می‌کند
\item
  \textbf{بازخورد \lr{\mbox{CQI}}} چشمان زمان‌بند است \lr{—} انتخاب \lr{\mbox{MCS}} و تخصیص منابع را هدایت می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{HARQ}}}}} قابلیت اطمینان را با 8 فرآیند موازی و \lr{\mbox{RTT 8~ms}} فراهم می‌کند
\item
  \textbf{تطبیق پیوند \lr{(AMC + OLLA)}} به‌طور پیوسته نرخ را با شرایط کانال تطبیق می‌دهد
\item
  \textbf{هیچ الگوریتم واحدی برنده نیست} \lr{—} انتخاب به سناریوی استقرار، ترکیب ترافیک و اولویت‌های اپراتور بستگی دارد
\item
  \lr{\textbf{\lr{Proportional Fair}}} استاندارد عملی در شبکه‌های تجاری است زیرا سازش خوبی بین توان عملیاتی و انصاف فراهم می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{SC-FDMA}}}}} پیوند بالاسو را به تخصیص پیوسته محدود می‌کند و انعطاف زمان‌بندی را نسبت به \lr{\mbox{OFDMA}} پیوند پایین‌سو کاهش می‌دهد
\end{enumerate}

\end{CNbox}
