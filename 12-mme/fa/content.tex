\CNsection{1}{چرا \lr{\mbox{MME}} وجود دارد}

\lr{\mbox{MME}} \textbf{موجودیت مرکزی صفحه کنترل} در معماری \lr{LTE/\allowbreak{}EPC} است. دلیل وجود آن:

\begin{itemize}
\tightlist
\item
  \textbf{تخلیه \lr{\mbox{RAN}} از سیگنالینگ هسته}: در \lr{2G/\allowbreak{}3G}، \lr{RNC/\allowbreak{}BSC} سیگنالینگ قابل توجهی از شبکه هسته را مدیریت می‌کرد. معماری مسطح \lr{\mbox{LTE}} تمام پایان‌دهی سیگنالینگ \lr{\mbox{NAS}} (لایه غیر-دسترسی) را به \lr{\mbox{MME}} منتقل کرد و طراحی \lr{eNodeB (eNB)} را ساده کرد.
\item
  \textbf{متمرکزسازی هوش صفحه کنترل}: احراز هویت، مدیریت تحرک، برقراری حامل و صفحه‌بندی همه توسط \lr{\mbox{MME}} هماهنگ می‌شوند.
\item
  \textbf{جداکردن صفحه کنترل از صفحه کاربر}: \lr{\mbox{MME}} به هیچ ترافیک صفحه کاربری دست نمی‌زند \lr{—} صرفاً سیگنالینگ را مدیریت می‌کند، که مقیاس‌پذیری مستقل \lr{\mbox{CP}} و \lr{\mbox{UP}} را ممکن می‌سازد.
\end{itemize}

\CNsection{2}{توابع \lr{\mbox{MME}}}

\CNchained\CNsubsection{2.1}{پایان‌دهی سیگنالینگ \lr{\mbox{NAS}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{MME}} نقطه انتهایی \lr{\mbox{NAS}} است \lr{—} تمام پیام‌های \lr{\mbox{NAS}} از \lr{\mbox{UE}} را پایان می‌دهد (که از طریق \lr{\mbox{eNB}} با \lr{\mbox{S1-AP}} به‌صورت شفاف منتقل می‌شوند).
\item
  پیام‌های \lr{\mbox{NAS}} بین \lr{\mbox{UE}} و \lr{\mbox{MME}} محافظت‌شده با یکپارچگی و رمزگذاری‌شده هستند.
\end{itemize}

\CNsubsection{2.2}{مدیریت تحرک}

\begin{itemize}
\tightlist
\item
  مدیریت ناحیه ردیابی \lr{\mbox{(TA)}} و تخصیص فهرست \lr{\mbox{TA}}
\item
  سیگنالینگ تحویل (مبتنی بر \lr{\mbox{S1}} و \lr{\mbox{X2}} با دخالت \lr{\mbox{MME}})
\item
  تحرک بین-\lr{\mbox{RAT}} (به/\allowbreak{}از \lr{2G/\allowbreak{}3G} از طریق \lr{S3/\allowbreak{}S10})
\end{itemize}

\CNsubsection{2.3}{احراز هویت و امنیت}

\begin{itemize}
\tightlist
\item
  بردارهای احراز هویت را از \lr{\mbox{HSS}} بازیابی می‌کند (از طریق \lr{\mbox{S6a}})
\item
  \lr{\mbox{EPS-AKA}} (احراز هویت و توافق کلید) را اجرا می‌کند
\item
  کلیدهای \lr{\mbox{NAS}} (\lr{\mbox{\ensuremath{K_{\mathrm{NASint}}}}}، \lr{\mbox{\ensuremath{K_{\mathrm{NASenc}}}}}) و کلیدهای \lr{AS (\ensuremath{K_{\mathrm{eNB}}})} را مشتق می‌کند
\item
  \lr{NAS Security Mode Command} را آغاز می‌کند
\end{itemize}

\CNsubsection{2.4}{مدیریت حامل}

\begin{itemize}
\tightlist
\item
  حامل‌های پیش‌فرض \lr{\mbox{EPS}} را در حین اتصال برقرار می‌کند
\item
  فعال‌سازی/\allowbreak{}اصلاح/\allowbreak{}غیرفعال‌سازی حامل اختصاصی را هماهنگ می‌کند
\item
  با \lr{S-GW (S11)} و به‌طور غیرمستقیم با \lr{\mbox{P-GW}} تعامل دارد
\end{itemize}

\CNsubsection{2.5}{صفحه‌بندی}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}}ها را در حالت \lr{\mbox{ECM-IDLE}} هنگام رسیدن داده پیوند پایین‌سو صفحه‌بندی می‌کند
\item
  پیام‌های \lr{S1-AP Paging} را به تمام \lr{\mbox{eNB}}ها در فهرست \lr{\mbox{TA}} ثبت‌شده \lr{\mbox{UE}} می‌فرستد
\end{itemize}

\CNsubsection{2.6}{دسترسی‌پذیری در حالت بیکار}

\begin{itemize}
\tightlist
\item
  زمینه \lr{\mbox{UE}} را حتی وقتی \lr{\mbox{UE}} در \lr{IDLE (ECM-IDLE)} است حفظ می‌کند
\item
  می‌داند \lr{\mbox{UE}} ممکن است در کدام نواحی ردیابی باشد
\item
  هنگام رسیدن داده یا سیگنالینگ، صفحه‌بندی را فعال می‌کند
\end{itemize}

\CNsubsection{2.7}{انتخاب، استخرسازی و توازن مجدد \lr{\mbox{MME}}}

\begin{itemize}
\tightlist
\item
  چندین \lr{\mbox{MME}} به همان ناحیه خدمات می‌دهند (استخر \lr{\mbox{MME}})
\item
  توازن بار از طریق ظرفیت نسبی (عامل وزن) که به \lr{\mbox{eNB}}ها اعلام می‌شود
\item
  توازن مجدد: انتقال تدریجی زمینه‌های \lr{\mbox{UE}} هنگام افزودن/\allowbreak{}حذف \lr{\mbox{MME}}ها
\end{itemize}

\CNsection{3}{پشته پروتکل \lr{\mbox{NAS}}}

\lr{\mbox{NAS}} (لایه غیر-دسترسی) بین \lr{\mbox{UE}} و \lr{\mbox{MME}} عمل می‌کند و نسبت به \lr{\mbox{eNB}} شفاف است.

\CNsubsection{3.1}{\lr{\mbox{EMM}} \lr{—} مدیریت تحرک \lr{\mbox{EPS}}}

ثبت‌نام و تحرک \lr{\mbox{UE}} را مدیریت می‌کند:

\begin{itemize}
\tightlist
\item
  \lr{Attach /\allowbreak{} Detach}
\item
  به‌روزرسانی ناحیه ردیابی \lr{\mbox{(TAU)}}
\item
  احراز هویت \lr{(EPS-AKA)}
\item
  کنترل حالت امنیتی
\item
  تخصیص مجدد \lr{\mbox{GUTI}}
\item
  درخواست خدمات
\end{itemize}

\textbf{حالت‌های \lr{\mbox{EMM}}:}

\begin{itemize}
\tightlist
\item
  \lr{EMM-DEREGISTERED}: \lr{\mbox{UE}} ثبت‌نام نشده است
\item
  \lr{EMM-REGISTERED}: \lr{\mbox{UE}} به شبکه متصل است
\end{itemize}

\CNsubsection{3.2}{\lr{\mbox{ESM}} \lr{—} مدیریت نشست \lr{\mbox{EPS}}}

اتصالات \lr{\mbox{PDN}} و حامل‌های \lr{\mbox{EPS}} را مدیریت می‌کند:

\begin{itemize}
\tightlist
\item
  درخواست/\allowbreak{}پذیرش اتصال \lr{\mbox{PDN}}
\item
  تخصیص منابع حامل
\item
  اصلاح /\allowbreak{} غیرفعال‌سازی حامل
\item
  قطع اتصال \lr{\mbox{PDN}}
\end{itemize}

\textbf{\lr{\mbox{ESM}} در حین اتصال به‌صورت حمل‌شده روی \lr{\mbox{EMM}} عمل می‌کند} (پیام \lr{\mbox{ESM}} درون \lr{Attach Request} جای می‌گیرد).

\CNsection{5}{مدیریت حامل}

\CNchained\CNsubsection{5.1}{برقراری حامل پیش‌فرض}

\begin{itemize}
\tightlist
\item
  به‌طور خودکار در حین اتصال اولیه ایجاد می‌شود (بخشی از \lr{Create Session})
\item
  یک حامل پیش‌فرض در هر اتصال \lr{\mbox{PDN}} (همیشه فعال، غیر-\lr{\mbox{GBR}})
\item
  \lr{\mbox{QCI}} معمولاً 9 (اینترنت بهترین تلاش) یا 5 (سیگنالینگ \lr{\mbox{IMS}})
\item
  بدون قطع اتصال \lr{\mbox{PDN}} نمی‌تواند غیرفعال شود
\end{itemize}

\item

\end{itemize}

\CNsection{6}{انتخاب \lr{\mbox{MME}}}

\CNchained\CNsubsection{6.1}{انتخاب مبتنی بر \lr{\mbox{DNS}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{eNB}} نام \lr{\mbox{FQDN}} مربوط به \lr{\mbox{MME}} را از \lr{\mbox{TAI}} حل می‌کند \ensuremath{\leftarrow} \lr{\mbox{DNS}} فهرست \lr{\mbox{IP}}های \lr{\mbox{MME}} را بازمی‌گرداند
\item
  قالب: \lr{\CNcode{tac-XXXX.}\allowbreak{}\CNcode{tac-YYYY.}\allowbreak{}\CNcode{mme.}\allowbreak{}\CNcode{epc.}\allowbreak{}\CNcode{mnc\textless{}MNC\textgreater{}.}\allowbreak{}\CNcode{mcc\textless{}MCC\textgreater{}.}\allowbreak{}\CNcode{3gppnetwork.}\allowbreak{}\CNcode{org}}
\end{itemize}

\CNsubsection{6.2}{توازن بار}

\begin{itemize}
\tightlist
\item
  هر \lr{\mbox{MME}} یک \textbf{ظرفیت نسبی \lr{\mbox{MME}}} (وزن، \CNnum{0-255)} را از طریق \lr{\mbox{S1 Setup}} اعلام می‌کند
\item
  \lr{\mbox{eNB UE}}های جدید را به‌تناسب در بین اعضای استخر توزیع می‌کند
\item
  مقدار ظرفیت بالاتر = \lr{\mbox{UE}}های بیشتر به آن \lr{\mbox{MME}} هدایت می‌شوند
\end{itemize}

\CNsubsection{6.3}{\lr{\mbox{GUMMEI}} \lr{—} شناسه یکتای جهانی \lr{\mbox{MME}}}

\CNdisplay{\mathrm{GUMMEI} = \mathrm{MCC} + \mathrm{MNC} + \text{\rl{شناسه گروه MME (MMEGI)}} + \text{\rl{کد MME (MMEC)}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{MMEGI}} استخر \lr{\mbox{MME}} را شناسایی می‌کند
\item
  \lr{\mbox{MMEC}} یک \lr{\mbox{MME}} مشخص درون استخر را شناسایی می‌کند
\item
  \lr{GUTI = GUMMEI + M-TMSI} (هویت موقت \lr{\mbox{UE}})
\end{itemize}

\CNsection{7}{\lr{\mbox{S1-Flex}} و استخرسازی \lr{\mbox{MME}}}

\CNchained\CNsubsection{}{مفهوم}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{S1-Flex}}}}}: هر \lr{\mbox{eNB}} به \textbf{چندین \lr{\mbox{MME}}} در یک استخر متصل می‌شود (نه فقط یکی)
\item
  \textbf{ناحیه استخر \lr{\mbox{MME}}}: ناحیه جغرافیایی که توسط مجموعه‌ای از \lr{\mbox{MME}}ها خدمات می‌گیرد
\item
  مزایا:

  \begin{itemize}
  \tightlist
  \item
    \textbf{افزونگی}: اگر یک \lr{\mbox{MME}} شکست بخورد، بقیه به \lr{\mbox{UE}}ها خدمات می‌دهند
  \item
    \textbf{اشتراک بار}: \lr{\mbox{UE}}های جدید بر اساس وزن توزیع می‌شوند
  \item
    \textbf{افزونگی جغرافیایی}: اعضای استخر می‌توانند در مراکز داده مختلف باشند
  \item
    \textbf{مقیاس‌دهی تدریجی}: افزودن/\allowbreak{}حذف \lr{\mbox{MME}}ها بدون وقفه سرویس
  \end{itemize}
\end{itemize}

\CNsubsection{}{اضافه‌بار و توازن مجدد \lr{\mbox{MME}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{MME}} یک \lr{\textbf{\lr{Overload Start}}} به \lr{\mbox{eNB}}ها می‌فرستد \ensuremath{\leftarrow} \lr{\mbox{eNB}} اتصالات جدید را به سایر اعضای استخر هدایت می‌کند
\item
  \textbf{توازن مجدد}: \lr{\mbox{MME}} از \lr{\mbox{UE}}ها می‌خواهد دوباره متصل شوند تا بار را بازتوزیع کند (از طریق جداسازی ضمنی با نشانه اتصال مجدد)
\end{itemize}

\item
  \textbf{شبکه‌بندی}: \lr{\mbox{AMF}} انتخاب برش (تعامل \lr{\mbox{NSSF}}) را انجام می‌دهد و به \lr{\mbox{SMF}} مختص برش مسیریابی می‌کند
\item
  \textbf{طراحی بدون حالت}: زمینه \lr{\mbox{UE}} می‌تواند به‌صورت خارجی \lr{\mbox{(UDSF)}} برای تاب‌آوری ذخیره شود
\item
  \textbf{ثبت‌نام یکپارچه}: بدون \lr{\mbox{Attach}} در برابر \lr{\mbox{TAU}} جداگانه \lr{—} رویه \lr{\mbox{Registration}} واحد هر دو را مدیریت می‌کند
\end{enumerate}
