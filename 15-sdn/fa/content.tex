\CNsection{2}{معماری سنتی در برابر \lr{\mbox{SDN}}}

\CNchained\CNsubsection{}{معماری سنتی}

\begin{itemize}
\tightlist

\end{itemize}

\begin{itemize}

  سوییچ‌ها به دستگاه‌های ارسال ساده تبدیل می‌شوند
\item
  کنترلر \textbf{نمای جهانی} از کل شبکه دارد
\item
  قابل برنامه‌ریزی از طریق \lr{\mbox{API}}ها (شمال‌سو و جنوب‌سو)
\end{itemize}

\CNsubsection{3.1}{لایه زیرساخت (صفحه داده)}

\begin{itemize}
\tightlist
\item
  \textbf{سوییچ‌ها و مسیریاب‌های} فیزیکی یا مجازی
\item
  ارسال بسته را بر اساس \textbf{جداول جریان} برنامه‌ریزی‌شده توسط کنترلر انجام می‌دهد
\item
  بدون تصمیم‌گیری مستقل \lr{—} دستورالعمل‌های لایه کنترل را دنبال می‌کند
\item
  سخت‌افزار: سوییچ‌های جعبه-سفید، \lr{\mbox{ASIC}}های قابل برنامه‌ریزی، \lr{\mbox{TCAM}} برای جستجوی سریع
\end{itemize}

\CNsubsection{3.2}{لایه کنترل (کنترلر \lr{\mbox{SDN}})}

\begin{itemize}
\tightlist
\item
  \textbf{«مغز»} شبکه
\item
  \textbf{نمای جهانی} از توپولوژی، پیوندها، میزبان‌ها و ترافیک را نگه می‌دارد
\item
  سیاست‌های سطح بالا را به قواعد جریان سطح پایین ترجمه می‌کند
\item
  نمونه‌ها: \lr{\mbox{ONOS}}، \lr{OpenDaylight (ODL)}، \lr{\mbox{Ryu}}، \lr{\mbox{Floodlight}}، \lr{Nokia NSP}، \lr{Cisco ACI}
\end{itemize}

\CNsubsection{3.3}{لایه کاربرد}

\begin{itemize}
\tightlist
\item
  \textbf{منطق تجاری و سیاست‌ها} به‌صورت کاربردهای نرم‌افزاری بیان می‌شوند
\item
  \lr{\mbox{API}} شمال‌سوی کنترلر را مصرف می‌کند
\item
  نمونه‌ها: مهندسی ترافیک، امنیت (خرد-قطعه‌بندی)، توازن بار، پایش، شبکه‌سازی مبتنی بر قصد
\end{itemize}

\CNsection{5}{واسط‌های جنوب‌سو}

واسط جنوب‌سو \textbf{کنترلر را به دستگاه‌های زیرساخت} (صفحه داده) متصل می‌کند.

\CNsubsection{5.1}{\lr{\mbox{OpenFlow}}}

\begin{itemize}
\tightlist
\item
  شناخته‌شده‌ترین پروتکل جنوب‌سوی \lr{\mbox{SDN}}
\item
  تعریف‌شده توسط \textbf{بنیاد شبکه‌سازی باز \lr{\mbox{(ONF)}}}
\item
  روشی استاندارد برای برنامه‌ریزی \textbf{جداول جریان} در سوییچ‌ها فراهم می‌کند
\item
  کنترلر ورودی‌های جریان را نصب/\allowbreak{}اصلاح/\allowbreak{}حذف می‌کند
\item
  سوییچ رویدادها (تغییرات توپولوژی، \lr{packet-in}) را به کنترلر گزارش می‌دهد
\item
  نسخه‌ها: \CNnum{1.0} \ensuremath{\to} \CNnum{1.3} (پراستقرارترین) \ensuremath{\leftarrow} \CNnum{1.5}
\end{itemize}

\CNsubsection{5.2}{\lr{\mbox{P4}} (پردازنده‌های بسته مستقل از پروتکل برنامه‌نویسی‌پذیر)}

\begin{itemize}
\tightlist
\item
  \textbf{زبان دامنه-خاص} برای برنامه‌ریزی خود صفحه داده
\item
  فراتر از \lr{\mbox{OpenFlow}} می‌رود: مشخص می‌کند سوییچ روی \textbf{چه چیزی} می‌تواند تطبیق دهد
\item
  خطوط لوله تجزیه و پردازش بسته قابل برنامه‌ریزی
\item
  مستقل از هدف: به سخت‌افزارهای مختلف کامپایل می‌شود (\lr{\mbox{Tofino}}، \lr{\mbox{FPGA}}، سوییچ‌های نرم‌افزاری)
\item
  مورد استفاده: سرآیندهای سفارشی، رایانش درون-شبکه، \lr{\mbox{INT}} (تله‌متری درون-باند شبکه)
\end{itemize}

\CNsubsection{5.3}{\lr{\mbox{ForCES}} (جداسازی عنصر ارسال و کنترل)}

\begin{itemize}
\tightlist
\item
  استاندارد \lr{IETF (RFC 5810)}
\item
  عنصر کنترل \lr{\mbox{(CE)}} را از عنصر ارسال \lr{\mbox{(FE)}} جدا می‌کند
\item
  پروتکلی برای ارتباط \lr{\mbox{CE-FE}} تعریف می‌کند
\item
  در عمل کمتر از \lr{\mbox{OpenFlow}} پذیرفته شده است
\end{itemize}

  لایه‌های رویی بر فراز چندین انتقال زیرین (\lr{\mbox{MPLS}}، اینترنت، \lr{\mbox{LTE}})

  مسیریابی آگاه به کاربرد (مسیریابی صدا روی \lr{\mbox{MPLS}}، وب روی اینترنت)
\item
  تأمین بدون-لمس برای سایت‌های شعب
\item
  سازندگان: \lr{Cisco Viptela}، \lr{VMware VeloCloud}، \lr{\mbox{Fortinet}}، \lr{\mbox{Versa}}
\end{itemize}

\CNsubsection{8.4}{برنامه‌پذیری \lr{\mbox{UPF}} از طریق \lr{\mbox{PFCP}}}

\lr{PFCP (3GPP TS 29.244)} پروتکل بین \lr{\mbox{SMF}} و \lr{\mbox{UPF}} است:

\begin{itemize}
\tightlist
\item
  \textbf{\lr{\mbox{PDR}} (قاعده تشخیص بسته)}: معیارهای تطبیق (واسط مبدأ، \lr{\mbox{IP}}، \lr{\mbox{TEID}} در \lr{\mbox{GTP}})
\item
  \textbf{\lr{\mbox{FAR}} (قاعده اقدام ارسال)}: اقدام (ارسال، دور ریختن، بافر، کپسوله‌سازی)
\item
  \textbf{\lr{\mbox{QER}} (قاعده اجرای \lr{\mbox{QoS}})}: محدودسازی نرخ (\lr{\mbox{MBR}}، \lr{\mbox{GBR}})
\item
  \textbf{\lr{\mbox{URR}} (قاعده گزارش‌دهی مصرف)}: اندازه‌گیری و گزارش‌دهی ترافیک
\end{itemize}

این از نظر عملکردی معادل مدل تطبیق-اقدام \lr{\mbox{OpenFlow}} است!

\CNsection{10}{چالش‌های \lr{\mbox{SDN}}}

\CNchained\CNsubsection{10.1}{مقیاس‌پذیری کنترلر}

\begin{itemize}
\tightlist
\item
  کنترلر واحد نمی‌تواند هزاران سوییچ + میلیون‌ها جریان را مدیریت کند
\item
  \textbf{راه‌حل}: خوشه‌بندی کنترلر (\lr{\mbox{ONOS}}، \lr{\mbox{ODL}} از کنترلرهای توزیع‌شده پشتیبانی می‌کنند)
\item
  مبادله: سازگاری در برابر دسترس‌پذیری (قضیه \lr{\mbox{CAP}} اعمال می‌شود)
\end{itemize}

\CNsubsection{10.2}{نقطه شکست واحد}

\begin{itemize}
\tightlist
\item
  اگر کنترلر شکست بخورد، هیچ جریان جدیدی نمی‌تواند نصب شود
\item
  صفحه داده ارسال جریان‌های موجود را ادامه می‌دهد (حالت فعالانه)
\item
  \textbf{کاهش اثر}: افزونگی کنترلر (فعال-آماده، فعال-فعال)
\end{itemize}

\CNsubsection{10.3}{واسط شرق-غرب}

\begin{itemize}
\tightlist
\item
  ارتباط بین نمونه‌های کنترلر در یک خوشه
\item
  بدون پروتکل استاندارد (هر کنترلر خوشه‌بندی اختصاصی دارد)
\item
  چالش‌ها: همگام‌سازی وضعیت، سناریوهای مغز-شکافته، سازگاری
\end{itemize}
