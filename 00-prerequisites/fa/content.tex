این سند مفاهیم بنیادی را که به‌عنوان دانش قبلی فرض می‌شوند مرور می‌کند. از آن به‌عنوان یادآوری پیش از ورود به ماژول‌های درس استفاده کنید.

\textbf{نکات کلیدی برای به خاطر سپردن:}

\begin{itemize}
\tightlist
\item
  لایه‌های \lr{\mbox{1–2}} جایی هستند که بیشتر نوآوری‌های مختص بی‌سیم رخ می‌دهد \lr{(PHY/\allowbreak{}MAC)}.
\item
  «صفحه کاربر» داده را حمل می‌کند (لایه \lr{\mbox{3}}+)، در حالی که «صفحه کنترل» سیگنالینگ را مدیریت می‌کند.
\item
  در \lr{\mbox{5G}}، پشته پروتکل تقسیم می‌شود: \lr{\mbox{CU}} (\lr{\mbox{PDCP}} و بالاتر) در برابر \lr{DU (RLC/\allowbreak{}MAC)} در برابر \lr{\mbox{RU (PHY)}}.
\item
  تونل‌های \lr{\mbox{GTP-U}} بسته‌های \lr{\mbox{IP}} کاربر را بین ایستگاه پایه و هسته کپسوله می‌کنند.
\end{itemize}

\CNsection{2}{مفاهیم پایه \lr{\mbox{RF}}}

\CNchained\CNsubsection{}{فرکانس و پهنای باند}

\begin{itemize}
\tightlist
\item
  \textbf{فرکانس \lr{\mbox{(f)}}:} تعداد نوسان‌ها در ثانیه \lr{\mbox{(Hz)}}. باندهای موبایل از \lr{\mbox{700~MHz}} تا \lr{\mbox{100~GHz}} گسترده‌اند.
\item
  \textbf{طول‌موج:} \lr{$\lambda = c/f$}. فرکانس بالاتر \ensuremath{\leftarrow} طول‌موج کوتاه‌تر \ensuremath{\leftarrow} آنتن‌های کوچک‌تر اما تلفات مسیر بیشتر.
\item
  \textbf{پهنای باند \lr{\mbox{(BW)}}:} محدوده فرکانس‌هایی که یک سیگنال اشغال می‌کند. پهنای باند بیشتر \ensuremath{\leftarrow} نرخ داده بالاتر (شانون).
\end{itemize}

\CNsubsection{}{مدولاسیون}

\begin{itemize}
\tightlist
\item
  \textbf{هدف:} نگاشت بیت‌های دیجیتال روی سیگنال‌های حامل آنالوگ.
\item
  \textbf{طرح‌های معمول:} \lr{\mbox{BPSK}}، \lr{\mbox{QPSK}}، \lr{\mbox{16-QAM}}، \lr{\mbox{64-QAM}}، \lr{\mbox{256-QAM}}.
\item
  مدولاسیون مرتبه بالاتر = بیت بیشتر در هر نماد اما نیازمند \lr{\mbox{SNR}} بهتر.
\end{itemize}

\CNsubsection{}{نسبت سیگنال به نویز \lr{\mbox{(SNR)}}}

\begin{itemize}
\tightlist
\item
  \lr{$\mathrm{SNR} = P_{\mathrm{signal}}/P_{\mathrm{noise}}$} (خطی) یا \lr{$10\log_{10}(P_{\mathrm{signal}}/P_{\mathrm{noise}})$} بر حسب \lr{\mbox{dB}}.
\item
  ظرفیت شانون: \lr{$C = \mathrm{BW}\cdot\log_2(1+\mathrm{SNR})$} [بیت/\allowbreak{}ثانیه].
\item
  تعیین می‌کند کدام طرح مدولاسیون و کدگذاری \lr{\mbox{(MCS)}} می‌تواند استفاده شود.
\end{itemize}

\CNsection{3}{مبانی ارتباطات دیجیتال}

\CNchained\CNsubsection{}{نرخ خطای بیت \lr{\mbox{(BER)}}}

\begin{itemize}
\tightlist
\item
  احتمال اینکه یک بیت دریافتی با بیت ارسالی متفاوت باشد.
\item
  به \lr{\mbox{SNR}}، مرتبه مدولاسیون و کدگذاری بستگی دارد.
\item
  اهداف معمول: \lr{\mbox{10\textsuperscript{-3}}} (صدا)، \lr{\mbox{10\textsuperscript{-6}}} (داده پیش از \lr{\mbox{ARQ}}).
\end{itemize}

\CNsubsection{}{کدگذاری کانال}

\begin{itemize}
\tightlist
\item
  \textbf{هدف:} افزودن افزونگی برای تشخیص/\allowbreak{}تصحیح خطا.
\item
  \textbf{کدهای توربو:} در \lr{3G/\allowbreak{}4G} برای کانال‌های داده استفاده می‌شوند.
\item
  \textbf{کدهای \lr{\mbox{LDPC}}:} در \lr{\mbox{5G NR}} برای داده \lr{\mbox{(eMBB)}} استفاده می‌شوند.
\item
  \textbf{کدهای پولار:} در \lr{\mbox{5G NR}} برای کانال‌های کنترل استفاده می‌شوند.
\item
  \textbf{نرخ کد \lr{\mbox{(R)}}:} نسبت بیت‌های اطلاعات به کل بیت‌ها. \lr{$R = k/n$} (مثلاً \lr{$R = 1/2$} یعنی 50\% افزونگی).
\end{itemize}

\CNsubsection{}{مفهوم \lr{\mbox{OFDM}}}

\begin{itemize}
\tightlist
\item
  \textbf{مسیله:} کانال‌های پهن‌باند از محوشدگی انتخابی-فرکانسی رنج می‌برند.
\item
  \textbf{راه‌حل:} تقسیم کانال پهن‌باند به بسیاری زیرحامل باریک، که هر یک محوشدگی تخت را تجربه می‌کند.
\item
  \textbf{پارامترهای کلیدی:}

  \begin{itemize}
  \tightlist
  \item
    فاصله زیرحامل \lr{\mbox{($\Delta f$)}}: \lr{15~kHz (LTE)}، \lr{15/\allowbreak{}30/\allowbreak{}60/\allowbreak{}120/\allowbreak{}240~kHz (5G NR)}.
  \item
    اندازه \lr{\mbox{FFT}}: تعداد زیرحامل‌ها را تعیین می‌کند.
  \item
    پیشوند چرخه‌ای \lr{\mbox{(CP)}}: بازه محافظ برای مقابله با تداخل بین-نمادی \lr{\mbox{(ISI)}}.
  \end{itemize}
\item
  \textbf{مزایا:} یکسان‌سازی ساده (1-ضربه در هر زیرحامل)، تخصیص منبع انعطاف‌پذیر.
\item
  \lr{\mbox{\textbf{\lr{\mbox{OFDMA:}}}}} چندین کاربر زیرحامل‌ها را به اشتراک می‌گذارند \ensuremath{\leftarrow} پایه پیوند پایین‌سو \lr{\mbox{LTE}} و \lr{\mbox{5G}}.
\end{itemize}

\CNsection{4}{مبانی شبکه‌سازی \lr{\mbox{IP}}}

\CNchained\CNsubsection{}{آدرس‌دهی}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{IPv4:}}}}} آدرس‌های 32 بیتی (مثلاً \CNnum{192.168.1.1)}، \lr{\mbox{NAT}} در شبکه‌های موبایل رایج است.
\item
  \lr{\mbox{\textbf{\lr{\mbox{IPv6:}}}}} آدرس‌های 128 بیتی، به‌طور فزاینده در \lr{\mbox{5G}} استفاده می‌شود (از \lr{\mbox{NAT}} اجتناب می‌کند).
\item
  \textbf{زیرشبکه‌بندی:} نشانه‌گذاری \lr{\mbox{CIDR}} (مثلاً \lr{10.0.0.0/\allowbreak{}24 = 256} آدرس).
\end{itemize}

\CNsubsection{}{مسیریابی}

\begin{itemize}
\tightlist
\item
  شبکه‌های هسته موبایل از مسیریابی \lr{\mbox{IP}} بین توابع شبکه استفاده می‌کنند.
\item
  \textbf{مسیرهای ایستا} برای توپولوژی‌های ساده؛ \lr{\textbf{\lr{BGP/\allowbreak{}OSPF}}} در شبکه‌های حامل.
\item
  \textbf{تونل‌سازی \lr{\mbox{GTP}}:} بسته‌های \lr{\mbox{IP}} مشترک را درون \lr{GTP/\allowbreak{}UDP/\allowbreak{}IP} بین \lr{\mbox{gNB}} و \lr{\mbox{UPF}} کپسوله می‌کند.
\end{itemize}

\CNsection{5}{احتمال پایه}

\CNchained\CNsubsection{}{چرا احتمال در مخابرات؟}

\begin{itemize}
\tightlist
\item
  دسترسی تصادفی \lr{\mbox{(RACH)}}: احتمال برخورد با چندین کاربر.
\item
  زمان‌بندی: مدل‌سازی رسیدن ترافیک (فرآیند پواسون).
\item
  محوشدگی: توزیع‌های رایلی/\allowbreak{}رایسی چندمسیری را مدل می‌کنند.
\item
  تحلیل \lr{\mbox{BER}}: تابع \lr{\mbox{Q}}، محاسبات احتمال خطا.
\end{itemize}

\CNsubsection{}{مفاهیم کلیدی}

\begin{itemize}
\tightlist
\item
  \textbf{متغیر تصادفی \lr{\mbox{(X)}}:} نتیجه یک آزمایش تصادفی.
\item
  \lr{\textbf{\lr{PDF /\allowbreak{} PMF:}}} تابع چگالی/\allowbreak{}جرم احتمال.
\item
  \textbf{امید ریاضی:} \lr{$E[X] = \sum_i x_i\,P(x_i)$} \lr{—} نتیجه میانگین.
\item
  \textbf{واریانس:} \lr{$\operatorname{Var}(X) = E[(X-\mu)^2]$} \lr{—} پراکندگی حول میانگین.
\end{itemize}

\CNsubsection{}{بودجه پیوند (پیش‌نمایش)}

\CNdisplay{P_{\mathrm{Rx}}\,[\mathrm{dBm}] = P_{\mathrm{Tx}}\,[\mathrm{dBm}] + G_{\mathrm{Tx}}\,[\mathrm{dBi}] - L_{\mathrm{path}}\,[\mathrm{dB}] + G_{\mathrm{Rx}}\,[\mathrm{dBi}]}

(توان دریافتی = توان ارسالی + بهره آنتن فرستنده \lr{\mbox{\ensuremath{-}}} تلفات مسیر + بهره آنتن گیرنده)

\CNsubsection{}{مسایل تمرینی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  یک فرستنده \lr{\mbox{2~W}} خروجی می‌دهد. بر حسب \lr{\mbox{dBm}} بیان کنید. \ensuremath{\leftarrow} \lr{$10\log_{10}(2000) = 33$ dBm}.
\item
  تلفات مسیر \lr{\mbox{130~dB}} است، \lr{Tx = 43 dBm}، بهره آنتن‌ها مجموعاً \lr{\mbox{18~dBi}}. توان \lr{\mbox{Rx}}؟ \ensuremath{\leftarrow} \lr{$43 + 18 - 130 = -69$ dBm}.
\item
  \lr{SNR = 20~dB}. نسبت خطی چیست؟ \ensuremath{\leftarrow} \lr{$10^{20/10} = 100$}.
\end{enumerate}
