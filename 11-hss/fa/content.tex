بدون یک پایگاه داده مرکزی:

\begin{itemize}
\tightlist
\item
  هر دستگاهی می‌توانست خود را جای مشترک جا بزند
\item
  شبکه نمی‌تواند تماس/\allowbreak{}داده ورودی را به مکان صحیح مسیریابی کند
\item
  تمایز خدمات (طرح‌های ممتاز در برابر پایه) غیرممکن می‌شد
\item
  رومینگ بین شبکه‌ها کار نمی‌کرد
\end{itemize}

\textbf{\lr{\mbox{HSS}} (سرور مشترک خانگی)} تحول \lr{\mbox{HLR}} (ثبت‌کننده مکان خانگی) نسل \lr{2G/\allowbreak{}3G} است که برای شبکه‌های تمام-\lr{IP LTE/\allowbreak{}EPC} با استفاده از پروتکل \lr{\mbox{Diameter}} به‌جای \lr{SS7/\allowbreak{}MAP} بازطراحی شده است.

\CNsubsection{}{مسیولیت‌های کلیدی:}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{ذخیره هویت دایمی مشترک \lr{\mbox{(IMSI)}}}

  \begin{itemize}
  \tightlist
  \item
    هویت بین‌المللی مشترک موبایل \lr{—} شناسه 15 رقمی یکتا در سطح جهانی
  \item
    قالب: \lr{\mbox{MCC}} (3 رقم) + \lr{\mbox{MNC}} (\CNnum{2-3} رقم) + \lr{\mbox{MSIN}} (\CNnum{9-10} رقم)
  \end{itemize}
\item
  \textbf{ذخیره اعتبارنامه‌های احراز هویت \lr{\mbox{(K, OPc)}}}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{\textbf{\lr{\mbox{K}}}}}: کلید مخفی دایمی 128 بیتی (فقط بین \lr{\mbox{USIM}} و \lr{\mbox{HSS}} مشترک)
  \item
    \lr{\mbox{\textbf{\lr{\mbox{OPc}}}}}: ثابت مختص اپراتور مشتق‌شده از \lr{\mbox{OP}} (کلید اپراتور) و \lr{\mbox{K}}
  \item
    هرگز روی هوا منتقل نمی‌شود \lr{—} برای تولید اعتبارنامه‌های موقت استفاده می‌شود
  \end{itemize}
\item
  \textbf{ذخیره پروفایل‌های اشتراک (\lr{\mbox{APN}}، \lr{\mbox{QoS}}، خدمات مجاز)}

  \begin{itemize}
  \tightlist
  \item
    مشترک می‌تواند به کدام \lr{\mbox{APN}}ها متصل شود
  \item
    حداکثر نرخ بیت \lr{\mbox{(AMBR)}} برای پیوند بالاسو/\allowbreak{}پایین‌سو
  \item
    شناسه‌های کلاس کیفیت خدمات \lr{\mbox{(QCI)}}
  \item
    خدمات مجاز و محدودیت‌ها
  \end{itemize}
\item
  \textbf{ردیابی مکان \lr{\mbox{UE}} (\lr{\mbox{MME}} خدمات‌دهنده)}

  \begin{itemize}
  \tightlist
  \item
    ثبت می‌کند که در حال حاضر کدام \lr{\mbox{MME}} به \lr{\mbox{UE}} خدمات می‌دهد
  \item
    مسیریابی خدمات ورودی (\lr{\mbox{MT-SMS}}، صفحه‌بندی) را ممکن می‌سازد
  \item
    هر بار که \lr{\mbox{UE}} به‌روزرسانی ناحیه ردیابی \lr{\mbox{(TAU)}} یا اتصال انجام می‌دهد به‌روزرسانی می‌شود
  \end{itemize}
\end{enumerate}

\CNsubsection{}{مدیریت \lr{\mbox{SQN}} (محافظت از بازپخش)}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{SQN}}}}} یک شمارنده 48 بیتی است که به‌طور مستقل توسط \lr{\mbox{HSS}} و \lr{\mbox{USIM}} نگهداری می‌شود
\item
  از بازپخش بردارهای احراز هویت قدیمی جلوگیری می‌کند
\item
  \lr{\mbox{UE}} تأیید می‌کند: \lr{\mbox{SQN}} دریافتی \lr{\textgreater{}} آخرین \lr{\mbox{SQN}} پذیرفته‌شده (در یک پنجره)
\item
  اگر \lr{\mbox{SQN}} خارج از محدوده باشد \ensuremath{\leftarrow} \lr{\mbox{UE}} یک \lr{\mbox{\textbf{\lr{\mbox{AUTS}}}}} (نشانه همگام‌سازی مجدد) می‌فرستد
\item
  \lr{\mbox{HSS}} از \lr{\mbox{f1*}} و \lr{\mbox{f5*}} برای همگام‌سازی مجدد \lr{\mbox{SQN}} خود با \lr{\mbox{USIM}} استفاده می‌کند
\end{itemize}

\CNsubsection{}{تحویل بردار به \lr{\mbox{MME}}}

\begin{itemize}

  اندازه دسته معمول: \CNnum{3-5} بردار در هر درخواست \lr{\mbox{AIR}}
\end{itemize}

\lr{\textbf{\lr{AIR (Authentication-Information-Request):}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{IMSI}} مشترک
\item
  شناسه \lr{\mbox{PLMN}} بازدیدشده
\item
  تعداد بردارهای درخواستی
\item
  اطلاعات همگام‌سازی مجدد (\lr{\mbox{AUTS}}، در صورت عدم تطابق \lr{\mbox{SQN}})
\end{itemize}

\lr{\textbf{\lr{AIA (Authentication-Information-Answer):}}}

\begin{itemize}
\tightlist
\item
  یک یا چند بردار احراز هویت \lr{\mbox{EPS}} (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}}، \lr{\mbox{XRES}}، \lr{\mbox{\ensuremath{K_{\mathrm{ASME}}}}})
\item
  کد نتیجه (موفقیت/\allowbreak{}شکست)
\end{itemize}

\lr{\textbf{\lr{ULR (Update-Location-Request):}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{IMSI}}
\item
  هویت \lr{\mbox{MME}} (نام میزبان + \lr{\mbox{realm}})
\item
  نوع \lr{\mbox{RAT}}
\item
  پرچم‌های \lr{\mbox{ULR}} (اتصال اولیه و غیره)
\end{itemize}

\lr{\textbf{\lr{ULA (Update-Location-Answer):}}}

\begin{itemize}
\tightlist
\item
  داده اشتراک (تمام \lr{\mbox{APN}}ها، \lr{\mbox{AMBR}}، پروفایل‌های \lr{\mbox{QoS}})
\item
  کد نتیجه
\item
  جداسازی \lr{\mbox{APN}} پیش‌فرض از \lr{\mbox{APN}}های اضافی
\end{itemize}

\CNsubsection{}{جداسازی /\allowbreak{} پاک‌سازی (\lr{PUR/\allowbreak{}PUA} یا \lr{CLR/\allowbreak{}CLA})}

\textbf{جداسازی صریح (با شروع \lr{\mbox{UE}}):}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{MME}} یک \lr{\mbox{\textbf{\lr{\mbox{PUR}}}}} می‌فرستد \ensuremath{\leftarrow} \lr{\mbox{HSS}} مشترک را «متصل‌نشده» علامت‌گذاری می‌کند
\item
  \lr{\mbox{HSS}} با \lr{\mbox{\textbf{\lr{\mbox{PUA}}}}} پاسخ می‌دهد
\end{itemize}

\textbf{لغو با شروع شبکه (\lr{\mbox{HSS}} ارسال می‌کند):}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{HSS}} یک \lr{\mbox{\textbf{\lr{\mbox{CLR}}}}} به \lr{\mbox{MME}} قدیمی می‌فرستد (مثلاً \lr{\mbox{UE}} در \lr{\mbox{MME}} جدید ثبت‌نام کرده است)
\item
  \lr{\mbox{MME}} قدیمی زمینه \lr{\mbox{UE}} را حذف می‌کند، با \lr{\mbox{\textbf{\lr{\mbox{CLA}}}}} پاسخ می‌دهد
\item
  دلایل \lr{\mbox{CLR}}: \lr{\mbox{UE}} جابه‌جا شده، اشتراک لغو شده، سیاست اپراتور
\end{itemize}

\CNsubsection{}{اصلاح اشتراک \lr{(IDR/\allowbreak{}IDA)}}

وقتی اپراتور یک اشتراک را اصلاح می‌کند (مثلاً مشتری طرح خود را ارتقا می‌دهد):

\begin{itemize}
\tightlist
\item
  \lr{\mbox{HSS}} یک \lr{\textbf{\lr{\mbox{IDR}}} (Insert-Subscriber-Data-Request)} به \lr{\mbox{MME}} خدمات‌دهنده می‌فرستد
\item
  \lr{\mbox{MME}} نسخه محلی داده اشتراک را به‌روزرسانی می‌کند
\item
  \lr{\mbox{MME}} با \lr{\mbox{\textbf{\lr{\mbox{IDA}}}}} پاسخ می‌دهد
\item
  تغییرات بلافاصله اعمال می‌شوند (نیازی به اتصال مجدد نیست)
\end{itemize}

\CNsubsection{}{هویت \lr{\mbox{5G}}: \lr{\mbox{SUPI}} و \lr{\mbox{SUCI}}}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
SUPI (Subscription Permanent Identifier)
  = IMSI (same format as 4G: MCC+MNC+MSIN)
  (never sent over the air in 5G!)

SUCI (Subscription Concealed Identifier)
  = Home Network ID + Routing Indicator + Protection Scheme + Encrypted MSIN
  
  Encryption: ECIES (Elliptic Curve Integrated Encryption Scheme)
  - Home network publishes public key
  - UE encrypts MSIN portion with home network public key
  - Only home network (UDM) can decrypt → reveals SUPI
\end{Verbatim}
\end{CNplaincode}

\CNsection{}{خلاصه}

\lr{\mbox{HSS}} لنگر امنیت و هویت شبکه \lr{LTE/\allowbreak{}EPC} است. این سرور:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{از کلیدها محافظت می‌کند} \lr{—} \lr{\mbox{K}} هرگز از \lr{\mbox{HSS}} (و \lr{\mbox{USIM}}) خارج نمی‌شود
\item
  \textbf{هویت را اثبات می‌کند} \lr{—} احراز هویت متقابل از طریق \lr{\mbox{EPS-AKA}}
\item
  \textbf{دسترسی را کنترل می‌کند} \lr{—} پروفایل‌های اشتراک تعیین می‌کنند کاربران چه کاری می‌توانند انجام دهند
\item
  \textbf{مکان را ردیابی می‌کند} \lr{—} می‌داند کدام \lr{\mbox{MME}} به هر مشترک خدمات می‌دهد
\item
  \textbf{به‌صورت تدریجی تحول می‌یابد} \lr{—} در \lr{\mbox{5G}} به \lr{UDM/\allowbreak{}AUSF/\allowbreak{}UDR} برای استقرار بومی-ابر تجزیه می‌شود
\end{enumerate}

درک \lr{\mbox{HSS}} برای موارد زیر حیاتی است:

\begin{itemize}
\tightlist
\item
  تحلیل امنیت شبکه
\item
  اشکال‌زدایی شکست‌های احراز هویت
\item
  برنامه‌ریزی تأمین مشترکین
\item
  درک گذار از هسته \lr{\mbox{4G}} به \lr{\mbox{5G}}
\end{itemize}
