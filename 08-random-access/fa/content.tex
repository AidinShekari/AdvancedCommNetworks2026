\item
  \lr{$N_{ZC}$} = طول توالی (839 برای بلند، 139 برای کوتاه)
\end{itemize}

\CNsubsection{64}{پیشوند در هر سلول}

هر سلول دقیقاً \textbf{64 پیشوند در دسترس} دارد که از موارد زیر مشتق می‌شود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{شاخص توالی ریشه} (\lr{\CNcode{rootSequenceIndex}}، پخش‌شده در \lr{\mbox{SIB2}})
\item
  \textbf{شیفت‌های چرخه‌ای} اعمال‌شده بر توالی‌های ریشه
\end{enumerate}

\CNdisplay{\mathrm{Preamble}(v) = \mathrm{CyclicShift}\left(x_u,\; v \times N_{CS}\right)}

که در آن:

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\CNcode{N\_CS}}} = اندازه شیفت چرخه‌ای \lr{(\CNcode{zeroCorrelationZoneConfig})}
\item
  تعداد پیشوند در هر ریشه = \lr{$\lfloor N_{\mathrm{ZC}} / N_{\mathrm{CS}} \rfloor$}
\item
  اگر یک ریشه 64 پیشوند فراهم نکند، چندین ریشه استفاده می‌شود
\end{itemize}

\CNsubsection{}{پوشش سلول و شیفت چرخه‌ای}

شیفت چرخه‌ای \lr{\mbox{\CNcode{N\_CS}}} \textbf{ناحیه همبستگی صفر} را تعیین می‌کند که باید از حداکثر تأخیر رفت-و-برگشت فراتر رود:

\CNdisplay{N_{CS} > \frac{2\, d_{\max}}{c\, T_{s,\mathrm{seq}}}}

که در آن \lr{$d_{\max}$} بیشینه شعاع سلول، \lr{$c$} سرعت نور \lr{($3 \times 10^{8}$ m/\allowbreak{}s)} و \lr{$T_{s,\mathrm{seq}}$} مدت هر نمونه دنباله است.

\textbf{مثال:} برای \lr{\ensuremath{N_{\mathrm{CS}}} = 13} (قالب 0):

\begin{itemize}
\tightlist
\item
  حداکثر تأخیر قابل تشخیص = \lr{$13 \times (1/1.25\,\mathrm{MHz}) = 10.4\ \mu\mathrm{s}$}
\item
  حداکثر شعاع سلول = \lr{$10.4\ \mu\mathrm{s} \times c/2 \approx 1.56\ \mathrm{km}$}
\item
  پیشوند در هر ریشه = \lr{$\lfloor 839/13 \rfloor = 64$} \ensuremath{\leftarrow} یک ریشه کافی است
\end{itemize}

\CNsubsection{}{شعاع سلول از پیشروی زمانی}

حداکثر \lr{\mbox{TA}} \textbf{تأخیر انتشار رفت-و-برگشت} را تصحیح می‌کند:

\CNdisplay{
\begin{aligned}
\text{\rl{تأخیر رفت‌وبرگشت}} &= \mathrm{TA} \times 16 \times T_s\\
\text{\rl{تأخیر یک‌طرفه}} &= \mathrm{TA} \times 16 \times T_s / 2\\
\text{\rl{بیشینه فاصله}} &= c \times \text{\rl{تأخیر یک‌طرفه}}
\end{aligned}
}

\item
  تأخیر \lr{\mbox{RACH 2}}-گام در \lr{\mbox{NR}} را با \lr{\mbox{RACH 4}}-گام در \lr{\mbox{LTE}} مقایسه کنید. چه زمانی هر یک را انتخاب می‌کنید؟
\end{enumerate}
