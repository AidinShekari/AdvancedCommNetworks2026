\textbf{شبکه‌بندی} این را با اجازه دادن به \textbf{یک شبکه فیزیکی} برای خدمت‌دهی به تمام این نیازمندی‌های متنوع به‌طور همزمان از طریق چندین شبکه منطقی حل می‌کند، که هر یک برای هدف خاص خود بهینه شده است.

\textbf{محرک‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{کارایی اقتصادی}: اشتراک‌گذاری زیرساخت گران (طیف، برج‌ها، فیبر) در بین خدمات
\item
  \textbf{تمایز خدمات}: ارایه \lr{\mbox{SLA}}های متناسب به صنایع مختلف (خودروسازی، بهداشت، صنعت)
\item
  \textbf{چابکی عملیاتی}: استقرار خدمات جدید بدون ساخت شبکه‌های فیزیکی جدید
\item
  \textbf{چند-مستأجری}: امکان فعالیت \lr{\mbox{MVNO}}ها و شرکت‌ها در شبکه‌های مجازی جدا
\end{itemize}

\CNsection{2}{مفهوم شبکه‌بندی}

یک \textbf{برش شبکه} یک شبکه منطقی کامل و سرتاسری است که بر پایه زیرساخت فیزیکی مشترک ساخته می‌شود.

\CNsubsection{2.1}{دامنه سرتاسری}

هر برش کل شبکه را در بر می‌گیرد:

\begin{itemize}
\tightlist
\item
  \textbf{برش \lr{\mbox{RAN}}}: تخصیص منابع رادیویی، سیاست‌های زمان‌بندی، پیکربندی شکل‌دهی پرتو
\item
  \textbf{برش انتقال}: تضمین‌های پهنای باند، مسیرهای مسیریابی، کران‌های تأخیر در بک‌هال/\allowbreak{}مید‌هال
\item
  \textbf{برش هسته}: توابع شبکه اختصاصی یا مشترک (\lr{\mbox{SMF}}، \lr{\mbox{UPF}}، \lr{\mbox{PCF}} و غیره)
\end{itemize}

\CNsubsection{2.2}{جداسازی بین برش‌ها}

سه بُعد جداسازی:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{جداسازی منابع}: طیف، محاسبه، ذخیره‌سازی و پهنای باند اختصاصی در هر برش
\item
  \textbf{جداسازی امنیتی}: زمینه‌های امنیتی جدا، سلسله‌مراتب کلید، کنترل‌های دسترسی
\item
  \textbf{جداسازی شکست}: یک خطا در یک برش به دیگری منتشر نمی‌شود
\end{enumerate}

\CNsubsection{2.3}{سفارشی‌سازی در هر برش}

هر برش می‌تواند به‌طور مستقل سفارشی شود:

\begin{itemize}
\tightlist
\item
  \textbf{توابع شبکه}: انتخاب کنید کدام \lr{\mbox{NF}}ها مستقر شوند (مثلاً \lr{\mbox{URLLC}} ممکن است از برخی \lr{\mbox{NF}}ها صرف‌نظر کند)
\item
  \textbf{توپولوژی}: مکان‌یابی \lr{\mbox{NF}}ها را تعریف کنید (مثلاً \lr{\mbox{UPF}} لبه برای \lr{\mbox{URLLC}})
\item
  \textbf{پارامترهای \lr{\mbox{QoS}}}: مقادیر \lr{\mbox{5QI}}، \lr{\mbox{AMBR}}، سطوح اولویت مختص برش را پیکربندی کنید
\item
  \textbf{سیاست‌ها}: صورت‌حساب، کنترل دسترسی و قواعد مدیریت تحرک مستقل
\end{itemize}

  \lr{\textbf{\lr{S-NSSAI = SST + SD}}} شناسه بنیادی برش است؛ \lr{\mbox{UE}} مقدار \lr{\mbox{NSSAI}} درخواستی را ارایه می‌دهد، شبکه \lr{\mbox{NSSAI}} مجاز را بازمی‌گرداند
\item
  \textbf{\lr{\mbox{NSSF}} مغز انتخاب برش است} \lr{—} تعیین می‌کند \lr{\mbox{UE}} به کدام برش‌ها دسترسی دارد و کدام \lr{\mbox{AMF}} باید به آن خدمات دهد
\item
  \textbf{جداسازی چند-بُعدی است} \lr{—} جداسازی منبع، امنیت و شکست همه باید مورد توجه قرار گیرند
\item
  \textbf{شبکه‌بندی \lr{\mbox{RAN}} نیازمند زمان‌بندی دو-سطحی است} \lr{—} تخصیص بین-برشی و سپس زمان‌بندی درون-برشی در هر \lr{\mbox{UE}}
\item
  \textbf{شبکه‌بندی \lr{\mbox{\ensuremath{\neq} QoS}}} \lr{—} شبکه‌بندی جداسازی شبکه منطقی سرتاسری فراهم می‌کند؛ \lr{\mbox{QoS}} رفتار در هر جریان را درون یک برش فراهم می‌کند
\item
  \textbf{مدیریت چرخه عمر} ایجاد تا ازکاراندازی برش را خودکار می‌کند، با محرک مشخصات مبتنی بر قصد
\end{enumerate}
