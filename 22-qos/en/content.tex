\CNsubsection{}{Without QoS:}

\begin{itemize}
\tightlist
\item
  Voice calls get choppy during congestion
\item
  Video freezes while background downloads continue
\item
  Emergency services compete with social media
\item
  Industrial control packets delayed by bulk transfers
\end{itemize}

\CNsubsection{}{QoS Goal:}

\textbf{Differentiate traffic and provide appropriate treatment to each service based on its requirements.}

\CNsubsection{}{2G (GSM): No Real QoS}

\begin{itemize}
\tightlist
\item
  All services are \textbf{circuit-switched}
\item
  Each call gets a dedicated timeslot \ensuremath{\to} same quality for everyone
\item
  No differentiation needed (only voice + SMS)
\end{itemize}

\CNsubsection{}{LTE (4G): Bearer-Based QoS}

\begin{itemize}
\tightlist
\item
  Introduced \textbf{EPS bearers} with specific QoS parameters
\item
  \textbf{QCI} (QoS Class Identifier): standardized classes
\item
  \textbf{ARP} (Allocation and Retention Priority)
\item
  \textbf{GBR} (Guaranteed Bit Rate) vs \textbf{Non-GBR} bearers
\end{itemize}

\begin{itemize}
\tightlist
\item
  \textbf{QoS Flows} identified by \textbf{QFI} (QoS Flow Identifier)
\item
  \textbf{5QI} replaces QCI with more values
\item
  \textbf{Reflective QoS} for simplified UL treatment
\item
  Finer granularity: multiple flows per PDU session
\end{itemize}

\textbf{Example Scenario:}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
Congestion situation - resources are full:

Bearer A: Priority=5, Pre-emption Capability=Yes
Bearer B: Priority=10, Pre-emption Vulnerability=Yes

→ Bearer A can preempt (remove) Bearer B to get resources
→ Emergency calls (Priority=1) can preempt regular calls
\end{Verbatim}
\end{CNplaincode}

\CNsubsection{}{LTE QoS Limitations}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Coarse granularity}: One QCI per bearer \ensuremath{\to} one bearer per service type
\item
  \textbf{Bearer overhead}: Each new QCI requires a new GTP tunnel
\item
  \textbf{Signaling heavy}: Dedicated bearer setup requires full NAS/\allowbreak{}S1AP signaling
\item
  \textbf{Inflexible}: Cannot mix traffic with different QoS in one bearer
\item
  \textbf{Limited QCIs}: Only 9 standard values (extended to \textasciitilde{}20 with MCX)
\end{enumerate}

\CNsubsection{}{QoS Rules}

QoS rules tell the UE how to classify uplink traffic:

\CNfigure{m22-qos-rule}{Structure of a QoS rule, with example rules}

\CNsubsection{}{Reflective QoS}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  The network sends a DL packet with QFI = X and RQI = 1 (Reflective QoS Indication).
\item
  The UE observes the DL packet headers:

  \begin{itemize}
  \tightlist
  \item
    source IP/\allowbreak{}port \ensuremath{\to} becomes the UL destination
  \item
    destination IP/\allowbreak{}port \ensuremath{\to} becomes the UL source
  \item
    QFI = X \ensuremath{\to} the UE uses the same QFI for its reply
  \end{itemize}
\item
  The UE creates a “derived QoS rule” automatically:

  \begin{itemize}
  \tightlist
  \item
    no explicit QoS-rule signalling is needed
  \item
    the rule is timer-based: valid for the RQ timer duration
  \end{itemize}
\end{enumerate}

\textbf{Benefit:} less signalling to establish QoS rules. \textbf{Use case:} the UE does not know the server IP in advance.

\CNsubsection{}{Rate Parameters Explained}

\begin{itemize}
\tightlist
\item
  \textbf{GFBR = Guaranteed Flow Bit Rate}

  \begin{itemize}
  \tightlist
  \item

  \item

  \end{itemize}
\item
  \textbf{MFBR = Maximum Flow Bit Rate}

  \begin{itemize}
  \tightlist
  \item
    the network MAY provide up to this rate
  \item
    traffic above MFBR is dropped or shaped
  \end{itemize}
\end{itemize}

  GBR flows are limited by their own MFBR, not AMBR
\item
  UE-AMBR \ensuremath{\ge} sum of Session-AMBRs (logically)
\item
  Enforced separately for UL and DL
\end{itemize}
