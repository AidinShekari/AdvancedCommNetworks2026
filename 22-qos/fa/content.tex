\CNsubsection{}{بدون \lr{\mbox{QoS}}:}

\begin{itemize}
\tightlist
\item
  تماس‌های صوتی در حین ازدحام بریده‌بریده می‌شوند
\item
  ویدیو متوقف می‌شود در حالی که دانلودهای پس‌زمینه ادامه دارند
\item
  خدمات اضطراری با شبکه‌های اجتماعی رقابت می‌کنند
\item
  بسته‌های کنترل صنعتی توسط انتقال‌های انبوه به تأخیر می‌افتند
\end{itemize}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
Congestion situation - resources are full:

Bearer A: Priority=5, Pre-emption Capability=Yes
Bearer B: Priority=10, Pre-emption Vulnerability=Yes

→ Bearer A can preempt (remove) Bearer B to get resources
→ Emergency calls (Priority=1) can preempt regular calls
\end{Verbatim}
\end{CNplaincode}

\begin{enumerate}

  \textbf{غیرانعطاف‌پذیر}: نمی‌توان ترافیک با \lr{\mbox{QoS}} متفاوت را در یک حامل ترکیب کرد

\end{enumerate}

\CNsubsection{}{قواعد \lr{\mbox{QoS}}}

قواعد \lr{\mbox{QoS}} به \lr{\mbox{UE}} می‌گویند چگونه ترافیک پیوند بالاسو را طبقه‌بندی کند:

\CNfigure{m22-qos-rule}{ساختار یک قاعده \lr{\mbox{QoS}} به همراه قواعد نمونه}

\CNsubsection{}{\lr{\mbox{QoS}} بازتابی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  شبکه یک بسته \lr{\mbox{DL}} با \lr{\mbox{QFI = X}} و \lr{\mbox{RQI = 1}} (نشانگر \lr{\mbox{QoS}} بازتابی) ارسال می‌کند.
\item
  \lr{\mbox{UE}} سرآیندهای بسته \lr{\mbox{DL}} را مشاهده می‌کند:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{IP}}/\allowbreak{}درگاه مبدأ \ensuremath{\leftarrow} به مقصد \lr{\mbox{UL}} تبدیل می‌شود
  \item
    \lr{\mbox{IP}}/\allowbreak{}درگاه مقصد \ensuremath{\leftarrow} به مبدأ \lr{\mbox{UL}} تبدیل می‌شود
  \item
    \lr{QFI = X \ensuremath{\leftarrow} UE} برای پاسخ از همان \lr{\mbox{QFI}} استفاده می‌کند
  \end{itemize}
\item
  \lr{\mbox{UE}} به‌طور خودکار یک «قاعده \lr{\mbox{QoS}} استخراج‌شده» می‌سازد:

  \begin{itemize}
  \tightlist
  \item
    نیازی به سیگنالینگ صریح قاعده \lr{\mbox{QoS}} نیست
  \item
    قاعده مبتنی بر زمان‌سنج است: به مدت زمان‌سنج \lr{\mbox{RQ}} معتبر است
  \end{itemize}
\end{enumerate}

\textbf{مزیت:} کاهش سیگنالینگ برای برقراری قواعد \lr{\mbox{QoS}}. \textbf{کاربرد:} وقتی \lr{\mbox{UE}} نشانی \lr{\mbox{IP}} سرور را از قبل نمی‌داند.
