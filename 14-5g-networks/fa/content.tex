\CNsection{1}{چرا \lr{\mbox{5G}}؟ \lr{—} محدودیت‌های \lr{LTE/\allowbreak{}4G}}

با وجود موفقیت \lr{\mbox{LTE (4G)}}، چند محدودیت بنیادی نیاز به یک نسل جدید را ایجاب کرد:

\CNsubsection{1.1}{کف تأخیر (حدود \lr{\mbox{10~ms}})}

\begin{itemize}
\tightlist
\item
  فاصله زمانی انتقال \lr{\mbox{(TTI)}} در \lr{\mbox{LTE}} به‌صورت ثابت \lr{\mbox{\textbf{\lr{\mbox{1~ms}}}}} زیرفریم است
\item
  با تأخیرهای زمان‌بندی، \lr{\mbox{HARQ}} و پردازش، \textbf{کف تأخیر رفت-و-برگشت عملی حدود \lr{\mbox{10~ms}}} است
\item
  این برای موارد زیر کافی نیست:

  \begin{itemize}
  \tightlist
  \item
    کنترل خودروی خودران (نیازمند کمتر از \lr{\mbox{5~ms}})
  \item
    جراحی از راه دور /\allowbreak{} بازخورد لمسی (نیازمند حدود \lr{\mbox{1~ms}})
  \item
    حلقه‌های کنترل بلادرنگ صنعتی
  \end{itemize}
\end{itemize}

\CNsubsection{1.2}{سقف ظرفیت}

\begin{itemize}
\tightlist
\item
  اوج \lr{\mbox{LTE}}: حدود \lr{\mbox{1~Gbps}} (\lr{\mbox{Cat-16}}، \lr{\mbox{3CA}}، \lr{\mbox{MIMO 4\ensuremath{\times}4}}، \lr{\mbox{256-QAM}}) \lr{—} نظری
\item
  توان عملیاتی عملی: \lr{100–300~Mbps} در شرایط ایده‌آل
\item
  طیف زیر \lr{\mbox{6~GHz}} اشباع است؛ پهنای باند محدود در هر حامل (حداکثر \lr{\mbox{20~MHz}})
\item
  \lr{\mbox{MIMO 4\ensuremath{\times}4}} حد عملی آنتن در قالب‌های \lr{\mbox{LTE}} است
\end{itemize}

\CNsubsection{1.3}{مقیاس \lr{\mbox{IoT}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{LTE}} برای ارتباط پهن‌باند \textbf{انسان‌محور} طراحی شده بود
\item
  \lr{LTE-M /\allowbreak{} NB-IoT} پشتیبانی \lr{\mbox{IoT}} را اضافه کردند، اما:

  \begin{itemize}
  \tightlist
  \item
    چگالی اتصال به حدود \lr{\mbox{10\textsuperscript{4}}} دستگاه/\allowbreak{}\lr{\mbox{km\textsuperscript{2}}} محدود است
  \item
    به‌طور بومی برای \textbf{\lr{\mbox{10\textsuperscript{6}}} دستگاه/\allowbreak{}\lr{\mbox{km\textsuperscript{2}}}} مورد نیاز شهر هوشمند/\allowbreak{}\lr{\mbox{IoT}} صنعتی طراحی نشده است
  \item
    سربار سیگنالینگ در هر دستگاه برای استقرارهای انبوه بیش از حد بالاست
  \end{itemize}
\end{itemize}

\CNsection{2}{نیازمندی‌های \lr{\mbox{5G}} \lr{—} اهداف \lr{\mbox{IMT-2020}} اتحادیه مخابرات بین‌المللی}

اتحادیه مخابرات بین‌المللی \lr{(ITU) \textbf{\lr{\mbox{IMT-2020}}}} را به‌عنوان چارچوب \lr{\mbox{5G}} با شاخص‌های کلیدی عملکرد \lr{\mbox{(KPI)}} زیر تعریف کرد:

\CNsubsection{2.1}{حداکثر نرخ داده}

\begin{itemize}
\tightlist
\item
  \textbf{پیوند پایین‌سو}: \lr{\mbox{20~Gbps}}
\item
  \textbf{پیوند بالاسو}: \lr{\mbox{10~Gbps}}
\item
  از طریق پهنای باند بیشتر (تا \lr{\mbox{400~MHz}} در هر حامل) + \lr{\mbox{MIMO}} مرتبه بالاتر + \lr{\mbox{256-QAM}} به دست می‌آید
\end{itemize}

\CNsubsection{2.2}{تأخیر}

\begin{itemize}
\tightlist
\item
  \textbf{تأخیر صفحه کاربر}: \lr{\mbox{\textbf{\lr{\mbox{1~ms}}}}} (سناریوی \lr{\mbox{URLLC}})
\item
  \textbf{تأخیر صفحه کنترل}: \lr{\mbox{20~ms}} (بیکار تا متصل)
\item
  کاربردهای کنترل بلادرنگ را که قبلاً روی شبکه سلولی غیرممکن بود ممکن می‌سازد
\end{itemize}

\CNsubsection{2.3}{چگالی اتصال}

\begin{itemize}
\tightlist
\item
  \textbf{\lr{\mbox{10\textsuperscript{6}}} دستگاه در هر \lr{\mbox{km\textsuperscript{2}}}} (سناریوی \lr{\mbox{mMTC}})
\item
  بهبود 100 برابری نسبت به \lr{\mbox{LTE}}
\item
  پشتیبانی از حسگرها، محرک‌ها، کنتورها با حداقل سربار
\end{itemize}

\CNsubsection{2.4}{نرخ داده تجربه‌شده توسط کاربر}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{100~Mbps}}}}} در همه‌جا در دسترس (لبه سلول، شهری متراکم)
\item
  \textbf{\lr{\mbox{50~Mbps}} در \lr{\mbox{UL}}} در همه‌جا
\item
  نمایانگر «حداقل تضمین‌شده» تجربه کاربر است
\end{itemize}

\CNsubsection{2.5}{قابلیت اطمینان}

\begin{itemize}
\tightlist
\item
  \textbf{\CNnum{99.999\%}} (پنج نه) برای \lr{\mbox{URLLC}}
\item
  تحویل بسته درون بودجه تأخیر با احتمال خطای \lr{\mbox{10\textsuperscript{-5}}}
\item
  حیاتی برای کاربردهای مرتبط با ایمنی جان
\end{itemize}

\CNsection{4}{\lr{\mbox{5G NR}} (رادیو جدید) \lr{—} لایه فیزیکی}

\lr{\mbox{5G NR}} فناوری دسترسی رادیویی تعریف‌شده در نسخه \lr{\mbox{15}}+ گروه \lr{\mbox{3GPP}} است که از پایه برای انعطاف‌پذیری طراحی شده است.

\CNsubsection{4.1}{عددشناسی انعطاف‌پذیر}

برخلاف \lr{\mbox{LTE}} (\lr{\mbox{SCS}} ثابت \lr{\mbox{15 kHz}}، \lr{\mbox{TTI 1 ms}})، \lr{\mbox{NR}} از \textbf{فاصله‌های زیرحامل \lr{\mbox{(SCS)}} متعدد} پشتیبانی می‌کند:

\begin{longtable}[c]{@{}ccccl@{}}
\toprule\noalign{}
\rowcolor{CNink}\CNth{عددشناسی \lr{\mbox{($\mu$)}}} & \CNth{\lr{SCS (kHz)}} & \CNth{مدت شکاف} & \CNth{نوع \lr{\mbox{CP}}} & \CNth{کاربرد معمول} \\
\CNheadrulesep
\endhead
\bottomrule\noalign{}
\endlastfoot

\end{longtable}

\textbf{روابط کلیدی}:

\begin{itemize}
\tightlist
\item
  مدت شکاف = \lr{$1\ \mathrm{ms}/2^{\mu}$}
\item
  نماد در هر شکاف = \lr{\mbox{14}} (\lr{\mbox{CP}} معمولی) یا 12 (\lr{\mbox{CP}} گسترده)
\item
  \lr{\mbox{SCS}} بالاتر \ensuremath{\leftarrow} شکاف کوتاه‌تر \ensuremath{\leftarrow} تأخیر کمتر (اما زیرحامل پهن‌تر \ensuremath{\leftarrow} تحمل نویز فاز بیشتر مورد نیاز برای \lr{\mbox{mmWave}})
\end{itemize}

\CNsubsection{4.3}{بخش‌های پهنای باند \lr{\mbox{(BWP)}}}

\begin{itemize}
\tightlist
\item
  یک \textbf{بخش پهنای باند} یک زیرمجموعه پیوسته از پهنای باند حامل است
\item
  \lr{\mbox{UE}} می‌تواند تا \textbf{\lr{\mbox{4 BWP}} در هر سلول خدمات‌دهنده} پیکربندی شود (1 فعال در هر زمان)
\item
  صرفه‌جویی در توان \lr{\mbox{UE}} را ممکن می‌سازد: \lr{\mbox{BWP}} باریک برای پایش، \lr{\mbox{BWP}} پهن برای داده
\item
  هر \lr{\mbox{BWP}} می‌تواند عددشناسی خود را داشته باشد (\lr{\mbox{SCS}}، \lr{\mbox{CP}})
\end{itemize}

\CNsubsection{4.4}{مینی-اسلات‌ها}

\begin{itemize}
\tightlist
\item
  شکاف استاندارد \lr{\mbox{NR = 14}} نماد
\item
  \textbf{مینی-اسلات} = \lr{\mbox{2}}، 4 یا 7 نماد \lr{\mbox{OFDM}}
\item
  زمان‌بندی زیر-شکاف را برای \lr{\mbox{\textbf{\lr{\mbox{URLLC}}}}} بدون انتظار برای مرز شکاف ممکن می‌سازد
\item
  می‌تواند ارسال‌های در جریان \lr{\mbox{eMBB}} را برای ترافیک فوری \lr{\mbox{URLLC}} \textbf{پیش‌گیری} کند
\end{itemize}

\CNsection{5}{\lr{\mbox{NSA}} در برابر \lr{\mbox{SA}} \lr{—} راهبردهای استقرار}

\CNchained\CNsubsection{5.1}{غیرمستقل \lr{\mbox{(NSA)}} \lr{—} گزینه \lr{3/\allowbreak{}3a/\allowbreak{}3x}}

\textbf{معماری}: رادیو \lr{\mbox{NR (gNB)}} + \textbf{هسته \lr{LTE (EPC)}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{eNB}} در \lr{\mbox{LTE}} به‌عنوان \textbf{گره اصلی} (لنگر) عمل می‌کند
\item
  \lr{\mbox{gNB}} در \lr{\mbox{NR}} \textbf{گره ثانویه} است (ظرفیت اضافه می‌کند)
\item
  صفحه کنترل از \lr{\mbox{LTE}} عبور می‌کند \lr{(eNB \ensuremath{\to} EPC)}
\item
  صفحه کاربر می‌تواند تقسیم یا سوییچ شود
\end{itemize}

\textbf{زیرگزینه‌ها}:

\begin{itemize}
\tightlist
\item
  \textbf{گزینه 3}: صفحه کاربر از طریق \lr{\mbox{eNB}} (تقسیم در \lr{\mbox{eNB}})
\item
  \textbf{گزینه \lr{\mbox{3a}}}: صفحه کاربر مستقیماً از \lr{\mbox{gNB}} به \lr{\mbox{EPC}}
\item
  \textbf{گزینه \lr{\mbox{3x}}}: صفحه کاربر در \lr{\mbox{gNB}} تقسیم می‌شود (\lr{\mbox{gNB}} داده \lr{\mbox{NR}} + بخشی از داده \lr{\mbox{LTE}} را مدیریت می‌کند)
\end{itemize}

\CNsubsection{5.2}{مستقل \lr{\mbox{(SA)}} \lr{—} گزینه 2}

\textbf{معماری}: رادیو \lr{\mbox{NR (gNB)}} + \textbf{هسته \lr{\mbox{5G (5GC)}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{gNB}} مستقیماً از طریق \textbf{واسط \lr{\mbox{NG}}} به هسته \lr{\mbox{5G}} متصل می‌شود
\item
  تمام قابلیت‌های \lr{\mbox{5G}} در دسترس است
\item
  بدون وابستگی به زیرساخت \lr{\mbox{LTE}}
\end{itemize}

\CNsubsection{5.3}{چرا \lr{\mbox{NSA}} ابتدا مستقر شد}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{زمان سریع‌تر به بازار}: استفاده مجدد از زیرساخت \lr{\mbox{EPC}} موجود
\item
  \textbf{\lr{\mbox{CAPEX}} کمتر}: نیاز به استقرار فوری هسته \lr{\mbox{5G}} نیست
\item
  \textbf{لنگر پوشش}: \lr{\mbox{LTE}} پوشش همه‌جاگیر فراهم می‌کند؛ \lr{\mbox{NR}} توان عملیاتی اضافه می‌کند
\item
  \textbf{هسته اثبات‌شده}: \lr{\mbox{EPC}} بالغ و شناخته‌شده است
\item
  \textbf{راهبرد طیف}: استقرارهای اولیه \lr{\mbox{NR}} در باندهای محدود
\end{enumerate}

\CNsubsection{5.4}{چرا \lr{\mbox{SA}} هدف است}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{شبکه‌بندی}: نیازمند هسته \lr{\mbox{5G}} مبتنی بر \lr{\mbox{SBA}} است
\item
  \lr{\mbox{\textbf{\lr{\mbox{URLLC}}}}}: تأخیر کم سرتاسری نیازمند \lr{5GC + MEC} است
\item
  \lr{\mbox{\textbf{\lr{\mbox{mMTC}}}}}: پشتیبانی بومی \lr{\mbox{5GC}} برای بهینه‌سازی‌های \lr{\mbox{IoT}}
\item
  \textbf{رایانش لبه}: انعطاف‌پذیری مکان \lr{\mbox{UPF}}
\item
  \textbf{استقلال}: بدون وابستگی به \lr{\mbox{LTE}}، معماری تمیزتر
\item
  \textbf{معماری خدمات‌محور}: ایجاد سریع خدمات را ممکن می‌سازد
\end{enumerate}

  \textbf{سه ستون} \lr{(eMBB/\allowbreak{}URLLC/\allowbreak{}mMTC)} به صنایع مختلف با \lr{\mbox{KPI}}های مختلف خدمات می‌دهند
\item
  \lr{\mbox{\textbf{\lr{\mbox{5G NR}}}}} عددشناسی انعطاف‌پذیر، پشتیبانی \lr{\mbox{mmWave}} و شکل‌دهی پرتو پویا را معرفی می‌کند
\item
  \textbf{\lr{\mbox{NSA}} استقرار سریع را ممکن می‌سازد} با استفاده مجدد از \lr{LTE/\allowbreak{}EPC}؛ \textbf{\lr{\mbox{SA}} پتانسیل کامل \lr{\mbox{5G}} را آزاد می‌کند}
\item
  \textbf{هسته \lr{\mbox{5G}} مبتنی بر \lr{\mbox{SBA}}} بومی-ابر و مبتنی بر میکروسرویس است و شبکه‌بندی و لبه را ممکن می‌سازد
\item
  \textbf{تجزیه \lr{\mbox{gNB}}} \lr{(CU/\allowbreak{}DU/\allowbreak{}RU)} استقرارهای انعطاف‌پذیر و مقیاس‌پذیر را ممکن می‌سازد
\item
  \textbf{نسخه‌های \lr{\mbox{3GPP}}} به‌تدریج ویژگی‌ها را اضافه می‌کنند: نسخه ۱۵: پایه؛ نسخه ۱۶: \lr{\mbox{URLLC}}؛ نسخه ۱۷: گسترش؛ نسخه ۱۸: \lr{5G-Advanced}
\end{enumerate}
