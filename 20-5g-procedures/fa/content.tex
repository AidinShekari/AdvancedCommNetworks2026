\CNsection{1}{ثبت‌نام (ثبت‌نام اولیه)}

\CNchained\CNsubsection{}{هدف}

رویه ثبت‌نام اولیه به \lr{\mbox{UE}} امکان می‌دهد برای اولین بار یا پس از لغو ثبت‌نام، در شبکه هسته \lr{\mbox{5G (5GC)}} ثبت‌نام کند. حضور \lr{\mbox{UE}} در شبکه را برقرار می‌کند، مشترک را احراز هویت می‌کند، یک \lr{\mbox{5G-GUTI}} (شناسه موقت یکتای جهانی) تخصیص می‌دهد و به‌صورت اختیاری یک نشست \lr{\mbox{PDU}} برای اتصال داده برقرار می‌کند.

\CNsubsection{}{پیش‌شرط‌ها}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} روشن است و یک \lr{\mbox{USIM}} معتبر با اعتبارنامه‌های اشتراک \lr{\mbox{5G}} دارد
\item
  \lr{\mbox{UE}} یک \lr{\mbox{PLMN}} انتخاب کرده و روی یک سلول \lr{\mbox{(gNB)}} کمپ کرده است
\item
  \lr{\mbox{UE}} در حالت \lr{RRC\_\allowbreak{}IDLE} یا \lr{RRC\_\allowbreak{}INACTIVE} است
\item
  \lr{\mbox{UE}} هیچ \lr{\mbox{5G-GUTI}} معتبری ندارد (ثبت‌نام بار اول) یا \lr{\mbox{GUTI}} قبلی منقضی شده است
\item
  اتصال \lr{\mbox{RRC}} برقرار شده است (رویه دسترسی تصادفی کامل شده)
\end{itemize}

\CNsubsection{}{جریان گام-به-گام}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{برقراری اتصال \lr{\mbox{RRC}}}: \lr{\mbox{UE}} دسترسی تصادفی انجام می‌دهد و \lr{RRCSetupRequest} را به \lr{\mbox{gNB}} می‌فرستد
\item
  \textbf{راه‌اندازی \lr{\mbox{RRC}}}: \lr{\mbox{gNB}} با پیام \lr{\mbox{RRCSetup}} پاسخ می‌دهد
\item
  \textbf{درخواست ثبت‌نام}: \lr{\mbox{UE}} یک \lr{RRCSetupComplete} حاوی \lr{NAS Registration Request} می‌فرستد (شامل \lr{\mbox{SUCI}} یا \lr{\mbox{5G-GUTI}}، نوع ثبت‌نام، قابلیت‌های امنیتی \lr{\mbox{UE}}، \lr{\mbox{NSSAI}} درخواستی)
\item
  \textbf{انتخاب \lr{\mbox{AMF}}}: \lr{\mbox{gNB AMF}} مناسب را انتخاب می‌کند (در صورت نیاز از طریق \lr{\mbox{NSSF}}) و از طریق \lr{NGAP Initial UE Message} ارسال می‌کند
\item
  \textbf{درخواست هویت (شرطی)}: اگر \lr{\mbox{AMF}} نتواند هویت را حل کند، \lr{Identity Request} را به \lr{\mbox{UE}} می‌فرستد
\item
  \textbf{حل \lr{\mbox{SUCI}}}: \lr{\mbox{AMF}} احراز هویت را آغاز می‌کند؛ \lr{\mbox{SUCI}} را به \lr{\mbox{AUSF}} می‌فرستد
\item
  \textbf{درخواست بردار احراز هویت}: \lr{\mbox{AUSF}} بردار احراز هویت را از \lr{UDM/\allowbreak{}ARPF} درخواست می‌کند
\item
  \textbf{تولید \lr{\mbox{AV}}}: \lr{UDM/\allowbreak{}ARPF} یک \lr{\mbox{5G HE AV}} تولید می‌کند (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}}، \lr{\mbox{XRES*}}، \lr{\mbox{\ensuremath{K_{\mathrm{AUSF}}}}})، \lr{\mbox{SUCI}} را به \lr{\mbox{SUPI}} حل می‌کند
\item
  \textbf{پاسخ بردار احراز هویت}: \lr{\mbox{AUSF}} بردار را دریافت می‌کند، \lr{\mbox{HXRES*}} را محاسبه می‌کند، \lr{\mbox{XRES*}} را ذخیره می‌کند
\item
  \textbf{درخواست احراز هویت}: \lr{\mbox{AMF}} یک \lr{NAS Authentication Request} به \lr{\mbox{UE}} می‌فرستد (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}})
\item
  \textbf{تأیید \lr{\mbox{UE}}}: \lr{\mbox{UE}} مقدار \lr{\mbox{AUTN}} را تأیید می‌کند، \lr{\mbox{RES*}}، \lr{\mbox{CK}}، \lr{\mbox{IK}} را محاسبه و \lr{\mbox{\ensuremath{K_{\mathrm{AUSF}}}}} را مشتق می‌کند
\item
  \textbf{پاسخ احراز هویت}: \lr{\mbox{UE}} یک \lr{Authentication Response (RES*)} به \lr{\mbox{AMF}} می‌فرستد
\item
  \textbf{تأیید \lr{\mbox{RES}}}\emph{: \lr{\mbox{AMF}} مقدار \lr{\mbox{RES}}} را به \lr{\mbox{AUSF}} می‌فرستد؛ \lr{\mbox{AUSF}} تأیید می‌کند \lr{\mbox{HRES*}} با \lr{\mbox{HXRES*}} مطابقت دارد
\item
  \textbf{تأیید احراز هویت}: \lr{\mbox{AUSF}} به \lr{\mbox{AMF}} تأیید می‌کند، \lr{\mbox{\ensuremath{K_{\mathrm{SEAF}}}}} را فراهم می‌کند؛ \lr{\mbox{AMF}} مقدار \lr{\mbox{\ensuremath{K_{\mathrm{AMF}}}}} را مشتق می‌کند
\item
  \textbf{امنیت \lr{\mbox{NAS}}}: \lr{\mbox{AMF}} یک \lr{Security Mode Command} می‌فرستد (الگوریتم‌های انتخاب‌شده)؛ \lr{\mbox{UE}} با \lr{Security Mode Complete} پاسخ می‌دهد
\item
  \textbf{ثبت‌نام در \lr{\mbox{UDM}}}: \lr{\mbox{AMF}} در \lr{\mbox{UDM}} ثبت‌نام می‌کند \lr{(Nudm\_\allowbreak{}UECM\_\allowbreak{}Registration)}
\item
  \textbf{بازیابی اشتراک}: \lr{\mbox{AMF}} داده اشتراک را از \lr{\mbox{UDM}} بازیابی می‌کند \lr{(Nudm\_\allowbreak{}SDM\_\allowbreak{}Get)}
\item
  \textbf{انجمن سیاست}: \lr{\mbox{AMF}} انجمن سیاست \lr{\mbox{AM}} را با \lr{\mbox{PCF}} برقرار می‌کند
\item
  \textbf{پذیرش ثبت‌نام}: \lr{\mbox{AMF}} یک \lr{Registration Accept} می‌فرستد (\lr{\mbox{5G-GUTI}}، \lr{\mbox{TAI List}}، \lr{Allowed NSSAI}، ناحیه ثبت‌نام)
\item
  \textbf{تکمیل ثبت‌نام}: \lr{\mbox{UE}} با \lr{Registration Complete} تأیید می‌کند؛ \lr{\mbox{5G-GUTI}} جدید را ذخیره می‌کند
\end{enumerate}

\CNsection{2}{احراز هویت \lr{\mbox{(5G-AKA)}}}

\CNchained\CNsubsection{}{هدف}

رویه \lr{\mbox{5G-AKA}} (احراز هویت و توافق کلید) احراز هویت متقابل بین \lr{\mbox{UE}} و شبکه را فراهم می‌کند و کلیدهای رمزنگاری مشترک را برقرار می‌سازد. تضمین می‌کند شبکه هویت مشترک را تأیید می‌کند و \lr{\mbox{UE}} تأیید می‌کند که با شبکه‌ای مشروع در ارتباط است. این رویه کلید لنگر \lr{\mbox{(\ensuremath{K_{\mathrm{AUSF}}})}} را تولید می‌کند که تمام کلیدهای امنیتی بعدی از آن مشتق می‌شوند.

\CNsubsection{}{پیش‌شرط‌ها}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} ثبت‌نام را آغاز کرده یا احراز هویت مجدد فعال شده است
\item
  \lr{\mbox{UE}} یک \lr{\mbox{USIM}} معتبر با کلید بلندمدت \lr{\mbox{K}} و \lr{OP/\allowbreak{}OPc} دارد
\item
  \lr{\mbox{AUSF}} از \lr{\mbox{AMF}} خدمات‌دهنده قابل دسترس است
\item
  \lr{UDM/\allowbreak{}ARPF} اعتبارنامه‌های اشتراک معتبر برای مشترک دارد
\item
  \lr{\mbox{SUCI}} ارایه شده است (یا از \lr{\mbox{UE}} یا حل‌شده از \lr{\mbox{5G-GUTI}})
\end{itemize}

\CNsubsection{}{جریان گام-به-گام}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{محرک}: \lr{\mbox{AMF}} تعیین می‌کند احراز هویت لازم است (ثبت‌نام اولیه، احراز هویت مجدد دوره‌ای، یا تحویل بین-\lr{\mbox{AMF}})
\item
  \textbf{انتخاب \lr{\mbox{AUSF}}}: \lr{\mbox{AMF}} بر اساس شبکه خانگی \lr{\mbox{SUPI}} یک \lr{\mbox{AUSF}} انتخاب می‌کند (از طریق \lr{\mbox{NRF}})
\item
  \textbf{آغاز احراز هویت}: \lr{\mbox{AMF}} مقدار \lr{\mbox{SUCI}} و نام شبکه خدمات‌دهنده \lr{(SN-name)} را به \lr{\mbox{AUSF}} می‌فرستد
\item
  \textbf{درخواست \lr{\mbox{UDM}}}: \lr{\mbox{AUSF}} درخواست را با \lr{\mbox{SUCI}} و \lr{\mbox{SN-name}} به \lr{\mbox{UDM}} ارسال می‌کند
\item
  \textbf{رمزگشایی \lr{\mbox{SUCI}}}: \lr{\mbox{SIDF}} (درون \lr{\mbox{UDM}}) \lr{\mbox{SUCI}} را با کلید خصوصی شبکه خانگی رمزگشایی می‌کند \ensuremath{\leftarrow} \lr{\mbox{SUPI}} را به‌دست می‌آورد
\item
  \textbf{تولید \lr{\mbox{AV}}}: \lr{\mbox{ARPF}} یک \lr{\mbox{5G HE AV}} با استفاده از موارد زیر تولید می‌کند:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{RAND}} (چالش تصادفی)
  \item
    \lr{AUTN = SQN\ensuremath{\oplus}AK \textbar{}\textbar{} AMF \textbar{}\textbar{} MAC} (نشانه احراز هویت شبکه)
  \item
    \lr{XRES* = KDF(CK, IK, SN-name, RAND, RES)} (پاسخ مورد انتظار)
  \item
    \lr{\ensuremath{K_{\mathrm{AUSF}}} = KDF(CK, IK, SN-name, SQN\ensuremath{\oplus}AK)} (کلید لنگر)
  \end{itemize}
\item
  \textbf{\lr{\mbox{AV}} به \lr{\mbox{AUSF}}}: \lr{\mbox{UDM}} مقدار \lr{\mbox{5G HE AV}} را به \lr{\mbox{AUSF}} بازمی‌گرداند
\item
  \textbf{پردازش \lr{\mbox{AUSF}}}: \lr{\mbox{AUSF}} مقدار \lr{\mbox{XRES*}} را ذخیره می‌کند، \lr{HXRES* = SHA-256(XRES\emph{\lr{[128..255] \textbar{}\textbar{} XRES}})} را محاسبه می‌کند، \lr{\mbox{\ensuremath{K_{\mathrm{SEAF}}}}} را از \lr{\mbox{\ensuremath{K_{\mathrm{AUSF}}}}} مشتق می‌کند
\item
  \textbf{\lr{\mbox{SE AV}} به \lr{\mbox{AMF}}}: \lr{\mbox{AUSF}} مقدار \lr{\mbox{5G SE AV}} (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}}، \lr{\mbox{HXRES*}}) را به \lr{\mbox{AMF}} می‌فرستد
\item
  \textbf{چالش به \lr{\mbox{UE}}}: \lr{\mbox{AMF}} یک \lr{Authentication Request} (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}}، \lr{\mbox{ngKSI}}) به \lr{\mbox{UE}} می‌فرستد
\item
  \textbf{تأیید \lr{\mbox{UE}}}: \lr{\mbox{USIM}} مقدار \lr{\mbox{AUTN}} را تأیید می‌کند:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{AK}} را از \lr{\mbox{RAND}} با استفاده از \lr{\mbox{f5}} استخراج می‌کند، \lr{\mbox{SQN}} را بازیابی می‌کند
  \item
    \lr{\mbox{XMAC}} را محاسبه و تأیید می‌کند \lr{XMAC == MAC} (احراز هویت شبکه)
  \item
    تأیید می‌کند \lr{\mbox{SQN}} در محدوده قابل قبول است (محافظت از بازپخش)
  \end{itemize}
\item
  \textbf{مشتق‌سازی کلید در \lr{\mbox{UE}}}: \lr{\mbox{UE}} موارد زیر را محاسبه می‌کند:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{RES}}، \lr{\mbox{CK}}، \lr{\mbox{IK}} از \lr{\mbox{RAND}} با استفاده از \lr{\mbox{f2}}، \lr{\mbox{f3}}، \lr{\mbox{f4}}
  \item
    \lr{RES* = KDF(CK, IK, SN-name, RAND, RES)}
  \item
    \lr{\ensuremath{K_{\mathrm{AUSF}}} = KDF(CK, IK, SN-name, SQN\ensuremath{\oplus}AK)}
  \item
    \lr{\ensuremath{K_{\mathrm{SEAF}}} = KDF(\ensuremath{K_{\mathrm{AUSF}}}, SN-name)}
  \item
    \lr{\ensuremath{K_{\mathrm{AMF}}} = KDF(\ensuremath{K_{\mathrm{SEAF}}}, SUPI, ABBA)}
  \end{itemize}
\item
  \textbf{پاسخ}: \lr{\mbox{UE}} یک \lr{Authentication Response} با \lr{\mbox{RES*}} به \lr{\mbox{AMF}} می‌فرستد
\item
  \textbf{تأیید \lr{\mbox{AMF}}}: \lr{\mbox{AMF}} مقدار \lr{\mbox{HRES*}} را محاسبه و تأیید می‌کند \lr{HRES* == HXRES*}
\item
  \textbf{تأیید \lr{\mbox{AUSF}}}: \lr{\mbox{AMF}} مقدار \lr{\mbox{RES*}} را به \lr{\mbox{AUSF}} ارسال می‌کند؛ \lr{\mbox{AUSF}} تأیید می‌کند \lr{RES* == XRES*}
\item
  \textbf{موفقیت}: \lr{\mbox{AUSF}} موفقیت، \lr{\mbox{\ensuremath{K_{\mathrm{SEAF}}}}} و \lr{\mbox{SUPI}} تأییدشده را به \lr{\mbox{AMF}} بازمی‌گرداند
\item
  \textbf{مشتق‌سازی \lr{\mbox{\ensuremath{K_{\mathrm{AMF}}}}}}: \lr{\mbox{AMF}} مقدار \lr{\ensuremath{K_{\mathrm{AMF}}} = KDF(\ensuremath{K_{\mathrm{SEAF}}}, SUPI, ABBA)} را مشتق می‌کند
\end{enumerate}

  \item

\item

\end{enumerate}

\CNsubsection{}{جریان گام-به-گام}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{پیکربندی اندازه‌گیری}: \lr{\mbox{gNB}} مبدأ \lr{\mbox{UE}} را با اشیای اندازه‌گیری و معیارهای گزارش‌دهی پیکربندی می‌کند
\item
  \textbf{گزارش اندازه‌گیری}: \lr{\mbox{UE}} تشخیص می‌دهد سیگنال سلول مقصد به آستانه می‌رسد؛ یک \lr{MeasurementReport} به \lr{\mbox{gNB}} مبدأ می‌فرستد
\item
  \textbf{تصمیم تحویل}: \lr{\mbox{gNB}} مبدأ گزارش اندازه‌گیری را ارزیابی می‌کند، سلول مقصد را انتخاب می‌کند، تصمیم به تحویل می‌گیرد
\item
  \textbf{آماده‌سازی تحویل}:

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{gNB}} مبدأ یک \lr{Handover Request} به \lr{\mbox{gNB}} مقصد روی \lr{\mbox{Xn}} می‌فرستد:

    \begin{itemize}
    \tightlist
    \item
      زمینه \lr{\mbox{UE}} (امنیت، قابلیت‌ها، اطلاعات نشست \lr{\mbox{PDU}})
    \item
      اطلاعات تاریخچه \lr{\mbox{UE}}
    \item
      محفظه شفاف مبدأ-به-مقصد (پیکربندی \lr{\mbox{RRC}})
    \end{itemize}
  \end{itemize}
\item
  \textbf{کنترل پذیرش}: \lr{\mbox{gNB}} مقصد کنترل پذیرش را انجام می‌دهد، منابع را تخصیص می‌دهد (\lr{\mbox{DRB}}ها، منابع رادیویی)
\item
  \textbf{تأیید درخواست تحویل}: \lr{\mbox{gNB}} مقصد با موارد زیر پاسخ می‌دهد:

  \begin{itemize}
  \tightlist
  \item
    محفظه شفاف مقصد-به-مبدأ (فرمان تحویل \lr{\mbox{RRC}})
  \item
    تونل‌های ارسال \lr{\mbox{DL}} تخصیص‌یافته (برای ارسال داده در حین تحویل)
  \end{itemize}
\item
  \textbf{فرمان تحویل}: \lr{\mbox{gNB}} مبدأ یک \lr{RRCReconfiguration (mobilityControlInfo)} به \lr{\mbox{UE}} می‌فرستد شامل:

  \begin{itemize}
  \tightlist
  \item
    شناسه سلول مقصد، \lr{\mbox{C-RNTI}} جدید
  \item
    پیکربندی الگوریتم امنیتی \lr{\mbox{gNB}} مقصد
  \item
    پیکربندی منابع رادیویی برای سلول مقصد
  \end{itemize}
\item
  \textbf{ارسال داده}: \lr{\mbox{gNB}} مبدأ شروع به ارسال داده \lr{\mbox{DL}} بافر‌شده/\allowbreak{}ورودی به \lr{\mbox{gNB}} مقصد می‌کند (از طریق تونل ارسال \lr{\mbox{Xn}})
\item
  \textbf{انتقال وضعیت \lr{\mbox{SN}}}: \lr{\mbox{gNB}} مبدأ یک \lr{PDCP SN Status Transfer} به \lr{\mbox{gNB}} مقصد می‌فرستد (برای تحویل بدون‌اتلاف روی \lr{\mbox{DRB}}های \lr{\mbox{AM}})
\item
  \textbf{دسترسی تصادفی}: \lr{\mbox{UE}} به سلول مقصد همگام می‌شود (رویه \lr{\mbox{RACH}})
\item
  \textbf{تکمیل تحویل}: \lr{\mbox{UE}} یک \lr{RRCReconfigurationComplete} به \lr{\mbox{gNB}} مقصد می‌فرستد
\item
  \textbf{تغییر مسیر}: \lr{\mbox{gNB}} مقصد یک \lr{NGAP Path Switch Request} به \lr{\mbox{AMF}} می‌فرستد:

  \begin{itemize}
  \tightlist
  \item
    اطلاعات زمینه امنیتی جدید (\lr{NH/\allowbreak{}NCC} در صورت نیاز)
  \item
    اطلاعات تونل \lr{\mbox{N3}} جدید برای هر نشست \lr{\mbox{PDU}} (\lr{\mbox{F-TEID}} مربوط به \lr{\mbox{gNB}} مقصد)
  \end{itemize}
\item
  \textbf{به‌روزرسانی مسیر \lr{\mbox{UP}}}: \lr{\mbox{AMF}} یک \lr{Nsmf\_\allowbreak{}PDUSession\_\allowbreak{}UpdateSMContext} با اطلاعات تونل \lr{\mbox{AN}} جدید به \lr{\mbox{SMF}} می‌فرستد
\item
  \textbf{به‌روزرسانی \lr{\mbox{UPF}}}: \lr{\mbox{SMF}} یک \lr{PFCP Session Modification} به \lr{\mbox{UPF}} می‌فرستد \ensuremath{\leftarrow} \lr{\mbox{FAR}} مربوط به \lr{\mbox{DL}} را به‌سمت \lr{\mbox{gNB}} مقصد به‌روزرسانی می‌کند
\item
  \textbf{نشانگر پایان}: \lr{\mbox{UPF}} یک بسته نشانگر پایان روی تونل \lr{\mbox{N3}} قدیمی (به \lr{\mbox{gNB}} مبدأ) می‌فرستد تا نشان دهد مسیر \lr{\mbox{DL}} سوییچ شده است
\item
  \textbf{تأیید تغییر مسیر}: \lr{\mbox{AMF}} یک \lr{NGAP Path Switch Request Acknowledge} به \lr{\mbox{gNB}} مقصد می‌فرستد (\lr{\mbox{NH}}، \lr{\mbox{NCC}} جدید برای تحویل آینده)
\item
  \textbf{آزادسازی زمینه \lr{\mbox{UE}}}: \lr{\mbox{gNB}} مقصد یک \lr{UE Context Release} به \lr{\mbox{gNB}} مبدأ روی \lr{\mbox{Xn}} می‌فرستد
\item
  \textbf{پاک‌سازی مبدأ}: \lr{\mbox{gNB}} مبدأ زمینه و منابع \lr{\mbox{UE}} را آزاد می‌کند
\end{enumerate}

\CNsection{9}{تحویل \lr{\mbox{N2}} (بین-\lr{\mbox{AMF}})}

\CNchained\CNsubsection{}{هدف}

رویه تحویل \lr{\mbox{N2}} (بین-\lr{\mbox{AMF}}) تحرک را زمانی مدیریت می‌کند که \lr{\mbox{gNB}} مقصد توسط \lr{\mbox{AMF}} متفاوتی نسبت به مبدأ خدمات‌دهی می‌شود. این پیچیده‌تر از تحویل \lr{\mbox{Xn}} است زیرا نیازمند انتقال زمینه \lr{\mbox{UE}} بین \lr{\mbox{AMF}}ها، بازبرقراری انجمن‌های \lr{\mbox{NF}} (\lr{\mbox{SMF}}، \lr{\mbox{PCF}}) و به‌طور بالقوه احراز هویت مجدد است. زمانی استفاده می‌شود که هیچ واسط \lr{\mbox{Xn}} بین \lr{\mbox{gNB}}های مبدأ و مقصد وجود ندارد، یا زمانی که مقصد در ناحیه خدمات \lr{\mbox{AMF}} متفاوتی قرار می‌گیرد.

\CNsubsection{}{پیش‌شرط‌ها}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} در حالت \lr{RRC\_\allowbreak{}CONNECTED} با نشست‌های \lr{\mbox{PDU}} فعال است
\item
  \lr{\mbox{gNB}} مبدأ تعیین می‌کند تحویل لازم است (گزارش‌های اندازه‌گیری)
\item
  \lr{\mbox{gNB}} مقصد توسط همان \lr{\mbox{AMF}} خدمات‌دهی \textbf{نمی‌شود} (منطقه/\allowbreak{}مجموعه \lr{\mbox{AMF}} متفاوت)
\item
  \textbf{یا} هیچ واسط \lr{\mbox{Xn}} بین \lr{\mbox{gNB}}های مبدأ و مقصد وجود ندارد
\item
  واسط \lr{\mbox{N2}} بین \lr{\mbox{gNB}} مبدأ و \lr{\mbox{AMF}} مبدأ در دسترس است
\end{itemize}

  \item

  \end{itemize}

  \item

  \end{itemize}

\item

\item

\item
