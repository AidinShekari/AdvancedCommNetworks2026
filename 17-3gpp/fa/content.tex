\CNsection{1}{\lr{\mbox{3GPP}} چیست؟}

\textbf{پروژه مشارکت نسل سوم \lr{\mbox{(3GPP)}}} یک همکاری جهانی از سازمان‌های استانداردسازی مخابرات است که مشخصات و گزارش‌های فنی برای سیستم‌های ارتباطی موبایل تدوین می‌کند. \lr{\mbox{3GPP}} که در دسامبر 1998 تأسیس شد، هفت شریک سازمانی \lr{\mbox{(OP)}} منطقه‌ای را برای تولید استانداردهای کاربردپذیر جهانی شامل دسترسی رادیویی، معماری شبکه هسته و قابلیت‌های خدمات متحد می‌کند.

\CNsubsection{}{ویژگی‌های کلیدی}

\begin{itemize}
\tightlist
\item
  \textbf{شخصیت حقوقی نیست} \lr{—} \lr{\mbox{3GPP}} یک پروژه مشارکتی است که توسط \lr{\mbox{OP}}های خود اداره می‌شود
\item
  \textbf{شرکای نمایندگی بازار \lr{\mbox{(MRP)}}} ورودی صنعت را فراهم می‌کنند (مثلاً \lr{\mbox{GSMA}}، \lr{\mbox{5GAA}}، \lr{\mbox{NGMN}})
\item
  \textbf{اعضای فردی} شرکت‌هایی هستند که از طریق \lr{\mbox{OP}} منطقه‌ای خود مشارکت می‌کنند
\item
  مشخصات توسط هر \lr{\mbox{OP}} به استانداردهای منطقه‌ای \textbf{تبدیل} می‌شوند (مثلاً \lr{\mbox{ETSI}} آن‌ها را به‌عنوان \lr{EN/\allowbreak{}ES/\allowbreak{}TS} منتشر می‌کند)
\item
  سیستم‌های \textbf{\lr{2G (GSM/\allowbreak{}GPRS/\allowbreak{}EDGE)}، \lr{3G (UMTS)}، \lr{4G (LTE/\allowbreak{}LTE-A)} و \lr{\mbox{5G (NR)}}} را پوشش می‌دهد
\end{itemize}

\CNsubsection{6.2}{مورد مطالعه \lr{\mbox{(SI)}}}

\begin{itemize}
\tightlist
\item
  \textbf{هدف:} بررسی امکان‌سنجی یک ویژگی یا فناوری جدید
\item
  \textbf{خروجی:} گزارش فنی \lr{\mbox{(TR)}} که یافته‌ها، گزینه‌ها، نتایج را مستند می‌کند
\item
  \textbf{مدت:} معمولاً \lr{\mbox{1–4}} جلسه \lr{\mbox{TSG}} (\lr{\mbox{9–18}} ماه)
\item
  \textbf{فرآیند:}

  \begin{enumerate}
  \def\labelenumi{\arabic{enumi}.}
  \tightlist
  \item
    یک شرکت توصیف \lr{\mbox{SI (SID)}} را به \lr{\mbox{TSG}} ارایه می‌دهد
  \item
    \lr{\mbox{TSG}} مطالعه را تصویب می‌کند
  \item
    گروه کاری مطالعه را انجام می‌دهد (مشارکت‌های شرکت‌های عضو)
  \item
    \lr{\mbox{TR}} با نتایج و توصیه‌ها تولید می‌شود
  \item
    در صورت امکان‌پذیری \ensuremath{\leftarrow} پیشنهاد \lr{\mbox{WI}} ارایه می‌شود
  \end{enumerate}
\end{itemize}

\CNsubsection{6.3}{مورد کار \lr{\mbox{(WI)}}}

\begin{itemize}
\tightlist
\item
  \textbf{هدف:} ایجاد یا اصلاح مشخصات هنجاری
\item
  \textbf{خروجی:} مشخصات فنی \lr{\mbox{(TS)}} جدید یا به‌روزشده
\item
  \textbf{مدت:} مقید به زمان‌بندی نسخه هدف
\item
  \textbf{فرآیند:}

  \begin{enumerate}
  \def\labelenumi{\arabic{enumi}.}
  \tightlist
  \item
    یک شرکت توصیف مورد کار \lr{\mbox{(WID)}} را بر اساس نتایج \lr{\mbox{SI}} ارایه می‌دهد
  \item
    \lr{\mbox{TSG WI}} را تصویب می‌کند (به گروه کاری اختصاص می‌دهد، نقاط عطف تعیین می‌کند)
  \item
    گروه کاری درخواست‌های تغییر \lr{\mbox{(CR)}} به مشخصات موجود یا مشخصات جدید پیش‌نویس می‌کند
  \item
    \lr{\mbox{CR}}ها در سطح \lr{WG/\allowbreak{}TSG} بازبینی، اصلاح و تصویب می‌شوند
  \item
    \lr{\mbox{CR}}های تصویب‌شده در نسخه مشخصات برای آن \lr{\mbox{Release}} جای می‌گیرند
  \end{enumerate}
\end{itemize}
