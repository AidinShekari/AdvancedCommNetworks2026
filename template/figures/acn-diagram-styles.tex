% protocol stack: \CNstack{prefix}{x}{y of top layer}{width}{Layer/style, Layer/style, …}
% defines nodes <prefix>-1 (top) … <prefix>-n; styles: CNcpL (signalling), CNupL (user data)
\tikzset{CNcpL/.style={fill=CNsig!9}, CNupL/.style={fill=CNcyan!14}, CNphyL/.style={fill=CNpaper}, CNsecL/.style={fill=CNamber!14}}
\newcommand{\CNstack}[5]{%
  \foreach \CNlab/\CNst [count=\CNi] in {#5}{%
    \node[CNlayer, \CNst, minimum width=#4, minimum height=0.66cm, inner ysep=1pt]
      (#1-\CNi) at (#2,{#3-(\CNi-1)*0.66}) {\CNlab};}}
% packet header bit layout (32 bits per row): \CNbitruler, then
% \CNfield{row}{first bit}{number of bits}{label}{fill}
\def\CNbu{0.46}
\newcommand{\CNbitruler}{%
  \foreach \CNb in {0,...,31}{\node[font=\sffamily\tiny, text=CNmuted, inner sep=1pt] at ({(\CNb+0.5)*\CNbu},0.22) {\CNb};}}
\newcommand{\CNfield}[5]{%
  \node[draw=CNinkL!75, fill=#5, line width=0.6pt, minimum height=0.62cm, minimum width={#3*\CNbu cm},
        anchor=north west, inner sep=1pt, font=\sffamily\scriptsize, align=center] at ({#2*\CNbu},{-#1*0.62}) {#4};}
% packet on the wire: \CNpacket{y}{label/width(cm)/fill, …} — a strip of header fields from x = 0
\newcommand{\CNpacket}[2]{%
  \def\CNpx{0}%
  \foreach \CNl/\CNw/\CNf in {#2}{%
    \node[draw=CNinkL!75, fill=\CNf, line width=0.6pt, minimum height=0.9cm, minimum width=\CNw cm,
          anchor=west, inner sep=1.5pt, font=\sffamily\scriptsize, align=center] at (\CNpx,#1) {\CNl};
    \pgfmathsetmacro{\CNpnext}{\CNpx+\CNw}\xdef\CNpx{\CNpnext}}}
