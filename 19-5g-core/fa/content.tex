\CNsection{1}{مرور هسته \lr{\mbox{5G}}}

شبکه هسته \lr{\mbox{5G (5GC)}} در \lr{\textbf{\lr{3GPP TS 23.501}}} تعریف شده است و یک بازطراحی بنیادی شبکه هسته موبایل را نمایندگی می‌کند. برخلاف \lr{\mbox{EPC}} در \lr{\mbox{4G}} که از واسط‌های نقطه-به-نقطه بین عناصر شبکه یکپارچه استفاده می‌کرد، \lr{\mbox{5GC}} یک \textbf{معماری خدمات‌محور \lr{\mbox{(SBA)}}} را در پیش می‌گیرد که در آن توابع شبکه \lr{\mbox{(NF)}} خدمات را از طریق \lr{\mbox{API}}های \lr{\mbox{RESTful}} روی \lr{HTTP/\allowbreak{}2} ارایه می‌دهند.

\CNsubsection{}{اصول طراحی کلیدی}

\begin{itemize}
\tightlist
\item
  \textbf{معماری خدمات‌محور \lr{\mbox{(SBA)}}:} \lr{\mbox{NF}}ها از طریق واسط‌های خدمات‌محور \lr{\mbox{(SBI)}} با استفاده از \lr{HTTP/\allowbreak{}2} و \lr{\mbox{JSON}} ارتباط برقرار می‌کنند
\item
  \textbf{جداسازی کنترل و صفحه کاربر \lr{\mbox{(CUPS)}}:} جداسازی کامل \lr{\mbox{CP}} و \lr{\mbox{UP}}
\item
  \textbf{طراحی \lr{\mbox{NF}} بدون حالت:} حالت به‌صورت خارجی ذخیره می‌شود \lr{\mbox{(UDSF)}} و مقیاس‌پذیری افقی را ممکن می‌سازد
\item
  \textbf{شبکه‌بندی بومی:} پشتیبانی درون‌ساخت از چندین شبکه منطقی روی زیرساخت مشترک
\item
  \textbf{بومی-ابر:} طراحی‌شده برای استقرار کانتینری و مبتنی بر میکروسرویس
\item
  \textbf{کشف \lr{\mbox{NF}} مبتنی بر ثبت‌نام:} \lr{\mbox{NF}}ها در \lr{\mbox{NRF}} ثبت‌نام می‌کنند و همتاها را به‌صورت پویا کشف می‌کنند
\end{itemize}

\CNchained\CNsubsection{3.1}{تابع مدیریت دسترسی و تحرک \lr{\mbox{(AMF)}}}

\lr{\mbox{AMF}} نقطه تماس واحد برای تمام سیگنالینگ مرتبط با \lr{\mbox{UE}} از \lr{\mbox{RAN}} است. واسط‌های \lr{\mbox{N1 (NAS)}} و \lr{N2 (NGAP)} را پایان می‌دهد.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{پایان‌دهی \lr{\mbox{NAS}}:} تمام پیام‌های \lr{\mbox{NAS}} (ثبت‌نام، درخواست خدمات، برقراری نشست \lr{\mbox{PDU}}) در \lr{\mbox{AMF}} پایان می‌یابند
\item
  \textbf{مدیریت ثبت‌نام:} ثبت‌نام، لغو ثبت‌نام و به‌روزرسانی‌های دوره‌ای ثبت‌نام \lr{\mbox{UE}} را مدیریت می‌کند
\item
  \textbf{مدیریت اتصال:} زمینه سیگنالینگ \lr{\mbox{RRC}} را از طریق \lr{\mbox{N2}} به‌سمت \lr{\mbox{gNB}} مدیریت می‌کند
\item
  \textbf{مدیریت تحرک:} سیگنالینگ تحویل، مدیریت ناحیه ردیابی، هماهنگی صفحه‌بندی
\item
  \textbf{امنیت:} احراز هویت را آغاز می‌کند (به \lr{\mbox{AUSF}} واگذار می‌کند)، زمینه امنیتی \lr{\mbox{NAS}} را مدیریت می‌کند (رمزگذاری + یکپارچگی)
\item
  \textbf{انتخاب برش:} با \lr{\mbox{NSSF}} برای تعیین برش شبکه مناسب تعامل دارد
\end{itemize}

\CNsubsection{3.2}{تابع مدیریت نشست \lr{\mbox{(SMF)}}}

\lr{\mbox{SMF}} نشست‌های \lr{\mbox{PDU}} را سرتاسری مدیریت می‌کند. \lr{\mbox{UPF(\rl{ها})}}یی را که مسیر صفحه کاربر را تشکیل می‌دهند انتخاب و کنترل می‌کند.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{مدیریت نشست \lr{\mbox{PDU}}:} برقراری، اصلاح و آزادسازی نشست‌های \lr{\mbox{PDU}}
\item
  \textbf{انتخاب و کنترل \lr{\mbox{UPF}}:} \lr{\mbox{UPF(\rl{های})}} مناسب را انتخاب می‌کند، قواعد ارسال را از طریق \lr{N4/\allowbreak{}PFCP} برنامه‌ریزی می‌کند
\item
  \textbf{تخصیص آدرس \lr{\mbox{IP}}:} آدرس‌های \lr{IPv4/\allowbreak{}IPv6} را به \lr{\mbox{UE}}ها تخصیص می‌دهد (یا به \lr{\mbox{DHCP}} خارجی واگذار می‌کند)
\item
  \textbf{اجرای \lr{\mbox{QoS}}:} قواعد \lr{\mbox{QoS}} را در \lr{\mbox{UPF}} بر اساس سیاست‌های \lr{\mbox{PCF}} نصب می‌کند
\item
  \textbf{رومینگ:} سناریوهای مسیریابی خانگی و شکست محلی را مدیریت می‌کند
\item
  \textbf{اعلان داده پیوند پایین‌سو:} هنگام رسیدن داده پیوند پایین‌سو برای \lr{\mbox{UE}} بیکار، صفحه‌بندی را از طریق \lr{\mbox{AMF}} فعال می‌کند
\end{itemize}

\CNsubsection{3.3}{تابع صفحه کاربر \lr{\mbox{(UPF)}}}

\lr{\mbox{UPF}} \textbf{تنها \lr{\mbox{NF}} صفحه کاربر} در هسته \lr{\mbox{5G}} است. تمام پردازش بسته، ارسال و اجرای سیاست برای داده کاربر را انجام می‌دهد.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{مسیریابی و ارسال بسته:} بسته‌ها را بین \lr{\mbox{RAN}} و شبکه داده مسیریابی می‌کند
\item
  \textbf{تشخیص و گزارش‌دهی ترافیک:} برای شناسایی کاربرد \lr{\mbox{DPI}} انجام می‌دهد
\item
  \textbf{مدیریت \lr{\mbox{QoS}}:} \lr{\mbox{QoS}} در هر جریان را اجرا می‌کند (علامت‌گذاری، پلیسینگ، شکل‌دهی)
\item
  \textbf{گزارش‌دهی مصرف:} حجم/\allowbreak{}مدت ترافیک را برای صورت‌حساب اندازه‌گیری می‌کند
\item
  \textbf{طبقه‌بند پیوند بالاسو \lr{\mbox{(UL-CL)}}:} ترافیک را بر اساس تطبیق ترافیک به شبکه‌های داده مختلف هدایت می‌کند
\item
  \textbf{نقطه انشعاب:} از نشست‌های \lr{\mbox{PDU}} چند-میزبانه پشتیبانی می‌کند
\item
  \textbf{بافرینگ:} بسته‌های پیوند پایین‌سو را برای \lr{\mbox{UE}}های بیکار بافر می‌کند، \lr{\mbox{DDN}} را به \lr{\mbox{SMF}} فعال می‌کند
\end{itemize}

\CNsubsection{3.4}{مدیریت داده یکپارچه \lr{\mbox{(UDM)}}}

\lr{\mbox{UDM}} مدیریت داده مشترک، مدیریت اعتبارنامه احراز هویت و مجوزدهی اشتراک را فراهم می‌کند. این معادل \lr{\mbox{HSS}} در \lr{\mbox{5G}} است.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{ذخیره \lr{\mbox{SUPI}}:} شناسه دایمی اشتراک (\lr{\mbox{SUPI}} = قالب \lr{\mbox{IMSI}} یا \lr{\mbox{NAI}}) را ذخیره و مدیریت می‌کند
\item
  \textbf{محاسبه اعتبارنامه احراز هویت:} بردارهای احراز هویت \lr{\mbox{(AV)}} را با استفاده از کلید بلندمدت مشترک \lr{\mbox{(K)}} و پارامترهای اپراتور \lr{(OP/\allowbreak{}OPc)} تولید می‌کند
\item
  \textbf{مدیریت پروفایل اشتراک:} خدمات اشتراک‌شده، مجوزهای \lr{\mbox{DNN}}، پروفایل‌های \lr{\mbox{QoS}} را نگه می‌دارد
\item
  \textbf{مدیریت ثبت‌نام:} \lr{AMF/\allowbreak{}SMF} خدمات‌دهنده را در هر مشترک ردیابی می‌کند (ثبت‌نام \lr{\mbox{NF}})
\item
  \textbf{مجوزدهی \lr{\mbox{DNN}}:} تأیید می‌کند که \lr{\mbox{UE}} مجاز به دسترسی به ترکیب \lr{\mbox{DNN}}/\allowbreak{}برش مشخص است
\item
  \textbf{رمزگشایی \lr{\mbox{SUCI}}:} \lr{\mbox{SUCI}} را به \lr{\mbox{SUPI}} بازمی‌گرداند (با استفاده از \lr{\mbox{SIDF}} \lr{—} تابع رمزگشایی شناسه اشتراک)
\end{itemize}

\textbf{داده‌های کلیدی ذخیره‌شده:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{SUPI}}، کلیدهای رمزگشایی \lr{\mbox{SUCI}}
\item
  اعتبارنامه‌های احراز هویت مشترک (\lr{\mbox{K}}، \lr{OP/\allowbreak{}OPc}، \lr{\mbox{SQN}})
\item
  \lr{\mbox{S-NSSAI}} و فهرست \lr{\mbox{DNN}} اشتراک‌شده
\item
  پارامترهای \lr{\mbox{QoS}} پیش‌فرض نشست
\item
  ثبت‌نام‌های \lr{\mbox{NF}} خدمات‌دهنده (\lr{\mbox{AMF}}، \lr{\mbox{SMF}})
\item
  داده اشتراک دسترسی و تحرک
\end{itemize}

\textbf{واسط‌ها:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{Nudm:}}}}} واسط خدمات‌محور ارایه‌شده توسط \lr{\mbox{UDM}}
\item
  \textbf{\lr{\mbox{N8}} (از طریق \lr{\mbox{Nudm}}):} \lr{\mbox{AMF}} داده اشتراک/\allowbreak{}تحرک را بازیابی می‌کند
\item
  \textbf{\lr{\mbox{N10}} (از طریق \lr{\mbox{Nudm}}):} \lr{\mbox{SMF}} داده اشتراک مدیریت نشست را بازیابی می‌کند
\item
  \textbf{\lr{\mbox{N13}} (از طریق \lr{\mbox{Nudm}}):} \lr{\mbox{AUSF}} اعتبارنامه‌های احراز هویت را بازیابی می‌کند

\item

\end{itemize}

\CNsubsection{3.8}{تابع مخزن شبکه \lr{\mbox{(NRF)}}}

\lr{\mbox{NRF}} ثبت مرکزی برای تمام نمونه‌های \lr{\mbox{NF}} در \lr{\mbox{5GC}} است. کشف و مجوزدهی خدمات را ممکن می‌سازد.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{ثبت‌نام \lr{\mbox{NF}}:} \lr{\mbox{NF}}ها پروفایل خود را ثبت می‌کنند (نوع، ظرفیت، خدمات پشتیبانی‌شده، \lr{\mbox{PLMN}}، برش، مکان)
\item
  \textbf{کشف \lr{\mbox{NF}}:} \lr{\mbox{NF}}های مصرف‌کننده از \lr{\mbox{NRF}} پرس‌وجو می‌کنند تا \lr{\mbox{NF}}های تولیدکننده منطبق با معیارها را بیابند
\item
  \textbf{اعلان وضعیت \lr{\mbox{NF}}:} \lr{\mbox{NF}}های مشترک‌شده را درباره تغییرات وضعیت \lr{\mbox{NF}} مطلع می‌کند (ثبت‌نام‌شده، لغو ثبت‌نام‌شده، معلق)
\item
  \textbf{صدور نشانه \lr{\mbox{OAuth2}}:} نشانه‌های دسترسی برای مجوزدهی واسط خدمات‌محور صادر می‌کند
\end{itemize}

\textbf{محتوای پروفایل \lr{\mbox{NF}}:}

\begin{itemize}
\tightlist
\item
  شناسه نمونه \lr{NF (UUID)}
\item

\item
  وضعیت \lr{\mbox{NF}} (\lr{\mbox{REGISTERED}}، \lr{\mbox{SUSPENDED}}، \lr{\mbox{UNDISCOVERABLE}})
\item
  شناسه(های) \lr{\mbox{PLMN}} خدمات‌داده‌شده
\item
  \lr{S-NSSAI(\rl{های})} پشتیبانی‌شده
\item
  آدرس‌های \lr{\mbox{FQDN}} یا \lr{\mbox{IP}}
\item
  خدمات و نسخه‌های پشتیبانی‌شده
\item
  اطلاعات ظرفیت، اولویت، بار
\item
  محلیت جغرافیایی
\end{itemize}

\textbf{پارامترهای پرس‌وجوی کشف:}

\item

\item

\CNsubsection{3.9}{تابع نمایش شبکه \lr{\mbox{(NEF)}}}

\lr{\mbox{NEF}} یک چارچوب ایمن برای نمایش قابلیت‌ها و رویدادهای \lr{\mbox{5GC}} به توابع کاربرد خارجی \lr{\mbox{(AF)}} فراهم می‌کند.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{چارچوب نمایش:} قابلیت‌های شبکه را از طریق \lr{\mbox{API}}های استاندارد به \lr{\mbox{AF}}های شخص سوم نمایش می‌دهد
\item
  \textbf{تأثیر ترافیک:} \lr{\mbox{AF}} می‌تواند مسیریابی ترافیک مشخص را درخواست کند (مثلاً مسیریابی ترافیک به کارساز لبه محلی)
\item
  \textbf{رویدادهای پایش:} \lr{\mbox{AF}} در رویدادهای \lr{\mbox{UE}} اشتراک می‌کند (دسترسی‌پذیری، مکان، وضعیت اتصال)

  \textbf{تأمین پارامتر:} \lr{\mbox{AF}} رفتار مورد انتظار \lr{\mbox{UE}}، سیاست‌های انتقال داده پس‌زمینه را تأمین می‌کند
\item
  \textbf{ترجمه:} بین شناسه‌های خارجی (\lr{\mbox{GPSI}}، شناسه گروه خارجی) و شناسه‌های داخلی \lr{\mbox{(SUPI)}} نگاشت می‌کند
\item
  \textbf{دروازه امنیتی:} از دسترسی مستقیم \lr{\mbox{AF}} به \lr{\mbox{NF}}های داخلی جلوگیری می‌کند

\lr{\mbox{UDR}} مخزن داده پشتیبان برای داده مشترک، سیاست و نمایش است. توسط \lr{\mbox{UDM}}، \lr{\mbox{PCF}} و \lr{\mbox{NEF}} دسترسی می‌شود.

\begin{itemize}
\tightlist
\item

\CNsubsection{3.11}{تابع ذخیره‌سازی داده غیرساخت‌یافته \lr{\mbox{(UDSF)}}}

\lr{\mbox{UDSF}} با فراهم کردن ذخیره‌سازی خارجی برای داده زمینه \lr{\mbox{NF}}، طراحی بدون حالت \lr{\mbox{NF}}ها را ممکن می‌سازد.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{ذخیره‌سازی داده غیرساخت‌یافته:} زمینه/\allowbreak{}حالت دلخواه \lr{\mbox{NF}} را به‌عنوان بلوک‌های داده مبهم ذخیره می‌کند
\item
  \textbf{پشتیبانی \lr{\mbox{NF}} بدون حالت:} \lr{\mbox{NF}}ها می‌توانند پس از راه‌اندازی مجدد حالت را بازیابی کنند یا بار را در بین نمونه‌ها متعادل کنند
\item
  \textbf{عملیات \lr{\mbox{CRUD}}:} عملیات ایجاد، خواندن، به‌روزرسانی، حذف روی رکوردهای ذخیره‌شده
\end{itemize}

\textbf{چرا \lr{\mbox{UDSF}} مهم است:}

\begin{itemize}
\tightlist
\item
  بدون \lr{\mbox{UDSF}}، هر نمونه \lr{\mbox{NF}} وضعیت نشست را در حافظه نگه می‌دارد
\item
  با \lr{\mbox{UDSF}}، هر نمونه \lr{\mbox{NF}} می‌تواند با بازیابی حالت از \lr{\mbox{UDSF}} به هر درخواستی خدمات دهد
\item
  مقیاس‌پذیری افقی واقعی، تغییر مسیر و استقرار بومی-ابر را ممکن می‌سازد
\item
  هر نوع \lr{\mbox{NF}} ممکن است نمونه \lr{\mbox{UDSF}} خود را داشته باشد
\end{itemize}

\textbf{واسط:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{Nudsf:}}}}} واسط خدمات‌محور برای ذخیره/\allowbreak{}بازیابی داده غیرساخت‌یافته
\end{itemize}

\CNsubsection{3.12}{عامل ارتباط خدمات \lr{\mbox{(SCP)}}}

\lr{\mbox{SCP}} یک عنصر مسیریابی و توازن بار است که در مسیر ارتباط \lr{\mbox{SBI}} بین \lr{\mbox{NF}}ها قرار می‌گیرد. در \lr{TS 23.501} نسخه \lr{\mbox{16}}+ تعریف شده است.

\textbf{مسیولیت‌های کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{ارتباط غیرمستقیم:} \lr{\mbox{NF}}ها درخواست‌ها را به \lr{\mbox{SCP}} می‌فرستند به‌جای مستقیماً به \lr{\mbox{NF}} هدف
\item
  \textbf{توازن بار:} درخواست‌ها را در بین چندین نمونه \lr{\mbox{NF}} توزیع می‌کند
\item
  \textbf{مسیریابی پیام:} پیام‌ها را بر اساس پارامترهای کشف مسیریابی می‌کند بدون نیاز \lr{\mbox{NF}} مصرف‌کننده به کشف کامل
\item
  \textbf{پنهان‌سازی توپولوژی:} توپولوژی \lr{\mbox{NF}} داخلی را از شبکه‌های خارجی پنهان می‌کند (رومینگ)
\item
  \textbf{کنترل اضافه‌بار:} مدیریت ازدحام و محدودسازی درخواست را پیاده‌سازی می‌کند
\end{itemize}

\CNsection{4}{صفحه کنترل در برابر صفحه کاربر در هسته \lr{\mbox{5G}}}

\lr{\mbox{5GC}} جداسازی سختگیرانه‌ای بین صفحه کنترل \lr{\mbox{(CP)}} و صفحه کاربر \lr{\mbox{(UP)}} اعمال می‌کند که به‌عنوان \lr{\mbox{CUPS}} (جداسازی کنترل و صفحه کاربر) شناخته می‌شود.

\CNsubsection{}{صفحه کنترل \lr{\mbox{(CP)}}}

\begin{itemize}
\tightlist
\item
  \textbf{تمام \lr{\mbox{NF}}ها به‌جز \lr{\mbox{UPF}}} در صفحه کنترل عمل می‌کنند
\item
  ارتباط از طریق \textbf{واسط‌های خدمات‌محور \lr{\mbox{(SBI)}}} با استفاده از \lr{HTTP/\allowbreak{}2 + JSON}
\item
  پشته پروتکل: \lr{HTTP/\allowbreak{}2 \ensuremath{\to} TCP \ensuremath{\to} IP} (یا از طریق \lr{\mbox{SCP}})
\item
  توابع: سیگنالینگ، سیاست، احراز هویت، ثبت‌نام، مدیریت نشست، کشف
\item
  بسته به نیازمندی‌های تأخیر، به‌صورت متمرکز یا منطقه‌ای مستقر می‌شود
\item
  \lr{\mbox{AMF}} و \lr{\mbox{SMF}} لنگرهای اصلی \lr{\mbox{CP}} برای سیگنالینگ \lr{\mbox{UE}} و مدیریت نشست هستند
\end{itemize}

\CNsubsection{}{صفحه کاربر \lr{\mbox{(UP)}}}

\begin{itemize}
\tightlist
\item
  \textbf{فقط \lr{\mbox{UPF}}} در صفحه کاربر عمل می‌کند
\item
  توسط \lr{\mbox{SMF}} از طریق \lr{\textbf{\lr{N4 /\allowbreak{} PFCP}}} (واسط کنترل) برنامه‌ریزی می‌شود
\item
  پشته پروتکل داده کاربر: داده کاربرد \ensuremath{\leftarrow} لایه \lr{PDU \ensuremath{\to} GTP-U \ensuremath{\to} UDP \ensuremath{\to} IP}
\item
  توابع: ارسال بسته، اجرای \lr{\mbox{QoS}}، تشخیص ترافیک، اندازه‌گیری مصرف، بافرینگ
\item
  بسته به نیازهای تأخیر و محلیت داده، می‌تواند در لبه، منطقه‌ای یا مکان‌های مرکزی مستقر شود
\item
  چندین \lr{\mbox{UPF}} می‌توانند زنجیره شوند (از طریق \lr{\mbox{N9}}) برای زنجیره‌سازی تابع سرویس یا شکست محلی
\end{itemize}

\tightlist
\item
  \lr{\mbox{PDU}}های \lr{\mbox{NAS}} را حمل می‌کند (\lr{Registration Request/\allowbreak{}Accept}، احراز هویت، حالت امنیتی، برقراری نشست \lr{\mbox{PDU}})
\item
  پیام‌های \lr{\mbox{NAS}} به‌صورت شفاف توسط \lr{\mbox{gNB}} رله می‌شوند (کپسوله‌شده در \lr{\mbox{NGAP}} روی \lr{\mbox{N2}})
\item
  پس از فعال‌سازی امنیت، از امنیت \lr{\mbox{NAS}} (یکپارچگی + رمزگذاری) پشتیبانی می‌کند
\end{itemize}

\lr{\textbf{\lr{N2 (RAN \ensuremath{\leftrightarrow} AMF):}}}

\begin{itemize}
\tightlist
\item
  پروتکل \lr{\mbox{NGAP}} روی \lr{\mbox{SCTP}}
\item
  رویه‌ها: \lr{Initial UE Message}، \lr{PDU Session Resource Setup}، \lr{Handover Required/\allowbreak{}Request}، \lr{\mbox{Paging}}
\item
  سیگنالینگ مرتبط با \lr{\mbox{UE}} و غیرمرتبط با \lr{\mbox{UE}} را حمل می‌کند
\end{itemize}

\lr{\textbf{\lr{N3 (RAN \ensuremath{\leftrightarrow} UPF):}}}

\begin{itemize}
\tightlist
\item
  تونل \lr{\mbox{GTP-U}} شناسایی‌شده با \lr{\mbox{TEID}} (شناسه نقطه انتهایی تونل)
\item
  یک تونل \lr{\mbox{GTP-U}} در هر نشست \lr{\mbox{PDU}} در هر جفت \lr{\mbox{gNB-UPF}}
\item
  شناسه جریان \lr{QoS (QFI)} در سرآیند افزونه \lr{\mbox{GTP-U}} برای نگاشت جریان‌های \lr{\mbox{QoS}} حمل می‌شود
\end{itemize}

\lr{\textbf{\lr{N4 (SMF \ensuremath{\leftrightarrow} UPF):}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{PFCP}} (پروتکل کنترل ارسال بسته) \lr{—} \lr{3GPP TS 29.244}
\item
  مدیریت سطح نشست (برقراری، اصلاح، حذف)
\item
  مدیریت سطح گره (برقراری ارتباط، ضربان)
\item
  گزارش‌ها: گزارش‌های مصرف، اعلان داده پیوند پایین‌سو، نشانگر پایان
\end{itemize}

\lr{\textbf{\lr{N9 (UPF \ensuremath{\leftrightarrow} UPF):}}}

\begin{itemize}
\tightlist
\item
  تونل \lr{\mbox{GTP-U}} برای مسیرهای صفحه کاربر چند-گامی
\item
  در سناریوهای \lr{\mbox{UL-CL}} (طبقه‌بند پیوند بالاسو) و نقطه انشعاب استفاده می‌شود
\item
  زنجیره‌سازی تابع سرویس و استقرار \lr{\mbox{UPF}} توزیع‌شده را ممکن می‌سازد
\end{itemize}

\CNsection{7}{گزینه‌های استقرار}

معماری پیمانه‌ای \lr{\mbox{5GC}} توپولوژی‌های استقرار انعطاف‌پذیر بر اساس نیازمندی‌های سرویس را ممکن می‌سازد.

\CNsubsection{7.1}{استقرار متمرکز}

تمام \lr{\mbox{NF}}ها (شامل \lr{\mbox{UPF}}) در یک مرکز داده مرکزی مستقر می‌شوند.

\textbf{ویژگی‌ها:}

\begin{itemize}
\tightlist
\item
  عملیات ساده، سایت واحد
\item
  تأخیر بالاتر برای صفحه کاربر (تمام ترافیک از سایت مرکزی عبور می‌کند)
\item
  مناسب برای پوشش گسترده با نیازمندی‌های تأخیر آسان‌گیر
\item
  هزینه زیرساخت کمتر
\end{itemize}

\CNsubsection{7.2}{استقرار توزیع‌شده}

صفحه کنترل متمرکز، صفحه کاربر توزیع‌شده نزدیک‌تر به کاربران.

\textbf{ویژگی‌ها:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{NF}}های \lr{\mbox{CP}} (\lr{\mbox{AMF}}، \lr{\mbox{SMF}}، \lr{\mbox{PCF}} و غیره) در \lr{\mbox{DC}} منطقه‌ای/\allowbreak{}مرکزی
\item
  \lr{\mbox{UPF}} در نقاط تجمیع منطقه‌ای مستقر می‌شود
\item
  تأخیر صفحه کاربر کاهش‌یافته
\item
  \lr{\mbox{SMF}} چندین نمونه \lr{\mbox{UPF}} را در مکان‌های مختلف مدیریت می‌کند
\end{itemize}

\CNsubsection{7.3}{استقرار \lr{\mbox{UPF}} در لبه (یکپارچگی \lr{\mbox{MEC}})}

نمونه‌های \lr{\mbox{UPF}} در لبه شبکه (سایت سلول یا نقطه تجمیع محلی) برای تأخیر فوق‌العاده کم مستقر می‌شوند.

\textbf{ویژگی‌ها:}

\begin{itemize}
\tightlist
\item
  شکست محلی ترافیک در لبه
\item
  رایانش لبه چند-دسترسی \lr{\mbox{(MEC)}} را ممکن می‌سازد \lr{—} چارچوب \lr{\mbox{MEC}} سازمان \lr{\mbox{ETSI}}
\item
  \lr{\mbox{SMF}} می‌تواند \lr{\mbox{UPF}}های لبه را بر اساس مکان \lr{\mbox{UE}} به‌صورت پویا وارد/\allowbreak{}حذف کند
\item
  \textbf{\lr{\mbox{UL-CL}} (طبقه‌بند پیوند بالاسو):} جریان‌های ترافیک مشخص را به \lr{\mbox{UPF}} لبه هدایت می‌کند در حالی که بقیه به \lr{\mbox{UPF}} مرکزی می‌رود
\item
  \textbf{شبکه داده ناحیه محلی \lr{\mbox{(LADN)}}:} شبکه داده فقط در ناحیه جغرافیایی مشخص در دسترس است
\item
  موارد استفاده معمول: \lr{AR/\allowbreak{}VR}، رانندگی خودران، خودکارسازی صنعتی، بازی ابری
\end{itemize}

\CNsubsection{}{مسیر صفحه کاربر نشست \lr{\mbox{PDU}}}

\CNfigure{m19-pdu-path}{مسیر صفحه کاربر یک نشست \lr{\mbox{PDU}} با \lr{\mbox{UPF}} میانی}

\textbf{جریان برقراری نشست \lr{\mbox{PDU}} (ساده‌شده):}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{UE \ensuremath{\to} AMF}}: \lr{NAS PDU Session Establishment Request} (از طریق \lr{\mbox{N1}})
\item
  \lr{AMF \ensuremath{\to} SMF}: \lr{Nsmf\_\allowbreak{}PDUSession\_\allowbreak{}CreateSMContext} (از طریق \lr{\mbox{N11}})
\item
  \lr{\mbox{SMF UPF}} را انتخاب می‌کند، آدرس \lr{\mbox{IP}} تخصیص می‌دهد
\item
  \lr{SMF \ensuremath{\to} UPF}: \lr{PFCP Session Establishment} (از طریق \lr{\mbox{N4}}) \lr{—} \lr{PDR/\allowbreak{}FAR/\allowbreak{}QER} را نصب می‌کند
\item
  \lr{SMF \ensuremath{\to} AMF}: \lr{N1N2MessageTransfer} (شامل \lr{NAS accept} + اطلاعات \lr{\mbox{N2}})
\item
  \lr{AMF \ensuremath{\to} gNB}: \lr{PDU Session Resource Setup Request} (از طریق \lr{\mbox{N2}})
\item
  \lr{gNB \ensuremath{\to} UPF}: تونل \lr{\mbox{GTP-U}} برقرار می‌شود \lr{\mbox{(N3)}}
\item
  \lr{\mbox{UE}} دریافت \lr{PDU Session Establishment Accept} با قواعد \lr{\mbox{QoS}}
\end{enumerate}

\CNsection{}{خلاصه}

هسته \lr{\mbox{5G}} یک تغییر پارادایم از \lr{\mbox{EPC}} در \lr{\mbox{4G}} را نمایندگی می‌کند:

\begin{itemize}
\tightlist
\item
  \textbf{\lr{\mbox{SBA}} جایگزین واسط‌های نقطه-به-نقطه می‌شود} \lr{—} انعطاف‌پذیری، قابلیت استفاده مجدد و تحول مستقل \lr{\mbox{NF}} را ممکن می‌سازد
\item
  \textbf{\lr{\mbox{CUPS}} کامل} \lr{—} \lr{\mbox{UPF}} تنها موجودیت صفحه کاربر است، قابل برنامه‌ریزی از طریق \lr{\mbox{PFCP}}
\item
  \textbf{بومی-ابر از طراحی} \lr{—} \lr{\mbox{NF}}های بدون حالت، \lr{\mbox{UDSF}}، کشف مبتنی بر \lr{\mbox{NRF}}، کانتینری‌سازی
\item
  \textbf{شبکه‌بندی بومی} \lr{—} \lr{\mbox{S-NSSAI}}، \lr{\mbox{NSSF}}، انتخاب \lr{\mbox{NF}} در هر برش از پایه
\item
  \textbf{امنیت تقویت‌شده} \lr{—} پنهان‌سازی \lr{\mbox{SUCI}}، احراز هویت کنترل‌شده توسط شبکه خانگی، \lr{\mbox{OAuth2}} در \lr{\mbox{SBI}}
\item
  \textbf{آماده-لبه} \lr{—} \lr{\mbox{UPF}} توزیع‌شده، \lr{\mbox{UL-CL}}، \lr{\mbox{LADN}} برای خدمات فوق‌کم-تأخیر
\end{itemize}

\emph{مراجع: \lr{3GPP TS 23.501} (معماری سیستم)، \lr{TS 23.502} (رویه‌ها)، \lr{TS 29.244 (PFCP)}، \lr{TS 33.501} (معماری امنیت)}
