\CNsubsection{}{پاسخ \lr{\mbox{Diameter}}}

\lr{\mbox{Diameter}} (نام آن به معنای «دو برابر \lr{\mbox{RADIUS}}») در \lr{\mbox{\textbf{\lr{\mbox{RFC 6733}}}}} برای رفع تمام این محدودیت‌ها تعریف شد:

\begin{itemize}
\tightlist
\item
  انتقال قابل اعتماد \lr{(TCP/\allowbreak{}SCTP)}
\item
  پیام‌های با شروع سرور (مدل ارسال)
\item
  فضای کد \lr{\mbox{AVP 32}} بیتی (بیش از 4 میلیارد نوع صفت)
\item
  تبادل قابلیت و مدیریت خطای داخلی
\item
  معماری عامل برای مسیریابی مقیاس‌پذیر
\item
  پشتیبانی بومی \lr{TLS/\allowbreak{}DTLS}
\end{itemize}

\begin{CNquote}

\lr{\mbox{\CNwarn{}}} \textbf{مفهوم کلیدی}: \lr{\mbox{Diameter}} فقط یک پروتکل \textbf{صفحه کنترل} است. سیگنالینگ را حمل می‌کند (احراز هویت، مجوزدهی، حسابداری، سیاست) \lr{—} هرگز ترافیک داده کاربر.

\end{CNquote}

\CNsubsection{3.2}{اتصالات همتا}

\begin{itemize}
\tightlist
\item
  \textbf{انتقال}: \lr{\mbox{TCP}} (پیش‌فرض) یا \lr{\mbox{SCTP}} (ترجیحی برای چند-میزبانی)
\item
  \textbf{پورت}: 3868 (متن ساده) یا \lr{5868 (TLS)}
\item
  \textbf{پایدار}: اتصال برقرار می‌ماند؛ با نگهبان \lr{(DWR/\allowbreak{}DWA)} پایش می‌شود
\item
  \textbf{همتا-به-همتا}: هر دو نقطه انتهایی می‌توانند پیام آغاز کنند
\item
  \textbf{تبادل قابلیت}: \lr{CER/\allowbreak{}CEA} باید پیش از پیام‌های کاربرد موفق شود
\end{itemize}

\CNsubsection{3.3}{مسیریابی مبتنی بر \lr{\mbox{Realm}}}

\lr{\mbox{Diameter}} پیام‌ها را بر اساس \textbf{\lr{\mbox{realm}}ها} (شناسه‌های شبه-دامنه) مسیریابی می‌کند:

\begin{itemize}
\tightlist
\item
  هر گره با یک \lr{\mbox{realm}} پیکربندی می‌شود (مثلاً \lr{\CNcode{epc.}\allowbreak{}\CNcode{mnc001.}\allowbreak{}\CNcode{mcc208.}\allowbreak{}\CNcode{3gppnetwork.}\allowbreak{}\CNcode{org}})
\item
  جدول مسیریابی \lr{\mbox{realm}}ها را به اتصالات همتا نگاشت می‌کند
\item
  عامل‌ها از \lr{AVP \CNcode{Destination-Realm}} برای تعیین گام بعدی استفاده می‌کنند
\item
  سناریوهای چند-اپراتوری و رومینگ را ممکن می‌سازد
\end{itemize}

\CNsubsection{8.2}{چرا \lr{\mbox{5G}} از \lr{\mbox{Diameter}} فاصله گرفت}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{ابزار ساده‌تر}: \lr{HTTP/\allowbreak{}2 + JSON} از زیرساخت وب موجود استفاده می‌کند (پروکسی‌ها، متعادل‌کننده‌های بار، دروازه‌های \lr{\mbox{API}})
\item
  \textbf{بومی-وب}: \lr{\mbox{API}}های \lr{\mbox{REST}} استاندارد \lr{—} آسان‌تر برای توسعه‌دهندگان، ابزارهای اشکال‌زدایی استاندارد
\item
  \textbf{طراحی بدون حالت}: میکروسرویس‌های بومی-ابر، مقیاس‌پذیری افقی، هماهنگ‌سازی \lr{\mbox{Kubernetes}} را ممکن می‌سازد
\item
  \textbf{سازگار با ابر}: بدون اتصالات همتای پایدار؛ توازن بار استاندارد کار می‌کند
\item
  \textbf{انعطاف‌پذیری}: تکامل طرح \lr{\mbox{JSON}} آسان‌تر از تعاریف \lr{\mbox{AVP}} دودویی است
\item
  \textbf{اکوسیستم}: اکوسیستم عظیم \lr{HTTP/\allowbreak{}2} (کتابخانه‌ها، ابزارها، پایش)
\end{enumerate}

\CNsubsection{8.4}{مهاجرت: هم‌کنش‌گری \lr{4G/\allowbreak{}5G}}

\lr{\mbox{Diameter}} از بین نرفته است \lr{—} در مرزهای هم‌کنش‌گری باقی می‌ماند:

\begin{itemize}
\tightlist
\item
  \textbf{عامل مسیریابی \lr{Diameter (DRA)}} همچنان در \lr{\mbox{4G}} استفاده می‌شود
\item
  \textbf{تابع هم‌کنش‌گری \lr{\mbox{(IWF)}}} در مرز \lr{4G/\allowbreak{}5G} بین \lr{Diameter \ensuremath{\leftrightarrow} HTTP/\allowbreak{}2} ترجمه می‌کند
\item
  \lr{\mbox{5G NSA}} (غیرمستقل) همچنان از \lr{\mbox{EPC}} با \lr{\mbox{Diameter}} استفاده می‌کند
\item
  مهاجرت کامل به \lr{\mbox{5G SA}} (مستقل) \lr{\mbox{Diameter}} را درون هسته حذف می‌کند
\end{itemize}
