\CNsection{1}{مبانی سوییچینگ مداری}

\CNchained\CNsubsection{}{مفهوم: مسیر اختصاصی}

در سوییچینگ مداری، یک \textbf{مسیر ارتباطی اختصاصی} بین دو نقطه انتهایی برای کل مدت تماس برقرار می‌شود. این مسیر به‌صورت انحصاری رزرو می‌شود \lr{—} هیچ کاربر دیگری نمی‌تواند تا زمان آزادسازی تماس از آن منابع استفاده کند.

\CNsubsection{}{رزرو منابع}

\begin{itemize}
\tightlist
\item
  \textbf{واسط رادیویی}: یک شکاف زمانی \lr{\mbox{TDMA}} اختصاصی (یکی از 8 شکاف در هر فرکانس حامل) تخصیص می‌یابد
\item
  \textbf{واسط \lr{\mbox{Abis}}} \lr{(BTS\ensuremath{\leftrightarrow}BSC)}: یک کانال \lr{\mbox{16~kbps}} یا \lr{\mbox{64~kbps}} روی لینک‌های \lr{E1/\allowbreak{}T1}
\item
  \textbf{واسط \lr{\mbox{A}}} \lr{(BSC\ensuremath{\leftrightarrow}MSC)}: یک شکاف زمانی \lr{\mbox{64~kbps}} روی لینک‌های \lr{E1/\allowbreak{}T1}
\item
  \textbf{ترانک} (\lr{\mbox{MSC\ensuremath{\leftrightarrow}PSTN}}/\allowbreak{}سایر \lr{\mbox{MSC}}): یک مدار \lr{\mbox{64~kbps}} روی لینک‌های \lr{\mbox{PCM}}
\end{itemize}

\CNsubsection{}{ویژگی‌های کلیدی}

\begin{itemize}
\tightlist
\item
  \textbf{اتصال‌گرا}: مسیر پیش از جریان داده برقرار می‌شود
\item
  \textbf{پهنای باند تضمین‌شده}: \lr{\mbox{64~kbps}} ثابت در هر جهت پس از اتصال
\item
  \textbf{تأخیر ثابت}: تأخیر قابل پیش‌بینی (هدف حدود \lr{\mbox{150ms}} سرتاسری)
\item
  \textbf{ناکارآمد برای داده انفجاری}: منابع در دوره‌های سکوت هدر می‌روند
\end{itemize}

\CNsection{2}{مفاهیم سیگنالینگ \lr{\mbox{SS7}}}

\CNchained\CNsubsection{}{\lr{\mbox{SS7}} چیست؟}

\textbf{سیستم سیگنالینگ \lr{\mbox{7 (SS7)}}} پشته پروتکل سیگنالینگ خارج از باند است که توسط شبکه‌های مخابراتی برای موارد زیر استفاده می‌شود:

\begin{itemize}
\tightlist

\item

\CNsection{3}{فرآیند برقراری تماس}

\CNchained\CNsubsection{3.1}{تماس با مبدأ موبایل \lr{\mbox{(MOC)}}}

تماس با مبدأ موبایل توسط مشترک موبایل آغاز می‌شود.

\CNsubsubsection{توالی رویدادها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{\lr{\mbox{UE}} شماره را می‌گیرد} \ensuremath{\leftarrow} پیام \lr{CHANNEL REQUEST} را روی \lr{\mbox{RACH}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{BTS}}}}} آن را به \lr{\mbox{BSC}} ارسال می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{BSC}}}}} یک کانال اختصاصی \lr{\mbox{(SDCCH)}} تخصیص می‌دهد \ensuremath{\leftarrow} \lr{IMMEDIATE ASSIGNMENT}
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}} پیام \lr{CM SERVICE REQUEST} را (نوع خدمت = \lr{\mbox{MOC}}) از طریق \lr{\mbox{BSC}} به \lr{\mbox{MSC}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} مشترک را احراز هویت می‌کند (به بخش 6 مراجعه کنید)
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} رمزگذاری را فعال می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}} پیام \lr{\mbox{SETUP}} را با شماره طرف مقابل می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} پیام \lr{CALL PROCEEDING} را به \lr{\mbox{UE}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} ارقام را تحلیل می‌کند، مسیریابی را تعیین می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} پیام \lr{IAM (ISUP)} را به‌سمت \lr{GMSC/\allowbreak{}PSTN} می‌فرستد
\item
  \lr{\textbf{\lr{GMSC/\allowbreak{}PSTN}}} پیام \lr{\mbox{ACM}} (آدرس کامل) را بازمی‌گرداند \ensuremath{\leftarrow} زنگ برگشتی
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} پیام \lr{\mbox{ALERTING}} را به \lr{\mbox{UE}} می‌فرستد
\item
  \textbf{طرف مقابل پاسخ می‌دهد} \ensuremath{\leftarrow} \lr{\mbox{ANM}} (پیام پاسخ) بازگردانده می‌شود
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} پیام \lr{\mbox{CONNECT}} را به \lr{\mbox{UE}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{BSC}}}}} یک \lr{\mbox{TCH}} (کانال ترافیک) برای صدا تخصیص می‌دهد
\item
  \textbf{مسیر صدا برقرار می‌شود} \lr{—} گفتگو آغاز می‌شود
\end{enumerate}

\CNsubsection{3.2}{تماس با مقصد موبایل \lr{\mbox{(MTC)}}}

تماس با مقصد موبایل تماسی است که برای یک مشترک موبایل می‌رسد.

\CNsubsubsection{توالی رویدادها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{\lr{\mbox{PSTN}}/\allowbreak{}طرف تماس‌گیرنده} پیام \lr{\mbox{IAM}} را با \lr{\mbox{MSISDN}} مقصد به \lr{\mbox{GMSC}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{GMSC}}}}} از \lr{\mbox{HLR}} پرس‌وجو می‌کند: «این مشترک کجاست؟» \lr{(MAP: Send Routing Info)}
\item
  \lr{\mbox{\textbf{\lr{\mbox{HLR}}}}} مقدار \lr{\mbox{MSRN}} (شماره رومینگ ایستگاه موبایل) را که به \lr{\mbox{MSC}} خدمات‌دهنده اشاره دارد بازمی‌گرداند
\item
  \lr{\mbox{\textbf{\lr{\mbox{GMSC}}}}} تماس را با استفاده از \lr{MSRN (IAM)} به \lr{\mbox{MSC}} خدمات‌دهنده مسیریابی می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} از \lr{\mbox{VLR}} وضعیت مشترک و ناحیه مکانی را پرس‌وجو می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}} UE}} را در ناحیه مکانی صفحه‌بندی می‌کند
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}} به صفحه‌بندی روی \lr{\mbox{RACH}} پاسخ می‌دهد
\item
  \lr{\mbox{\textbf{\lr{\mbox{BSC}}}}} کانال \lr{\mbox{SDCCH}} را تخصیص می‌دهد \ensuremath{\leftarrow} احراز هویت و رمزگذاری
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} پیام \lr{\mbox{SETUP}} را به \lr{\mbox{UE}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}} پیام \lr{CALL CONFIRMED} را می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{UE}}}}} به کاربر هشدار می‌دهد (آهنگ زنگ)
\item
  \textbf{کاربر پاسخ می‌دهد} \ensuremath{\leftarrow} \lr{\mbox{UE}} پیام \lr{\mbox{CONNECT}} را می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{MSC}}}}} پیام \lr{\mbox{ANM}} را به‌سمت \lr{GMSC/\allowbreak{}PSTN} بازمی‌گرداند
\item
  \textbf{کانال ترافیک تخصیص می‌یابد} \lr{—} مسیر صدا فعال است
\end{enumerate}

\CNsection{4}{مسیریابی تماس}

\CNchained\CNsubsection{}{چگونه \lr{MSC/\allowbreak{}GMSC} تماس‌ها را با استفاده از \lr{\mbox{HLR}} مسیریابی می‌کند}

\lr{\mbox{HLR}} \textbf{پایگاه داده اصلی} تمام مشترکین است. می‌داند که در حال حاضر کدام \lr{MSC/\allowbreak{}VLR} به هر مشترک خدمات می‌دهد.

\CNsubsection{}{مسیریابی یک تماس با مقصد موبایل}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
Step 1: PSTN delivers call to GMSC (based on MSISDN number plan)
Step 2: GMSC sends MAP "Send Routing Information" to HLR
Step 3: HLR identifies the serving MSC/VLR for that subscriber
Step 4: HLR asks serving VLR to allocate an MSRN (temporary routing number)
Step 5: VLR returns MSRN to HLR
Step 6: HLR returns MSRN to GMSC
Step 7: GMSC uses MSRN to route the ISUP IAM to the correct MSC
\end{Verbatim}
\end{CNplaincode}

\CNsubsection{}{\lr{\mbox{MSRN}} (شماره رومینگ ایستگاه موبایل)}

\begin{itemize}
\tightlist
\item
  شماره موقت در قالب \lr{\mbox{E.164}}
\item
  مانند یک شماره تلفن عادی در طرح شماره‌گذاری شبکه خدمات‌دهنده به نظر می‌رسد
\item
  به \lr{\mbox{GMSC}} امکان می‌دهد تماس را با استفاده از \lr{\mbox{ISUP}} استاندارد و بدون اطلاع از جزییات داخلی \lr{\mbox{GSM}} مسیریابی کند
\item
  به‌ازای هر تماس تخصیص می‌یابد و پس از مسیریابی آزاد می‌شود
\end{itemize}

\CNsubsection{}{مسیریابی یک تماس با مبدأ موبایل}

برای \lr{\mbox{MOC}}، مسیریابی ساده‌تر است:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{MSC}} ارقام گرفته‌شده را از \lr{\mbox{UE}} دریافت می‌کند
\item
  \lr{\mbox{MSC}} تحلیل ارقام را انجام می‌دهد (طرح شماره‌گذاری، پیشوندها، خدمات تکمیلی)
\item
  \lr{\mbox{MSC}} تماس را از طریق \lr{\mbox{ISUP}} به‌سمت ترانک/\allowbreak{}دروازه \lr{GMSC/\allowbreak{}PSTN} مناسب مسیریابی می‌کند
\item
  اگر تماس به موبایل دیگری در همان شبکه باشد \ensuremath{\leftarrow} \lr{\mbox{MSC}} ممکن است از \lr{\mbox{HLR}} برای \lr{\mbox{MSRN}} پرس‌وجو کند
\end{enumerate}

\CNsection{5}{رویه به‌روزرسانی مکان}

\CNchained\CNsubsection{}{چرا به‌روزرسانی‌های مکان لازم هستند}

شبکه باید بداند \textbf{مشترک کجاست} تا تماس‌ها و پیام‌ها را تحویل دهد. بدون ردیابی مکان:

\begin{itemize}
\tightlist
\item
  شبکه باید هر سلول در کل کشور را صفحه‌بندی کند
\item
  این کار ظرفیت کانال صفحه‌بندی را اشباع می‌کرد
\item
  زمان برقراری تماس غیرقابل قبول می‌شد
\end{itemize}

به‌روزرسانی‌های مکان به شبکه امکان می‌دهند \textbf{صفحه‌بندی را به یک ناحیه مکانی مشخص} (گروهی از سلول‌ها) محدود کند.

\CNsubsection{}{به‌روزرسانی مکان دوره‌ای}

\begin{itemize}
\tightlist

\item

\item
  اگر تایمر بدون \lr{\mbox{LU}} منقضی شود، \lr{\mbox{VLR}} مشترک را «به‌صورت ضمنی جدا شده» علامت‌گذاری می‌کند
\item
  هدف: تشخیص گوشی‌هایی که بدون \lr{IMSI Detach} خاموش شده‌اند (باتری تمام شده و غیره)
\end{itemize}

\CNsection{6}{رویه احراز هویت}

\CNchained\CNsubsection{}{هدف}

احراز هویت تضمین می‌کند که مشترک همان کسی است که ادعا می‌کند، و از موارد زیر جلوگیری می‌کند:

\begin{itemize}
\tightlist
\item
  دسترسی غیرمجاز به شبکه
\item
  کلاهبرداری با \lr{\mbox{SIM}} کلون‌شده
\item
  شنود (در ترکیب با رمزگذاری)
\end{itemize}

\CNsubsection{}{رومینگ چگونه کار می‌کند}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{\lr{\mbox{UE}} در کشور خارجی روشن می‌شود} \ensuremath{\leftarrow} \lr{\mbox{VPLMN}} را انتخاب می‌کند (بر اساس فهرست \lr{\mbox{PLMN}} ترجیحی \lr{\mbox{SIM}})
\item
  \textbf{\lr{\mbox{UE}} به‌روزرسانی مکان را انجام می‌دهد} به \lr{MSC/\allowbreak{}VLR} مربوط به \lr{\mbox{VPLMN}}
\item
  \textbf{\lr{\mbox{VLR}} مربوط به \lr{\mbox{VPLMN}}} با \textbf{\lr{\mbox{HLR}} مربوط به \lr{\mbox{HPLMN}}} از طریق \lr{SS7/\allowbreak{}MAP} (از طریق لینک‌های سیگنالینگ بین‌المللی) تماس می‌گیرد
\item
  \lr{\mbox{\textbf{\lr{\mbox{HLR}}}}} مشترک را احراز هویت می‌کند (سه‌گانه‌ها را به \lr{\mbox{VPLMN}} می‌فرستد)
\item
  \lr{\mbox{\textbf{\lr{\mbox{HLR}}}}} پروفایل مشترک را به \lr{\mbox{VLR}} مربوط به \lr{\mbox{VPLMN}} می‌فرستد
\item
  \lr{\mbox{\textbf{\lr{\mbox{HLR}}}}} آدرس \lr{\mbox{VLR}} مربوط به \lr{\mbox{VPLMN}} را به‌عنوان مکان فعلی مشترک ثبت می‌کند
\item
  \textbf{مشترک اکنون می‌تواند از طریق \lr{\mbox{VPLMN}} تماس بگیرد/\allowbreak{}دریافت کند}
\end{enumerate}

\CNsubsection{}{تماس ورودی به یک مشترک در حال رومینگ}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  تماس به \lr{\mbox{GMSC}} مربوط به \lr{\mbox{HPLMN}} می‌رسد
\item
  \lr{\mbox{GMSC}} از \lr{\mbox{HLR}} پرس‌وجو می‌کند \ensuremath{\leftarrow} می‌فهمد که مشترک در \lr{\mbox{VPLMN}} است
\item
  \lr{\mbox{HLR}} از \lr{\mbox{VLR}} مربوط به \lr{\mbox{VPLMN}} درخواست \lr{\mbox{MSRN}} می‌کند
\item
  \lr{\mbox{VLR}} مربوط به \lr{\mbox{VPLMN}} مقدار \lr{\mbox{MSRN}} را بازمی‌گرداند
\item
  \lr{\mbox{HLR}} مقدار \lr{\mbox{MSRN}} را به \lr{\mbox{GMSC}} می‌دهد
\item
  \lr{\mbox{GMSC}} تماس را با استفاده از \lr{\mbox{MSRN}} (از طریق ترانک‌های بین‌المللی) به \lr{\mbox{MSC}} مربوط به \lr{\mbox{VPLMN}} مسیریابی می‌کند
\item
  \lr{\mbox{MSC}} مربوط به \lr{\mbox{VPLMN}} مشترک را صفحه‌بندی و متصل می‌کند
\end{enumerate}

\CNsubsection{}{تماس خروجی از یک مشترک در حال رومینگ}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{UE}} تماسی را در \lr{\mbox{VPLMN}} آغاز می‌کند
\item
  \lr{\mbox{MSC}} مربوط به \lr{\mbox{VPLMN}} تماس را به‌صورت محلی مدیریت می‌کند
\item
  تماس از \lr{\mbox{VPLMN}} به مقصد مسیریابی می‌شود (ممکن است از \lr{\mbox{HPLMN}} عبور کند یا مستقیم برود)
\item
  پروتکل \lr{CAMEL/\allowbreak{}CAP} ممکن است برای کنترل پیش‌پرداخت از \lr{\mbox{HPLMN}} استفاده شود
\end{enumerate}
