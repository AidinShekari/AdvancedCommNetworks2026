\CNsection{6.2}{توابع \lr{\mbox{eNodeB}}}

\lr{\mbox{eNodeB}} \textbf{تنها گره} در \lr{\mbox{E-UTRAN}} است. تمام توابعی را که قبلاً بین \lr{\mbox{NodeB}} و \lr{\mbox{RNC}} تقسیم شده بود انجام می‌دهد:

\CNsubsection{}{مدیریت منابع رادیویی \lr{\mbox{(RRM)}}}

\begin{itemize}
\tightlist
\item
  \textbf{کنترل حامل رادیویی:} برقراری، نگهداری و آزادسازی حامل‌های رادیویی
\item
  \textbf{کنترل پذیرش رادیویی \lr{\mbox{(RAC)}}:} پذیرش/\allowbreak{}رد درخواست‌های حامل رادیویی جدید بر اساس تداخل و بار
\item
  \textbf{کنترل تحرک اتصال:} تصمیمات تحویل بر اساس اندازه‌گیری‌های \lr{\mbox{UE}}
\item
  \textbf{تخصیص پویا منابع (زمان‌بندی):} بلوک‌های منبع را به \lr{\mbox{UE}}ها در هر \lr{TTI (1~ms)} تخصیص می‌دهد
\end{itemize}

\CNsubsection{}{زمان‌بندی}

\begin{itemize}
\tightlist
\item
  در هر \lr{\mbox{\textbf{\lr{\mbox{TTI 1~ms}}}}} (فاصله زمانی انتقال) عمل می‌کند
\item
  زمان‌بندی در دامنه فرکانس و دامنه زمان
\item
  الگوریتم‌ها: \lr{Round Robin}، \lr{Proportional Fair}، \lr{Max Throughput}
\item
  از گزارش‌های \lr{\mbox{\textbf{\lr{\mbox{CQI}}}}} (نشانگر کیفیت کانال) از \lr{\mbox{UE}}ها استفاده می‌کند
\item
  تخصیص منابع \lr{\mbox{DL}} و \lr{\mbox{UL}} را کنترل می‌کند
\end{itemize}

\CNsubsection{}{\lr{\mbox{HARQ}} (درخواست تکرار خودکار ترکیبی)}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{HARQ}} توقف-و-انتظار با \textbf{8 فرآیند موازی}
\item
  همگام در \lr{\mbox{UL}}، ناهمگام در \lr{\mbox{DL}}
\item
  ترکیب نرم: \lr{Chase Combining} یا افزونگی افزایشی
\item
  زمان رفت-و-برگشت: \lr{\mbox{8~ms}} (\lr{\mbox{4~ms}} در هر جهت)
\end{itemize}

\CNsubsection{}{کدگذاری کانال}

\begin{itemize}
\tightlist
\item
  \textbf{کدگذاری توربو} (نرخ \lr{1/\allowbreak{}3}) برای کانال‌های داده
\item
  \textbf{کدگذاری کانولوشنی} برای کانال‌های کنترل
\item
  تطبیق نرخ و مدیریت بافر \lr{\mbox{HARQ}}
\item
  الصاق \lr{\mbox{CRC}} برای تشخیص خطا
\end{itemize}

\CNsubsection{}{سایر توابع}

\begin{itemize}
\tightlist
\item
  فشرده‌سازی سرآیند \lr{IP (ROHC)}
\item
  رمزگذاری و حفاظت یکپارچگی
\item
  توزیع پیام صفحه‌بندی
\item
  اطلاعات پخش (\lr{\mbox{SIB}}ها، \lr{\mbox{MIB}}ها)
\item
  پیکربندی و گزارش‌دهی اندازه‌گیری
\end{itemize}

\CNsection{6.4}{لایه‌های پروتکل به‌تفصیل}

\CNchained\CNsubsection{6.4.1}{لایه فیزیکی \lr{\mbox{(PHY)}}}

\CNchained\CNsubsubsection{پیوند پایین‌سو: \lr{\mbox{OFDMA}} (دسترسی چندگانه تقسیم فرکانس متعامد)}

\begin{itemize}
\tightlist
\item
  پهنای باند را به بسیاری \textbf{زیرحامل} باریک تقسیم می‌کند (فاصله \lr{\mbox{15~kHz}})
\item
  هر زیرحامل متعامد است \ensuremath{\leftarrow} بدون تداخل بین-حامل
\item
  مقاوم در برابر محو شدن چندمسیری (پیشوند چرخه‌ای گسترش تأخیر را جذب می‌کند)
\item
  زمان‌بندی انعطاف‌پذیر در دامنه فرکانس را ممکن می‌سازد
\end{itemize}

\textbf{شبکه منابع:}

\begin{itemize}
\tightlist
\item
  \textbf{عنصر منبع \lr{\mbox{(RE)}}:} 1 زیرحامل \ensuremath{\times} 1 نماد \lr{\mbox{OFDM}} = کوچک‌ترین واحد قابل تخصیص
\item
  \textbf{بلوک منبع \lr{\mbox{(RB)}}:} 12 زیرحامل \ensuremath{\times} 7 نماد \lr{\mbox{OFDM}} (1 شکاف = \lr{\mbox{0.5~ms}}) = \lr{\mbox{\textbf{\lr{\mbox{84 RE}}}}}
\item
  \textbf{جفت بلوک منبع:} \lr{\mbox{2 RB = 1}} زیرفریم \lr{(TTI 1~ms)}
\end{itemize}

\textbf{ساختار فریم \lr{\mbox{(FDD)}}:}

\begin{itemize}
\tightlist
\item
  1 فریم رادیویی = \lr{10~ms = 10} زیرفریم
\item
  1 زیرفریم = \lr{\mbox{1~ms = 2}} شکاف
\item
  1 شکاف = \lr{0.5~ms = 7} نماد \lr{\mbox{OFDM}} (\lr{\mbox{CP}} معمولی)
\end{itemize}

\CNsubsubsection{پیوند بالاسو: \lr{\mbox{SC-FDMA}} (\lr{\mbox{FDMA}} تک-حامله)}

\begin{itemize}
\tightlist
\item
  همچنین \lr{\mbox{OFDM}} با گسترش \lr{\mbox{DFT}} نامیده می‌شود \lr{(DFT-s-OFDM)}
\item
  \lr{\mbox{\textbf{\lr{\mbox{PAPR}}}}} پایین‌تر (نسبت توان اوج به میانگین) نسبت به \lr{\mbox{OFDMA}} \ensuremath{\leftarrow} بهتر برای \lr{\mbox{UE}}های باتری‌دار
\item
  خاصیت تک-حامله با پیش‌کدگذاری \lr{\mbox{DFT}} پیش از \lr{\mbox{IFFT}} به دست می‌آید
\item
  همان ساختار شبکه منابع \lr{\mbox{DL}} (12 زیرحامل در هر \lr{\mbox{RB}})
\item
  \textbf{محدودیت:} تخصیص \lr{\mbox{UL}} باید در فرکانس پیوسته باشد
\end{itemize}

\CNsubsubsection{رویه‌های کلیدی \lr{\mbox{PHY}}}

\begin{itemize}
\tightlist
\item
  جستجوی سلول (همگام‌سازی \lr{PSS/\allowbreak{}SSS})
\item
  تخمین کانال (سیگنال‌های مرجع: \lr{\mbox{CRS}}، \lr{\mbox{DM-RS}}، \lr{\mbox{SRS}})
\item
  \lr{\mbox{MIMO}}: تا \lr{\mbox{4\ensuremath{\times}4}} در \lr{\mbox{DL}}، \lr{\mbox{1\ensuremath{\times}2}} یا \lr{\mbox{2\ensuremath{\times}2}} در \lr{UL (Rel-8/\allowbreak{}10)}
\item
  تطبیق پیوند از طریق بازخورد \lr{CQI/\allowbreak{}PMI/\allowbreak{}RI}
\item
  کنترل توان (حلقه باز و حلقه بسته)
\end{itemize}

\CNsubsection{6.4.2}{لایه \lr{\mbox{MAC}} (کنترل دسترسی رسانه)}

لایه \lr{\mbox{MAC}} بین \lr{\mbox{RLC}} و \lr{\mbox{PHY}} قرار دارد و مسیول \textbf{زمان‌بندی بلادرنگ} و چندگانه‌سازی داده است.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{زمان‌بندی (\lr{\mbox{DL}} و \lr{\mbox{UL}})}

  \begin{itemize}
  \tightlist
  \item
    در هر \lr{\mbox{TTI 1~ms}} عمل می‌کند
  \item
    \lr{\mbox{RB}}ها را بر اساس \lr{\mbox{CQI}}، وضعیت بافر، اولویت \lr{\mbox{QoS}}، انصاف تخصیص می‌دهد
  \item
    \lr{\mbox{DL}}: \lr{\mbox{eNB}} تصمیم می‌گیرد؛ \lr{\mbox{UL}}: \lr{\mbox{eNB}} منابع را به \lr{\mbox{UE}}ها اعطا می‌کند
  \end{itemize}
\item
  \textbf{چندگانه‌سازی/\allowbreak{}چندگانه‌گشایی}

  \begin{itemize}
  \tightlist
  \item
    داده از چندین کانال منطقی را در یک بلوک انتقال چندگانه می‌کند
  \item
    از اولویت‌بندی کانال منطقی \lr{\mbox{(LCP)}} در \lr{\mbox{UL}} استفاده می‌کند
  \end{itemize}
\item
  \textbf{موجودیت \lr{\mbox{HARQ}}}

  \begin{itemize}
  \tightlist
  \item
    8 فرآیند توقف-و-انتظار موازی
  \item
    \lr{\mbox{DL}}: \lr{\mbox{HARQ}} تطبیقی ناهمگام
  \item
    \lr{\mbox{UL}}: \lr{\mbox{HARQ}} همگام (زمان‌بندی ثابت)
  \item
    تولید \lr{ACK/\allowbreak{}NACK}؛ ترکیب نرم در گیرنده
  \end{itemize}
\item
  \textbf{رویه دسترسی تصادفی}

  \begin{itemize}
  \tightlist
  \item
    مبتنی بر رقابت (4-گام) یا بدون رقابت (2-گام)
  \item
    برای دسترسی اولیه، تحویل، بازیابی همگام‌سازی \lr{\mbox{UL}} استفاده می‌شود
  \end{itemize}
\item
  \textbf{\lr{\mbox{BSR}} (گزارش وضعیت بافر)}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{UE}} به \lr{\mbox{eNB}} درباره داده \lr{\mbox{UL}} در انتظار اطلاع می‌دهد
  \item
    اعطاهای زمان‌بندی \lr{\mbox{UL}} را فعال می‌کند
  \end{itemize}
\item
  \textbf{\lr{\mbox{PHR}} (گزارش حاشیه توان)}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{UE}} حاشیه توان باقیمانده را گزارش می‌دهد
  \item
    به \lr{\mbox{eNB}} در تصمیمات زمان‌بندی \lr{\mbox{UL}} کمک می‌کند
  \end{itemize}
\end{enumerate}

  \end{itemize}
\item
  \textbf{بازترتیبی \lr{(UM/\allowbreak{}AM)}:} \lr{\mbox{SDU}}ها را به‌ترتیب به لایه‌های بالاتر تحویل می‌دهد
\item
  \textbf{تشخیص تکراری \lr{\mbox{(AM)}}:} \lr{\mbox{PDU}}های تکراری را دور می‌ریزد
\end{itemize}

\item

\item
  ترکیبی: رسیدن به \lr{\mbox{BLER}} هدف \lr{\mbox{10\textsuperscript{-6}}} برای حامل‌های \lr{\mbox{AM}}
\end{itemize}

\CNsubsection{6.4.4}{لایه \lr{\mbox{PDCP}} (پروتکل همگرایی داده بسته)}

\lr{\mbox{PDCP}} بالای \lr{\mbox{RLC}} قرار دارد و توابع امنیت و کارایی را فراهم می‌کند.

\CNsubsubsection{توابع کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{فشرده‌سازی سرآیند (فقط صفحه کاربر)}

  \begin{itemize}
  \tightlist
  \item
    از \lr{\mbox{\textbf{\lr{\mbox{ROHC}}}}} (فشرده‌سازی مقاوم سرآیند، \lr{RFC 3095/\allowbreak{}4995}) استفاده می‌کند
  \item
    سرآیندهای \lr{IP/\allowbreak{}UDP/\allowbreak{}RTP} را از حدود \lr{\mbox{40–60}} بایت \ensuremath{\leftarrow} \lr{\mbox{1–4}} بایت فشرده می‌کند
  \item
    حیاتی برای \lr{\mbox{VoIP}}: سرآیند 40 بایتی روی بار 32 بایتی \ensuremath{\leftarrow} حدود 56\% سربار بدون \lr{\mbox{ROHC}}
  \item
    پروفایل‌های متعدد \lr{\mbox{ROHC}} برای پشته‌های پروتکل مختلف
  \end{itemize}
\item
  \textbf{رمزگذاری (صفحه کاربر + صفحه کنترل)}

  \begin{itemize}
  \tightlist
  \item
    تمام داده کاربر و سیگنالینگ را رمزنگاری می‌کند
  \item
    الگوریتم‌ها: \lr{\mbox{EEA0}} (تهی)، \lr{128-EEA1 (SNOW 3G)}، \lr{128-EEA2 (AES-CTR)}، \lr{128-EEA3 (ZUC)}
  \item
    از \lr{\mbox{COUNT}} (شماره ترتیب + \lr{\mbox{HFN}}) + \lr{BEARER + DIRECTION} به‌عنوان ورودی استفاده می‌کند
  \end{itemize}
\item
  \textbf{حفاظت یکپارچگی (فقط صفحه کنترل)}

  \begin{itemize}
  \tightlist
  \item
    سیگنالینگ \lr{\mbox{RRC}} را از دستکاری محافظت می‌کند
  \item
    الگوریتم‌ها: \lr{EIA1 (SNOW 3G)}، \lr{EIA2 (AES-CMAC)}، \lr{EIA3 (ZUC)}
  \item
    \lr{\mbox{MAC-I 32}} بیتی تولید می‌کند که به هر پیام \lr{\mbox{RRC}} الصاق می‌شود
  \item
    \textbf{در \lr{\mbox{LTE}} به صفحه کاربر اعمال نمی‌شود} (در \lr{\mbox{5G NR}} برای صفحه کاربر اضافه شد)
  \end{itemize}
\item
  \textbf{شماره‌گذاری ترتیب و بازترتیبی}

  \begin{itemize}
  \tightlist
  \item
    به هر \lr{\mbox{SDU}} یک \lr{\mbox{PDCP SN}} تخصیص می‌دهد
  \item
    \lr{\mbox{PDU}}های خارج از ترتیب را بازمرتب می‌کند (مهم در حین تحویل)
  \item
    موارد تکراری را تشخیص می‌دهد
  \end{itemize}
\item
  \textbf{پشتیبانی تحویل}

  \begin{itemize}
  \tightlist
  \item
    تحویل به‌ترتیب و حذف تکراری در حین تحویل
  \item
    گزارش‌دهی وضعیت: به \lr{\mbox{eNB}} مبدأ می‌گوید کدام \lr{\mbox{PDU}}ها با موفقیت تحویل داده شدند
  \item
    تحویل بدون اتلاف را برای حامل‌های \lr{\mbox{AM}} ممکن می‌سازد
  \end{itemize}
\end{enumerate}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{مدیریت اتصال}

  \begin{itemize}
  \tightlist
  \item
    برقراری /\allowbreak{} پیکربندی مجدد /\allowbreak{} آزادسازی اتصال \lr{\mbox{RRC}}
  \item
    فعال‌سازی امنیت (رمزگذاری + یکپارچگی)
  \item
    برقراری \lr{\mbox{SRB}} (حامل رادیویی سیگنالینگ)
  \end{itemize}
\item
  \textbf{پیکربندی و گزارش‌دهی اندازه‌گیری}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{UE}} را برای اندازه‌گیری سلول خدمات‌دهنده و همسایه پیکربندی می‌کند
  \item
    رویدادها: \lr{\mbox{A1–A6}} (درون/\allowbreak{}بین-فرکانسی)، \lr{\mbox{B1–B2}} (بین-\lr{\mbox{RAT}})
  \item
    گزارش‌دهی: دوره‌ای یا محرک-رویداد
  \item
    کمیت‌ها: \lr{\mbox{RSRP}}، \lr{\mbox{RSRQ}}، \lr{\mbox{SINR}}
  \end{itemize}
\item
  \textbf{کنترل تحویل}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{eNB}} بر اساس گزارش‌های اندازه‌گیری تصمیم تحویل می‌گیرد
  \item
    پیام \lr{\CNcode{RRCConnectionReconfiguration}} را با اطلاعات کنترل تحرک می‌فرستد
  \item
    \lr{\mbox{UE}} تحویل را انجام می‌دهد، پیام \lr{\CNcode{RRCConnectionReconfigurationComplete}} را می‌فرستد
  \end{itemize}
\item
  \textbf{پخش اطلاعات سیستم}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{\textbf{\lr{\mbox{MIB:}}}}} روی \lr{\mbox{PBCH}} هر \lr{\mbox{40~ms}} ارسال می‌شود (پهنای باند، \lr{\mbox{SFN}}، پیکربندی \lr{\mbox{PHICH}})
  \item
    \lr{\mbox{\textbf{\lr{\mbox{SIB1:}}}}} اطلاعات دسترسی سلول، زمان‌بندی سایر \lr{\mbox{SIB}}ها (دوره \lr{\mbox{80~ms}})
  \item
    \lr{\mbox{\textbf{\lr{\mbox{SIB2:}}}}} پیکربندی کانال مشترک (\lr{\mbox{RACH}}، صفحه‌بندی، کنترل توان \lr{\mbox{UL}})
  \item
    \lr{\mbox{\textbf{\lr{\mbox{SIB3–SIB8:}}}}} انتخاب مجدد سلول، اطلاعات بین-فرکانسی/\allowbreak{}بین-\lr{\mbox{RAT}}
  \item
    \lr{\mbox{\textbf{\lr{\mbox{SIB9–SIB19:}}}}} اطلاعات اضافی (\lr{\mbox{eNB}} خانگی، \lr{\mbox{MBMS}} و غیره)
  \end{itemize}
\item
  \textbf{صفحه‌بندی}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{UE}}های در حالت \lr{\mbox{IDLE}} را از تماس/\allowbreak{}داده ورودی مطلع می‌کند
  \item
    مبتنی بر \lr{\mbox{DRX}}: \lr{\mbox{UE}} در فرصت‌های صفحه‌بندی مشخص بیدار می‌شود
  \end{itemize}
\item
  \textbf{انتقال پیام \lr{\mbox{NAS}}}

  \begin{itemize}
  \tightlist
  \item
    پیام‌های \lr{\mbox{NAS}} را به‌صورت شفاف بین \lr{\mbox{UE}} و \lr{\mbox{MME}} حمل می‌کند
  \end{itemize}
\end{enumerate}

\lr{Path Switch Request} & \lr{\mbox{eNB}} مقصد \ensuremath{\leftarrow} \lr{\mbox{MME}} & مسیر \lr{\mbox{S-GW}} را پس از تحویل \lr{\mbox{X2}} به‌روزرسانی می‌کند \\

\item

\end{enumerate}

\CNsubsection{}{گذارهای حالت}

\CNchained\CNsubsubsection{\lr{IDLE \ensuremath{\to} CONNECTED} (برقراری اتصال)}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item

\item

\item

\item
  \lr{MME \textbf{\lr{Initial Context Setup}}} را آغاز می‌کند (امنیت + برقراری حامل)
\end{enumerate}

\CNsubsubsection{\lr{CONNECTED \ensuremath{\to} IDLE} (آزادسازی اتصال)}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{MME}} از طریق \lr{S1AP \CNcode{UE Context Release Command}} فعال می‌کند
\item
  \lr{\mbox{eNB}} پیام \lr{\textbf{\lr{RRC Connection Release}}} را به \lr{\mbox{UE}} می‌فرستد
\item
  \lr{\mbox{UE}} زمینه \lr{\mbox{AS}} را حذف می‌کند، به حالت بیکار بازمی‌گردد
\item
  محرک‌ها: تایمر عدم فعالیت، جداسازی صریح \lr{\mbox{NAS}}، شکست بازیابی شکست لینک رادیویی
\end{itemize}
