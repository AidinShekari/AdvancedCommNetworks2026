\CNsection{1}{مقدمه}

این ماژول یک \textbf{ترکیب جامع} از تحول شبکه موبایل از سیستم‌های آنالوگ \lr{\mbox{1G}} تا \lr{\mbox{5G}} و فراتر از آن فراهم می‌کند. این ماژول به‌عنوان یک مرجع اصلی عمل می‌کند که تمام ماژول‌های قبلی را به هم متصل می‌کند و نشان می‌دهد چگونه هر نسل مسایل مشخصی را حل کرد در حالی که چالش‌های جدیدی ایجاد کرد که انگیزه تحول بعدی شد.

تحول یک مسیر روشن را دنبال می‌کند:

\begin{itemize}
\tightlist
\item
  \textbf{صدا \ensuremath{\leftarrow} داده \ensuremath{\leftarrow} همه چیز}
\item
  \textbf{مدار \ensuremath{\leftarrow} بسته \ensuremath{\leftarrow} تمام-\lr{\mbox{IP}} \ensuremath{\leftarrow} خدمات‌محور}
\item
  \textbf{سلسله‌مراتبی \ensuremath{\leftarrow} مسطح \ensuremath{\leftarrow} بومی-ابر}
\item
  \textbf{یک‌اندازه-برای-همه \ensuremath{\leftarrow} شبکه‌بندی \ensuremath{\leftarrow} بومی-\lr{\mbox{AI}}}
\end{itemize}

\CNsection{3}{تحلیل نسل-به-نسل}

\CNchained\CNsubsection{3.1}{\lr{\mbox{GSM (2G)}} \lr{—} انقلاب صدای دیجیتال (1991)}

\textbf{مسیله حل‌شده:} \lr{\mbox{GSM}} سیستم‌های آنالوگ \lr{\mbox{1G}} (\lr{\mbox{AMPS}}، \lr{\mbox{TACS}}) را با انتقال دیجیتال جایگزین کرد و رمزنگاری، رومینگ بین‌المللی از طریق کارت‌های \lr{\mbox{SIM}} و استفاده کارآمد از طیف از طریق \lr{\mbox{TDMA}} را فراهم کرد. برای اولین بار، یک مشترک می‌توانست در سراسر اروپا سفر کند و با همان تلفن و شماره تماس بگیرد. پروتکل \lr{\mbox{GSM MAP}} مدیریت تحرک جهانی را ممکن ساخت و معماری سلسله‌مراتبی \lr{NSS (MSC/\allowbreak{}VLR/\allowbreak{}HLR/\allowbreak{}AuC)} به الگویی برای شبکه‌های موبایل تبدیل شد.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{GSM}} اساساً مدار-گزارشی بود \lr{—} یک شکاف زمانی اختصاصی برای کل مدت تماس بدون توجه به جریان واقعی داده تخصیص می‌یافت. خدمات داده \lr{\mbox{(CSD)}} به \lr{\mbox{9.6~kbps}} محدود بود و دسترسی به اینترنت را غیرعملی می‌کرد. با انفجار اینترنت در اواخر دهه 1990، اپراتورها به داده بسته-گزارشی بدون جایگزینی کل زیرساخت \lr{\mbox{GSM}} خود نیاز داشتند.

\CNsubsection{3.2}{\lr{GPRS (2.5G)} \lr{—} لایه داده بسته (2000)}

\textbf{مسیله حل‌شده:} \lr{\mbox{GPRS}} بسته-گزارشی را به‌عنوان یک \textbf{لایه رویی} بر رادیو و شبکه هسته موجود \lr{\mbox{GSM}} معرفی کرد. با افزودن گره‌های \lr{\mbox{SGSN}} و \lr{\mbox{GGSN}}، اپراتورها می‌توانستند اتصال اینترنت «همیشه-فعال» ارایه دهند که در آن کاربران منابع رادیویی را به اشتراک می‌گذاشتند و بر اساس حجم داده به‌جای زمان اتصال صورت‌حساب می‌شدند. \lr{\mbox{GPRS}} مفهوم زمینه \lr{\mbox{PDP}}، آدرس‌دهی \lr{\mbox{IP}} برای دستگاه‌های موبایل و تونل‌سازی \lr{\mbox{GTP}} را معرفی کرد \lr{—} مفاهیمی که تا \lr{\mbox{5G}} باقی مانده‌اند.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{GPRS}} از واسط رادیویی \lr{\mbox{TDMA}} در \lr{\mbox{GSM}} استفاده مجدد کرد و شکاف‌های زمانی را به اشتراک می‌گذاشت که همچنان عمدتاً برای صدا تخصیص می‌یافتند. اوج نظری \lr{171.2~kbps} به‌ندرت به دست می‌آمد (معمول: \lr{30-50~kbps}). تأخیر به دلیل ساختار فریم \lr{\mbox{TDMA}} بالا می‌ماند \lr{(300-600~ms)}. واسط رادیویی گلوگاه بود \lr{—} بدون تغییر فناوری واسط رادیویی نمی‌شد آن را رفع کرد.

\CNsubsection{3.3}{\lr{EDGE (2.75G)} \lr{—} داده بهبودیافته روی \lr{GSM (2003)}}

\textbf{مسیله حل‌شده:} \lr{\mbox{EDGE}} مدولاسیون \lr{\mbox{8PSK}} را معرفی کرد (در برابر \lr{\mbox{GMSK}} در \lr{GSM/\allowbreak{}GPRS}) و بیت در هر نماد را از 1 به 3 سه برابر کرد. با مدولاسیون و کدگذاری تطبیقی (9 طرح \lr{\mbox{MCS}})، \lr{\mbox{EDGE}} می‌توانست تا اوج نظری \lr{473.6~kbps} ارایه دهد. این یک ارتقای صرفاً نرم‌افزاری/\allowbreak{}فرستنده-گیرنده بود که نیازی به زیرساخت جدید نداشت و به اپراتورها افزایش ظرفیت مقرون‌به‌صرفه می‌داد در حالی که استقرارهای \lr{\mbox{3G}} خود را برنامه‌ریزی می‌کردند.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{EDGE}} همچنان به پهنای باند حامل \lr{\mbox{200~kHz}} در \lr{\mbox{GSM}} و ساختار \lr{\mbox{TDMA}} محدود بود. سرعت‌های دنیای واقعی \lr{100-200~kbps} نمی‌توانست از پخش جریانی ویدیو یا کاربردهای وب غنی پشتیبانی کند. محدودیت بنیادی واسط رادیویی باریک-باند بود \lr{—} دستیابی به پهن‌باند موبایل واقعی نیازمند فناوری رادیویی پهن‌باند کاملاً جدید بود.

\CNsubsection{3.4}{\lr{UMTS (3G)} \lr{—} پهن‌باند موبایل (2001)}

\textbf{مسیله حل‌شده:} \lr{\mbox{UMTS}} فناوری \lr{\mbox{WCDMA}} را با حامل‌های \lr{\mbox{5~MHz}} معرفی کرد و نرخ داده اوج تا \lr{\mbox{2~Mbps}} فراهم کرد. رویکرد طیف-گسترده کارایی طیفی بهتر، تحویل نرم برای عملکرد بهبودیافته لبه سلول و پشتیبانی بومی از صدا و داده همزمان ارایه می‌کرد. معماری \lr{UTRAN (Node B + RNC)} مدیریت منابع رادیویی متمرکز فراهم می‌کرد و هسته ترکیبی \lr{\mbox{CS+PS}} هم از صدای مدار-گزارشی و هم داده بسته پشتیبانی می‌کرد.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{RNC}} یک گلوگاه و نقطه شکست واحد بود. تأخیر به دلیل پشته پروتکل پیچیده بالا می‌ماند (رفت-و-برگشت \lr{100-200~ms}). رویکرد کانال اختصاصی برای ترافیک داده انفجاری ناکارآمد بود. صدا همچنان از سوییچینگ مدار استفاده می‌کرد و پیچیدگی دو-دامنه‌ای را حفظ می‌نمود. ضریب گسترش ثابت، تطبیق با شرایط متغیر کانال را محدود می‌کرد.

\CNsubsection{3.5}{\lr{HSPA (3.5G)} \lr{—} دسترسی بسته پرسرعت \CNnum{(2005-2008)}}

\textbf{مسیله حل‌شده:} \lr{HSPA (HSDPA + HSUPA)} کانال‌های مشترک، زمان‌بندی سریع در \lr{Node B (TTI 2ms)}، مدولاسیون و کدگذاری تطبیقی \lr{\mbox{(AMC)}} و \lr{\mbox{HARQ}} را معرفی کرد \lr{—} هوش را از \lr{\mbox{RNC}} به ایستگاه پایه منتقل کرد. \lr{\mbox{HSDPA}} تا \lr{14.4~Mbps (Rel-5)} ارایه داد و بعدها \lr{\mbox{HSPA}}+ با \lr{\mbox{64-QAM}} و \lr{\mbox{MIMO}} به \lr{\mbox{42~Mbps}} رسید. این پخش جریانی ویدیوی موبایل و فروشگاه‌های اپلیکیشن را عملی ساخت و عصر گوشی هوشمند را آغاز کرد.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{HSPA}} همچنان بر حامل \lr{\mbox{5~MHz}} در \lr{\mbox{WCDMA}} ساخته شده بود و نرخ اوج را محدود می‌کرد. صدا مدار-گزارشی باقی ماند (به‌جز آزمایش‌های \lr{\mbox{VoIP}}). \lr{\mbox{RNC}} همچنان وجود داشت و تأخیر و هزینه اضافه می‌کرد. معماری نمی‌توانست به‌طور کارآمد مقیاس یابد \lr{—} هر \lr{\mbox{RNC}} صدها \lr{\mbox{Node B}} را مدیریت می‌کرد و چالش‌های برنامه‌ریزی ظرفیت ایجاد می‌نمود. اپراتورها به یک طراحی از پایه نیاز داشتند.

\CNsubsection{3.6}{\lr{\mbox{LTE (4G)}} \lr{—} معماری مسطح تمام-\lr{IP (2009)}}

\textbf{مسیله حل‌شده:} \lr{\mbox{LTE}} یک شکست انقلابی ایجاد کرد: تمام-\lr{\mbox{IP}} از روز اول (بدون سوییچینگ مدار)، \lr{\mbox{OFDMA}} برای پیوند پایین‌سو (حذف مسایل تداخل بین-سلولی)، معماری مسطح (حذف \lr{\mbox{RNC}}، \lr{\mbox{eNodeB}} مستقیماً به \lr{\mbox{EPC}} متصل می‌شود) و هسته بسته تکامل‌یافته (جداسازی \lr{MME/\allowbreak{}S-GW/\allowbreak{}P-GW}). با پهنای باند \lr{\mbox{20~MHz}} و \lr{\mbox{MIMO 2\ensuremath{\times}2}}، \lr{\mbox{LTE}} اوج \lr{150-300~Mbps} ارایه داد. \lr{\mbox{VoLTE}} جایگزین صدای مدار-گزارشی شد. واسط \lr{\mbox{X2}} تحویل‌های مستقیم بین-\lr{\mbox{eNodeB}} بدون دخالت هسته را ممکن ساخت.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{LTE}} فاقد پشتیبانی بومی برای \lr{\mbox{IoT}} انبوه بود (\lr{\mbox{NB-IoT}} بعدها در \lr{\mbox{Rel-13}} آمد)، نمی‌توانست تأخیر زیر-میلی‌ثانیه برای کاربردهای صنعتی دست یابد، \lr{\mbox{MIMO}} محدود داشت (حداکثر 8 لایه) و واسط‌های نقطه-به-نقطه مبتنی بر \lr{\mbox{GTP}} در \lr{\mbox{EPC}} نمی‌توانستند برای عصر بومی-ابر مقیاس یابند. شبکه‌بندی با گره‌های یکپارچه \lr{\mbox{EPC}} ممکن نبود.

\CNsubsection{3.7}{\lr{LTE-Advanced (4G+)} \lr{—} عصر تجمیع حامل \CNnum{(2011-2014)}}

\textbf{مسیله حل‌شده:} \lr{LTE-Advanced} تجمیع حامل (ترکیب تا \lr{5\ensuremath{\times}20~MHz = 100 MHz})، \lr{\mbox{MIMO 8\ensuremath{\times}8}} (8 لایه \lr{\mbox{DL}})، هماهنگی بهبودیافته تداخل بین-سلولی \lr{(eICIC/\allowbreak{}FeICIC)}، گره‌های رله و چند-نقطه هماهنگ \lr{\mbox{(CoMP)}} را معرفی کرد. نرخ‌های اوج به \lr{3~Gbps (Rel-12)} رسید. شبکه‌های ناهمگون \lr{\mbox{(HetNet)}} با سلول‌های کوچک ظرفیت در نواحی متراکم را برطرف کردند. \lr{\mbox{LAA}} (دسترسی کمکی مجاز) طیف غیرمجاز را اضافه کرد.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{LTE-A}} یک تحول بود، نه انقلاب \lr{—} نمی‌توانست ساختار فریم را برای تأخیر فوق‌العاده کم به‌طور بنیادی تغییر دهد، نمی‌توانست از طیف \lr{\mbox{mmWave}} استفاده کند و \lr{\mbox{EPC}} یکپارچه باقی ماند. پشتیبانی از صنایع متنوع (خودروسازی، صنعت، بهداشت) با یک پیکربندی شبکه غیرممکن بود. صنعت به یک رادیو جدید \textbf{و} یک هسته جدید نیاز داشت.

\CNsubsection{3.8}{\lr{\mbox{5G NSA}} \lr{—} غیرمستقل (2018)}

\textbf{مسیله حل‌شده:} \lr{\mbox{5G NSA}} (گزینه \lr{3/\allowbreak{}3a/\allowbreak{}3x}) فناوری \lr{\mbox{NR}} (رادیو جدید) را با استفاده از هسته \lr{\mbox{LTE}} موجود \lr{\mbox{(EPC)}} به‌عنوان لنگر مستقر کرد. این به اپراتورها اجازه داد خدمات \lr{\mbox{5G}} را سریع بدون ساخت یک شبکه هسته کاملاً جدید راه‌اندازی کنند. اتصال دوگانه \lr{\mbox{(EN-DC)}} حامل‌های \lr{\mbox{LTE}} و \lr{\mbox{NR}} را ترکیب کرد و توان عملیاتی چند-\lr{\mbox{Gbps}} با استفاده از \lr{\mbox{mmWave}} و طیف \lr{\mbox{NR}} زیر-\lr{\mbox{6~GHz}} ارایه داد در حالی که \lr{\mbox{LTE}} را برای صفحه کنترل و صدا \lr{\mbox{(VoLTE)}} حفظ می‌کرد.

\textbf{آنچه نمی‌توانست انجام دهد:} \lr{\mbox{5G NSA}} نمی‌توانست وعده کامل \lr{\mbox{5G}} را محقق کند: بدون شبکه‌بندی (\lr{\mbox{EPC}} از آن پشتیبانی نمی‌کند)، بدون کم-تأخیر فوق‌قابل اعتماد (\lr{\mbox{URLLC}} نیازمند \lr{\mbox{5GC}} است)، بدون معماری خدمات‌محور و لنگر \lr{\mbox{LTE}} تأخیر اضافه می‌کرد. این «رادیو \lr{\mbox{5G}} با مغز \lr{\mbox{4G}}» بود \lr{—} یک سنگ پله ضروری، نه مقصد.

\CNsubsection{3.9}{\lr{\mbox{5G SA}} \lr{—} مستقل (2020)}

\textbf{مسیله حل‌شده:} \lr{\mbox{5G SA}} فناوری \lr{\mbox{NR}} را با هسته \lr{\mbox{5G (5GC)}} مستقر کرد و وابستگی به \lr{\mbox{LTE}} را حذف نمود. این شبکه‌بندی (شبکه‌های مجازی جدا برای هر مورد استفاده)، \lr{\mbox{URLLC}} با تأخیر \lr{\mbox{1ms}}، پشتیبانی بومی از \lr{\mbox{IoT}} انبوه \lr{\mbox{(mMTC)}}، یکپارچگی رایانش لبه \lr{\mbox{(MEC)}} و معماری خدمات‌محور را ممکن ساخت. جداسازی \lr{AMF/\allowbreak{}SMF/\allowbreak{}UPF} مقیاس‌پذیری مستقل را ممکن ساخت و \lr{\mbox{SBI}} (\lr{HTTP/\allowbreak{}2}، \lr{\mbox{JSON}}) استقرارهای بومی-ابر را ممکن کرد.

\textbf{آنچه نمی‌توانست انجام دهد:} پوشش \lr{\mbox{mmWave}} محدود باقی می‌ماند (برد کوتاه، حساس به انسداد)، مصرف انرژی به‌طور قابل توجهی بالاتر از \lr{\mbox{4G}} است، پیچیدگی مدیریت برش‌های شبکه در بین اپراتورها حل‌نشده است و تأخیر قطعی واقعی برای خودکارسازی صنعتی نیازمند بهبودهای بیشتر است. مهاجرت از \lr{\mbox{NSA}} به \lr{\mbox{SA}} نیازمند سرمایه‌گذاری قابل توجه با \lr{\mbox{ROI}} کوتاه‌مدت نامطمین است.

\CNsubsection{3.10}{هسته \lr{\mbox{5G}} \lr{—} معماری خدمات‌محور \CNnum{(2020-2022)}}

\textbf{مسیله حل‌شده:} هسته \lr{\mbox{5G}} واسط‌های نقطه-به-نقطه را با یک معماری خدمات‌محور جایگزین کرد که در آن توابع شبکه \lr{\mbox{(NF)}} خدمات را از طریق \lr{\mbox{API}}های \lr{RESTful (HTTP/\allowbreak{}2)} ارایه می‌دهند. \lr{\mbox{NRF}} کشف خدمات را ممکن می‌سازد، \lr{\mbox{NSSF}} انتخاب برش را مدیریت می‌کند، \lr{\mbox{PCF}} سیاست را فراهم می‌کند و \lr{\mbox{UDM}} داده مشترک را یکپارچه می‌سازد. اصول بومی-ابر (میکروسرویس‌ها، طراحی بدون حالت، کانتینری‌سازی) مقیاس‌دهی کشسان، هم‌کنش‌گری چند-سازنده و استقرار سریع ویژگی‌ها را ممکن می‌سازند.

\textbf{آنچه نمی‌توانست انجام دهد:} پیچیدگی \lr{\mbox{SBA}} اپراتورهای کوچک‌تر را به چالش می‌کشد. رومینگ بین-\lr{\mbox{PLMN}} با \lr{SBA (SEPP)} همچنان در حال بلوغ است. کارایی انرژی زیرساخت ابر توزیع‌شده یک نگرانی است. خودکارسازی سرتاسری واقعی (تأمین بدون-لمس) نیازمند یکپارچگی \lr{AI/\allowbreak{}ML} است که همچنان در حال تحول است. معماری برای تحول \lr{\mbox{6G}} آماده است اما نیازمند قابلیت‌های بومی-\lr{\mbox{AI}} است که از پایه ساخته شوند.

\CNsubsection{3.11}{\lr{\mbox{6G}} (آینده) \lr{—} شبکه‌های بومی-\lr{\mbox{AI}} (حدود 2030)}

\textbf{مسیله حل‌شده (مورد انتظار):} \lr{\mbox{6G}} هدف ارایه نرخ اوج \lr{\mbox{1~Tbps}} با استفاده از فرکانس‌های \lr{Sub-THz (100 GHz–3 THz)}، تأخیر زیر-\lr{\mbox{0.1 ms}}، سنجش و ارتباط یکپارچه \lr{\mbox{(ISAC)}}، واسط رادیویی و مدیریت شبکه بومی-\lr{\mbox{AI}}، سطوح هوشمند قابل بازپیکربندی \lr{\mbox{(RIS)}} برای انتشار قابل برنامه‌ریزی، یکپارچگی شبکه‌های غیرزمینی \lr{\mbox{(NTN)}} برای پوشش جهانی و ارتباطات معنایی/\allowbreak{}هدف-محور که معنا را به‌جای بیت منتقل می‌کنند.

\textbf{آنچه نمی‌توانست انجام دهد (چالش‌های پیش‌بینی‌شده):} انتشار \lr{\mbox{Sub-THz}} به‌شدت محدود است (متر تا ده‌ها متر) و نیازمند استقرارهای فوق-متراکم است. کارایی انرژی در این فرکانس‌ها ضعیف است. استانداردسازی پروتکل‌های بومی-\lr{\mbox{AI}} بی‌سابقه است. نگرانی‌های حریم خصوصی با سنجش فراگیر. چارچوب‌های مقرراتی برای تخصیص طیف \lr{\mbox{Sub-THz}} تعریف‌نشده‌اند. قابلیت تجاری برای بسیاری از ویژگی‌های پیشنهادی اثبات‌نشده باقی می‌ماند.

\CNsection{10}{نتیجه‌گیری}

تحول از \lr{\mbox{GSM}} به \lr{\mbox{6G}} یکی از قابل توجه‌ترین سفرهای مهندسی در تاریخ را نمایندگی می‌کند \lr{—} از تماس‌های صوتی \lr{\mbox{9.6~kbps}} تا شبکه‌های هوشمند ترابیت-در-ثانیه در چهار دهه. هر نسل محدودیت‌های نسل قبلی را حل کرد در حالی که پیچیدگی جدیدی معرفی کرد که تحول بعدی را طلب می‌نمود.

\textbf{سه ثابت در تمام نسل‌ها:}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{تقاضا همیشه از عرضه پیشی می‌گیرد} \lr{—} کاربران همیشه راه‌هایی برای مصرف ظرفیت موجود می‌یابند
\item
  \textbf{ساده‌سازی پس از پیچیدگی می‌آید} \lr{—} هر معماری بیش‌ازحد پیچیده در نهایت مسطح می‌شود
\item
  \textbf{مرز بین شبکه و رایانش محو می‌شود} \lr{—} از لوله‌های ساده تا پلتفرم‌های هوشمند
\end{enumerate}

درک این تحول فقط دانش تاریخی نیست \lr{—} به ما می‌آموزد چگونه فناوری‌های آینده را ارزیابی کنیم، تصمیمات معماری را پیش‌بینی کنیم و سیستم‌هایی طراحی کنیم که بتوانند به‌صورت تدریجی تحول یابند.
