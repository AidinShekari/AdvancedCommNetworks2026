\begin{itemize}
\tightlist

\item

\item
  سازوکارهای انتخاب مجدد سلول در حالت بیکار را توصیف کنید
\item
  انواع تحویل را در نسل‌های مختلف طبقه‌بندی و مقایسه کنید
\item
  فرآیند تصمیم تحویل را با استفاده از رویدادهای اندازه‌گیری تحلیل کنید
\item
  رویه تحویل \lr{\mbox{X2}} در \lr{\mbox{LTE}} را گام‌به‌گام دنبال کنید
\item

\item

\CNsection{1}{چرا مدیریت تحرک لازم است}

\CNchained\CNsubsection{}{چرا}

کاربران متحرک هستند \lr{—} راه می‌روند، رانندگی می‌کنند و سفر می‌کنند در حالی که انتظار تماس‌های صوتی، پخش جریانی ویدیو و نشست‌های داده بدون وقفه دارند. بدون مدیریت تحرک، کاربری که از مرز یک سلول عبور می‌کند، قطعی فوری سرویس را تجربه می‌کند.

\CNsubsection{}{چه چیزی}

مدیریت تحرک مجموعه‌ای از رویه‌های شبکه است که:

\begin{itemize}
\tightlist
\item
  مکان کاربر را \textbf{ردیابی} می‌کند (تا شبکه برای تماس‌ها/\allowbreak{}داده‌های ورودی به او دسترسی یابد)
\item
  تداوم سرویس را هنگام جابه‌جایی کاربر بین سلول‌ها \textbf{حفظ} می‌کند (تحویل)
\item
  با اتصال کاربران به بهترین سلول خدمات‌دهنده، استفاده از منابع رادیویی را \textbf{بهینه} می‌کند
\end{itemize}

\CNsubsection{}{چگونه \lr{—} ناحیه مسیریابی \lr{\mbox{(GPRS)}}}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{RA}} یک \textbf{زیرمجموعه} از یک \lr{\mbox{LA}} است (یک \lr{\mbox{LA}} شامل چندین \lr{\mbox{RA}} است)
\item
  \lr{\mbox{UE}} برای خدمات بسته‌ای \textbf{به‌روزرسانی ناحیه مسیریابی \lr{\mbox{(RAU)}}} را انجام می‌دهد
\item
  توسط \lr{\mbox{SGSN}} مدیریت می‌شود
\item
  امکان مکان دقیق‌تر برای کاربران داده را فراهم می‌کند
\end{itemize}

\CNsubsection{}{چگونه \lr{—} ناحیه ردیابی \lr{(LTE/\allowbreak{}5G)}}

\begin{itemize}
\tightlist
\item
  \textbf{فهرست ناحیه ردیابی \lr{\mbox{(TAL)}}}: به \lr{\mbox{UE}} یک فهرست از \lr{\mbox{TA}}ها تخصیص می‌یابد \lr{—} هنگام جابه‌جایی درون فهرست نیازی به به‌روزرسانی نیست
\item
  سیگنالینگ را برای کاربران پرتحرک کاهش می‌دهد (\lr{\mbox{TAL}} بزرگ‌تر)
\item
  کاربران کم‌تحرک \lr{\mbox{TAL}} کوچک‌تری می‌گیرند تا ناحیه صفحه‌بندی کاهش یابد
\end{itemize}

\item

\item
  تحرک دو-سطحی: \lr{\mbox{RNA}} (مدیریت‌شده توسط \lr{\mbox{RAN}}) + \lr{\mbox{TA}} (مدیریت‌شده توسط هسته)
\end{itemize}

\CNsection{3}{انتخاب مجدد سلول (تحرک حالت بیکار)}

\CNchained\CNsubsection{}{چرا}

حتی وقتی در تماس/\allowbreak{}نشست فعال نیست، \lr{\mbox{UE}} باید روی بهترین سلول اردو بزند تا:

\begin{itemize}
\tightlist
\item
  پیام‌های صفحه‌بندی را به‌طور کارآمد دریافت کند
\item
  هنگام نیاز به اتصال، کیفیت سیگنال خوبی داشته باشد
\item
  از به‌روزرسانی‌های غیرضروری مکان اجتناب کند
\end{itemize}

\CNsubsection{}{چه چیزی}

انتخاب مجدد سلول، فرآیند \textbf{خودمختار \lr{\mbox{UE}}} برای انتخاب سلول جدید جهت اردو زدن در حالت بیکار است. شبکه پارامترها را از طریق پخش اطلاعات سیستم (\lr{\mbox{SIB}}ها) فراهم می‌کند، اما \lr{\mbox{UE}} تصمیم را به‌صورت مستقل می‌گیرد.

\Needspace{11\baselineskip}

\CNsubsection{}{چگونه \lr{—} معیارهای انتخاب مجدد سلول}

\CNchained\CNsubsubsection{معیار رتبه‌بندی (معیار \lr{\mbox{S}})}

یک سلول برای اردو زدن مناسب است اگر معیار \lr{\mbox{S}} را برآورده کند:

\CNdisplay{
\begin{aligned}
S_{\mathrm{rxlev}} &= Q_{\mathrm{rxlevmeas}} - \left(Q_{\mathrm{rxlevmin}} + Q_{\mathrm{rxlevminoffset}}\right) - P_{\mathrm{compensation}} > 0\\
S_{\mathrm{qual}} &= Q_{\mathrm{qualmeas}} - \left(Q_{\mathrm{qualmin}} + Q_{\mathrm{qualminoffset}}\right) > 0
\end{aligned}
}

که در آن:

\begin{itemize}
\tightlist
\item
  \lr{\CNcode{Qrxlevmeas}: RSRP} اندازه‌گیری‌شده \lr{\mbox{(dBm)}}
\item
  \lr{\CNcode{Qrxlevmin}: RSRP} حداقل مورد نیاز (-124 تا \lr{\mbox{-44 dBm}}، معمول: \lr{\mbox{-124~dBm}})
\item
  \lr{\CNcode{Qqualmeas}: RSRQ} اندازه‌گیری‌شده \lr{\mbox{(dB)}}
\item
  \lr{\CNcode{Qqualmin}: RSRQ} حداقل مورد نیاز (-20 تا \lr{\mbox{-3 dB}}، معمول: \lr{\mbox{-20~dB}})
\end{itemize}

\CNsubsubsection{رتبه‌بندی سلول (معیار \lr{\mbox{R}})}

برای انتخاب مجدد درون-فرکانسی، سلول‌ها رتبه‌بندی می‌شوند:

\CNdisplay{
\begin{aligned}
R_s &= Q_{\mathrm{meas},s} + Q_{\mathrm{Hyst}} &&\text{\rl{(سلول خدمت‌دهنده)}}\\
R_n &= Q_{\mathrm{meas},n} - Q_{\mathrm{offset}} &&\text{\rl{(سلول همسایه)}}
\end{aligned}
}

اگر رابطه \lr{$R_n > R_s$} به مدت \lr{$T_{\mathrm{reselection}}$} برقرار بماند، انتخاب مجدد به سلول همسایه انجام می‌شود.

\CNsubsection{4.1}{تحویل درون-فرکانسی}

\CNchained\CNsubsubsection{چرا}

رایج‌ترین تحویل \lr{—} زمانی رخ می‌دهد که کاربر بین سلول‌های فعال روی \textbf{همان فرکانس حامل} جابه‌جا می‌شود. این سناریوی معمول در استقرارهای شهری متراکم است.

\CNsubsubsection{چه چیزی}

\begin{itemize}
\tightlist
\item
  سلول مبدأ و مقصد از همان فرکانس استفاده می‌کنند (مثلاً هر دو روی \lr{\mbox{1800~MHz}} باند 3)
\item
  \lr{\mbox{UE}} می‌تواند همسایه‌ها را \textbf{بدون شکاف‌های اندازه‌گیری} اندازه بگیرد
\item
  سریع‌ترین و ساده‌ترین نوع تحویل
\end{itemize}

\CNsubsubsection{چگونه}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} به‌طور پیوسته \lr{RSRP/\allowbreak{}RSRQ} سلول خدمات‌دهنده و همسایه‌های درون-فرکانسی را اندازه می‌گیرد
\item
  نیازی به وقفه در اندازه‌گیری نیست (فرکانس یکسان = می‌توان در حین دریافت اندازه گرفت)
\item
  توسط رویداد \lr{\mbox{A3}} فعال می‌شود: همسایه با آفست بهتر از خدمات‌دهنده می‌شود
\end{itemize}

\textbf{سناریوی معمول}: راه رفتن بین سلول‌های مجاور در یک مرکز خرید (تمام سلول‌ها روی فرکانس یکسان)

\Needspace{11\baselineskip}

\CNsubsection{4.2}{تحویل بین-فرکانسی}

\CNchained\CNsubsubsection{چرا}

شبکه‌ها چندین لایه فرکانسی برای ظرفیت مستقر می‌کنند (مثلاً ماکرو روی \lr{\mbox{800~MHz}} برای پوشش، سلول‌های کوچک روی \lr{\mbox{2600~MHz}} برای ظرفیت). کاربران باید بین این لایه‌ها جابه‌جا شوند.

\CNsubsubsection{چه چیزی}

\begin{itemize}
\tightlist
\item
  سلول مبدأ و مقصد روی \textbf{فرکانس‌های حامل متفاوت} کار می‌کنند
\item
  \lr{\mbox{UE}} برای تنظیم گیرنده خود روی فرکانس‌های دیگر نیازمند \textbf{شکاف‌های اندازه‌گیری} است
\item
  وقفه کمی طولانی‌تر از درون-فرکانسی
\end{itemize}

\CNsubsubsection{چگونه}

\begin{itemize}
\tightlist
\item
  شبکه شکاف‌های اندازه‌گیری را پیکربندی می‌کند (\lr{\mbox{6~ms}} در هر 40 یا \lr{\mbox{80~ms}})
\item
  \lr{\mbox{UE}} در حین شکاف‌ها همسایه‌های بین-فرکانسی را اندازه می‌گیرد
\item
  توسط رویداد \lr{\mbox{A4}} (همسایه روی فرکانس دیگر بهتر از آستانه می‌شود) یا \lr{\mbox{A5}} (خدمات‌دهنده افت می‌کند و همسایه خوب است) فعال می‌شود
\end{itemize}

\textbf{سناریوی معمول}: کاربر از سلول کوچک داخلی \lr{(2600~MHz)} به ماکرو بیرونی \lr{(800~MHz)} می‌رود

\Needspace{11\baselineskip}

\CNsubsection{4.3}{تحویل بین-\lr{\mbox{RAT}}}

\CNchained\CNsubsubsection{چرا}

همه نواحی پوشش \lr{LTE/\allowbreak{}5G} را ندارند. وقتی سیگنال \lr{\mbox{LTE}} افت می‌کند، \lr{\mbox{UE}} ممکن است برای حفظ سرویس نیاز به بازگشت به \lr{3G (UMTS)} یا حتی \lr{\mbox{2G (GSM)}} داشته باشد. همچنین برای بازگشت \lr{\mbox{CS}} (تماس‌های صوتی) استفاده می‌شود.

\CNsubsubsection{چه چیزی}

\begin{itemize}
\tightlist
\item
  تحویل بین فناوری‌های دسترسی رادیویی مختلف
\item
  مثال‌ها: \lr{LTE \ensuremath{\to} UMTS}، \lr{LTE \ensuremath{\to} GSM}، \lr{\mbox{NR \ensuremath{\to} LTE}}
\item
  پیچیده‌ترین نوع \lr{—} پروتکل‌های مختلف، شبکه‌های هسته مختلف
\end{itemize}

\CNsubsubsection{چگونه}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} در حین شکاف‌های اندازه‌گیری همسایه‌های بین-\lr{\mbox{RAT}} را اندازه می‌گیرد
\item
  توسط رویداد \lr{\mbox{B1}} (همسایه بین-\lr{\mbox{RAT}} بالای آستانه) یا \lr{\mbox{B2}} (خدمات‌دهنده زیر آستانه و بین-\lr{\mbox{RAT}} بالای آستانه) فعال می‌شود
\item
  نیازمند هماهنگی بین-\lr{\mbox{RAT}} بین \lr{\mbox{eNB}} و \lr{RNC/\allowbreak{}BSC} است
\end{itemize}

\textbf{سناریوی معمول}:

\begin{itemize}
\tightlist
\item
  \lr{\mbox{VoLTE}} در دسترس نیست \ensuremath{\leftarrow} بازگشت \lr{\mbox{CS}} به \lr{\mbox{3G}} برای تماس صوتی
\item
  کاربر به ناحیه روستایی با پوشش فقط \lr{\mbox{3G}} رانندگی می‌کند
\end{itemize}

\Needspace{11\baselineskip}

\CNsubsection{4.4}{تحویل سخت (قطع-پیش-از-وصل)}

\CNchained\CNsubsubsection{چرا}

وقتی اتصال همزمان به دو سلول ممکن نیست (فرکانس‌های مختلف، فناوری‌های مختلف، یا انتخاب طراحی سیستم)، اتصال قدیمی باید پیش از برقراری اتصال جدید آزاد شود.

\CNsubsubsection{چه چیزی}

\begin{itemize}
\tightlist
\item
  اتصال به سلول مبدأ \textbf{آزاد می‌شود} پیش از اتصال به مقصد
\item
  وقفه کوتاه در سرویس (معمولاً \lr{\mbox{20–50~ms}} در \lr{\mbox{LTE}})
\item
  مورد استفاده در: \lr{\mbox{GSM}}، \lr{\mbox{LTE}}، \lr{\mbox{5G NR}}
\end{itemize}

\CNkeepfig{m04-hard-ho-gap}{2}

\CNsubsubsection{چگونه}

\CNfigure{m04-hard-ho-gap}{تحویل سخت: ابتدا قطع، سپس اتصال}

\begin{itemize}
\tightlist
\item
  \textbf{مزیت}: ساده‌تر، بدون نیاز به ترکیب/\allowbreak{}تقسیم
\item
  \textbf{عیب}: وقفه کوتاه (قابل قبول برای داده، قابل توجه برای صدا در \lr{\mbox{2G}})
\end{itemize}

\Needspace{11\baselineskip}

\CNsubsection{4.5}{تحویل نرم (وصل-پیش-از-قطع) \lr{—} مختص \lr{\mbox{3G UMTS}}}

\CNchained\CNsubsubsection{چرا}

سیستم‌های \lr{\mbox{3G}} مبتنی بر \lr{\mbox{CDMA}} می‌توانند همان سیگنال را از چندین سلول به‌طور همزمان دریافت کنند (فرکانس یکسان، تقسیم کد). این امکان تحویل بی‌وقفه بدون هیچ وقفه‌ای را فراهم می‌کند.

\CNsubsubsection{چه چیزی}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} به \textbf{چندین سلول به‌طور همزمان} متصل است (مجموعه فعال)
\item
  سیگنال‌ها از چندین سلول برای بهره تنوع \textbf{ترکیب} می‌شوند
\item
  منحصر به \lr{CDMA/\allowbreak{}WCDMA (3G)} \lr{—} در \lr{\mbox{LTE}} یا \lr{\mbox{5G}} استفاده \textbf{نمی‌شود}
\end{itemize}

\CNsubsubsection{چگونه \lr{—} مدیریت مجموعه فعال}

\begin{CNplaincode}

\end{CNplaincode}

\begin{itemize}
\tightlist
\item
  \textbf{مجموعه فعال}: مجموعه سلول‌هایی که در حال حاضر به \lr{\mbox{UE}} خدمات می‌دهند (حداکثر \lr{\mbox{3–6}} سلول)
\item
  \textbf{تحویل نرم}: بین سلول‌های یک \lr{\mbox{Node B}} یا \lr{\mbox{Node B}}های مختلف، فرکانس یکسان
\item
  \textbf{تحویل نرم‌تر}: بین سکتورهای \textbf{همان} \lr{\mbox{Node B}} (ترکیب \lr{\mbox{MRC}} در \lr{\mbox{Node B}})
\end{itemize}

\textbf{مزایا}:

\begin{itemize}
\tightlist
\item
  بدون وقفه
\item
  بهره تنوع بزرگ \lr{\mbox{(2–3~dB)}}
\item
  کاهش پینگ-پونگ
\end{itemize}

\textbf{معایب}:

\begin{itemize}
\tightlist
\item
  استفاده از منابع در چندین سلول به‌طور همزمان
\item
  افزایش نیازمندی‌های بک‌هال و پردازش
\item
  افزودن پیچیدگی به کنترل توان
\end{itemize}

\CNsection{5}{فرآیند تصمیم تحویل}

\CNchained\CNsubsection{}{چرا}

شبکه باید تصمیم بگیرد چه زمانی تحویل دهد، به کدام سلول، و تضمین کند که تصمیم در برابر محو شدن و نویز مقاوم است. تحویلی که زمان‌بندی ضعیفی داشته باشد باعث قطع می‌شود؛ تحویل خوب-تنظیم‌شده برای کاربر نامحسوس است.

\CNsubsection{}{چه چیزی}

تصمیم تحویل در \lr{\mbox{LTE}} بر اساس موارد زیر است:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{گزارش‌های اندازه‌گیری \lr{\mbox{UE}}} \lr{—} دوره‌ای یا محرک-رویداد
\item
  \textbf{رویدادهای اندازه‌گیری} \lr{—} شرایطی که گزارش را فعال می‌کنند
\item
  \textbf{هیسترزیس و زمان-تا-فعال‌سازی} \lr{—} فیلترکردن برای اجتناب از محرک‌های کاذب
\item
  \textbf{تصمیم شبکه} \lr{—} \lr{\mbox{eNB}} گزارش را ارزیابی و تصمیم می‌گیرد
\end{enumerate}

\end{itemize}

\begin{itemize}
\tightlist
\item

\item
  تحویل مبتنی بر \lr{\mbox{RSRQ}}: خوب برای سناریوهای محدود-تداخل (شهری متراکم)
\end{itemize}

\CNsubsection{5.4}{حاشیه تحویل}

\CNchained\CNsubsubsection{چه چیزی}

\textbf{حاشیه کل تحویل} اثر ترکیبی آفست \lr{\mbox{A3}}، هیسترزیس و \lr{\mbox{TTT}} است که تعیین می‌کند همسایه باید چقدر بهتر باشد تا تحویل رخ دهد.

\CNdisplay{\text{\rl{حاشیه مؤثر تحویل}} \approx \mathrm{A3\_Offset} + \mathrm{Hysteresis} + \underbrace{\left(\text{\rl{سرعت UE}} \times \mathrm{TTT}\right)}_{\text{\rl{تغییر سیگنال در طول TTT}}}}

\CNsubsubsection{مثال محاسبه}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
Given:
  A3 Offset = 3 dB
  Hysteresis = 2 dB  
  TTT = 320 ms
  UE speed = 50 km/h → path loss changes ~0.5 dB during TTT

Effective margin = 3 + 2 + 0.5 = 5.5 dB

Meaning: Neighbor must be 5.5 dB better than serving 
         before handover is triggered
\end{Verbatim}
\end{CNplaincode}

\CNsubsection{}{چه چیزی}

\begin{itemize}
\tightlist
\item

\item
  کمک‌شده توسط \lr{\mbox{UE}}، کنترل‌شده توسط شبکه: \lr{\mbox{UE}} اندازه می‌گیرد، \lr{\mbox{eNB}} تصمیم می‌گیرد
\item
  فاز آماده‌سازی تضمین می‌کند که مقصد \textbf{پیش از} فرمان تحویل منابع داشته باشد
\item
  زمان کل وقفه: \lr{\mbox{\textbf{\lr{\mbox{20–50~ms}}}}} (برای اکثر کاربردها نامحسوس)
\end{itemize}

\CNsection{7}{شکست‌های تحویل}

\CNchained\CNsubsection{}{چرا}

شکست‌های تحویل مستقیماً بر تجربه کاربر تأثیر می‌گذارند \lr{—} قطع تماس، وقفه ویدیو، از دست رفتن داده. درک حالت‌های شکست برای بهینه‌سازی شبکه ضروری است.

\CNsubsection{}{چه چیزی}

چهار حالت اصلی شکست وجود دارد که هر یک علل و راه‌حل‌های متمایزی دارند:

\Needspace{11\baselineskip}

\CNsubsection{7.1}{تحویل دیر-هنگام}

\CNchained\CNsubsubsection{چرا رخ می‌دهد}

\lr{\mbox{UE}} خیلی سریع حرکت می‌کند یا حاشیه تحویل خیلی بزرگ است، بنابراین سیگنال سلول خدمات‌دهنده \textbf{پیش از} تکمیل تحویل به زیر سطح قابل استفاده می‌افتد.

\CNkeepfig{m04-too-late}{2}

\CNsubsubsection{چه اتفاقی می‌افتد}

\CNfigure{m04-too-late}{تحویل دیرهنگام: پیوند خدمت‌دهنده پیش از تحویل از دست می‌رود}

\CNsubsubsection{نشانگرها}

\begin{itemize}
\tightlist
\item
  شکست لینک رادیویی \lr{\mbox{(RLF)}} در سلول مبدأ \textbf{پیش از} دریافت فرمان تحویل
\item
  شمارنده \lr{\mbox{RLF}} بالا در سلول مبدأ
\item
  گزارش \lr{\mbox{RLF}} همسایه قوی را در زمان شکست نشان می‌دهد
\end{itemize}

\CNsubsubsection{کاهش}

\begin{itemize}
\tightlist
\item
  \textbf{کاهش} آفست \lr{\mbox{A3}} (فعال‌سازی تحویل زودتر)
\item
  \textbf{کاهش} \lr{\mbox{TTT}} (واکنش سریع‌تر)
\item
  \textbf{کاهش} هیسترزیس
\item
  افزودن گزارش‌دهی اندازه‌گیری زودهنگام (رویداد \lr{\mbox{A2}} با آستانه پایین‌تر)
\end{itemize}

\Needspace{11\baselineskip}

\CNsubsection{7.2}{تحویل زود-هنگام}

\CNchained\CNsubsubsection{چرا رخ می‌دهد}

تحویل خیلی سریع فعال می‌شود (پارامترها بیش از حد تهاجمی). \lr{\mbox{UE}} به سلولی تحویل داده می‌شود که هنوز به‌اندازه کافی قوی نیست تا اتصال را حفظ کند.

\CNsubsubsection{نشانگرها}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{RLF}} در سلول مقصد مدت کوتاهی پس از تکمیل تحویل رخ می‌دهد
\item
  \lr{\mbox{UE}} تلاش می‌کند به سلول مبدأ بازگردد و دوباره برقرار کند
\item
  زمان کوتاه بین تکمیل تحویل و \lr{\mbox{RLF}}
\end{itemize}

\begin{itemize}

\item

\item

\CNsubsection{7.4}{تحویل به سلول اشتباه}

\CNchained\CNsubsubsection{چرا رخ می‌دهد}

بهترین سلول مقصد، سلولی نیست که انتخاب شده است. این می‌تواند به دلایل زیر رخ دهد:

\begin{itemize}
\tightlist
\item
  رابطه همسایگی گمشده (مقصد در فهرست همسایه نیست)
\item
  گزارش اندازه‌گیری بهترین سلول واقعی را شامل نمی‌شود
\item
  ناهنجاری انتشار (سیگنال از سلول دور به‌طور لحظه‌ای قوی است)
\end{itemize}

\CNsubsubsection{چه اتفاقی می‌افتد}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} به سلول \lr{\mbox{X}} تحویل داده می‌شود، اما سلول \lr{\mbox{Y}} مقصد صحیح بود
\item
  \lr{\mbox{UE}} به‌سرعت سیگنال ضعیفی در سلول \lr{\mbox{X}} تجربه می‌کند
\item
  ممکن است تحویل دیگری (به سلول \lr{\mbox{Y}}) یا \lr{\mbox{RLF}} را فعال کند
\end{itemize}

\CNsubsubsection{کاهش}

\begin{itemize}
\tightlist
\item
  \textbf{رابطه همسایگی خودکار \lr{\mbox{(ANR)}}}: کشف و افزودن خودکار همسایه‌ها
\item
  \textbf{فهرست‌های جامع همسایه}: اطمینان از پیکربندی تمام مقاصد بالقوه
\item
  \textbf{آفست‌های مختص سلول}: کاهش اولویت سلول‌هایی که می‌دانیم مشکل ایجاد می‌کنند
\item
  \textbf{فیلترکردن لایه-3}: میانگین‌گیری اندازه‌گیری‌ها برای کاهش ناهنجاری‌ها
\end{itemize}

\CNsubsubsection{تحویل شرطی \lr{\mbox{(CHO)}} \lr{—} نوآوری \lr{\mbox{5G}}}

\textbf{چرا}: در \lr{\mbox{LTE}} قدیمی، \lr{\mbox{eNB}} باید منتظر گزارش اندازه‌گیری بماند، تصمیم بگیرد، مقصد را آماده کند، سپس فرمان بفرستد. اگر شرایط رادیویی به‌سرعت افت کند (سرعت بالا، \lr{\mbox{mmWave}})، این خیلی کند است.

\textbf{چه چیزی}: شبکه \textbf{چندین سلول مقصد را پیش-آماده} می‌کند و به \lr{\mbox{UE}} شرایطی می‌دهد تا تحویل را \textbf{به‌صورت خودمختار} فعال کند.

\CNfigure{m04-cho-compare}{تحویل سنتی در برابر تحویل شرطی}

\textbf{چگونه}:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  شبکه \lr{\mbox{CHO}} را پیکربندی می‌کند: «اگر رویداد \lr{\mbox{A3}} برای سلول \lr{\mbox{X}} یا سلول \lr{\mbox{Y}}، تحویل به آن سلول را اجرا کن»
\item
  سلول‌های مقصد پیش-آماده می‌شوند (منابع رزرو می‌شوند)
\item
  \lr{\mbox{UE}} شرایط را پایش می‌کند
\item
  وقتی شرط برآورده شد \ensuremath{\leftarrow} \lr{\mbox{UE}} بلافاصله اجرا می‌کند (بدون رفت-و-برگشت به شبکه)
\item
  \textbf{مزیت}: تأخیر آماده‌سازی تحویل را حذف می‌کند \ensuremath{\leftarrow} شکست‌های کمتر در سرعت بالا
\end{enumerate}

\CNsubsubsection{پشته پروتکل فعال دوگانه \lr{\mbox{(DAPS)}} \lr{—} وقفه \lr{\mbox{0~ms}}}

\textbf{چرا}: حتی وقفه \lr{\mbox{20~ms}} برای \lr{\mbox{URLLC}} (ارتباط فوق‌قابل اعتماد کم-تأخیر) مهم است.

\textbf{چه چیزی}: \lr{\mbox{UE}} در حین تحویل اتصال به \textbf{هر دو} مبدأ و مقصد را به‌طور همزمان حفظ می‌کند \lr{—} مفهومی مشابه تحویل نرم \lr{\mbox{3G}} اما با پیاده‌سازی متفاوت.

\CNfigure{m04-daps}{تحویل \lr{\mbox{DAPS}}: ابتدا اتصال، سپس قطع}

\textbf{چگونه}:

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UE}} دو پشته پروتکل فعال دارد (از این رو نام آن)
\item
  داده در حین انتقال از طریق هر دو مسیر ارسال/\allowbreak{}دریافت می‌شود
\item
  پس از تأیید اتصال موفق توسط مقصد، مبدأ آزاد می‌شود
\item
  \textbf{مبادله}: افزایش پیچیدگی و مصرف توان \lr{\mbox{UE}}
\end{itemize}

\CNsubsubsection{اتصال دوگانه \lr{\mbox{(MR-DC)}} \lr{—} چند-\lr{\mbox{RAT}}}

\textbf{چه چیزی}: \lr{\mbox{UE}} به‌طور همزمان به دو گره مختلف متصل است:

\begin{itemize}

\item

\item
  \lr{\textbf{\lr{\mbox{NR-DC}}}: NR} اصلی + \lr{\mbox{NR}} ثانویه
\item
  \lr{\textbf{\lr{\mbox{NE-DC}}}: NR} اصلی + \lr{\mbox{LTE}} ثانویه
\end{itemize}

\CNsubsubsection{رویه}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\mbox{UE}} روی \lr{\mbox{LTE}}، \lr{\mbox{RSRP}} به زیر \lr{\mbox{-105~dBm}} می‌افتد \ensuremath{\leftarrow} رویداد \lr{\mbox{A2}} \ensuremath{\leftarrow} شروع اندازه‌گیری \lr{\mbox{UMTS}}
\item
  \lr{\mbox{RSRP}} در \lr{\mbox{LTE}} به زیر \lr{\mbox{-110~dBm}} می‌افتد و \lr{\mbox{RSCP}} در \lr{\mbox{UMTS}} برای \lr{\mbox{640~ms}} بیش از \lr{\mbox{-95~dBm}} است \ensuremath{\leftarrow} رویداد \lr{\mbox{B2}}
\item
  \lr{\mbox{eNB}} تحویل بین-\lr{\mbox{RAT}} به \lr{\mbox{UMTS}} را فعال می‌کند
\item
  \lr{\mbox{UE}} تحویل سخت به \lr{\mbox{3G}} انجام می‌دهد، اتصال دامنه \lr{\mbox{CS}} را برقرار می‌کند
\item
  تماس صوتی روی \lr{\mbox{3G}} سوییچینگ مداری ادامه می‌یابد
\item
  پس از پایان تماس: \lr{\mbox{UE}} به \lr{\mbox{LTE}} بازمی‌گردد (بازگشت سریع /\allowbreak{} تغییر مسیر)
\end{enumerate}

\CNsubsubsection{فرمول فیلترکردن لایه-3}

\CNdisplay{F_n = (1 - a)\,F_{n-1} + a\,M_n, \qquad a = \left(\tfrac{1}{2}\right)^{k/4}}

که در آن \lr{$k$} ضریب فیلتر \lr{(filterCoefficient)} است (0، 1، 2، \lr{…}، 8، \lr{…}):

\begin{itemize}
\tightlist
\item

\item
  \lr{$k = 4$: $a = 0.5$} \ensuremath{\leftarrow} ۵۰٪ جدید + ۵۰٪ قدیم \ensuremath{\leftarrow} حافظه حدود \lr{\mbox{200~ms}}

\CNboxsection{پرسش‌های مرور}

\begin{CNbox}[unbreakable]{q}{پرسش‌های مرور}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{توضیح دهید} چرا \lr{\mbox{LTE}} از تحویل سخت به‌جای تحویل نرم مانند \lr{\mbox{3G}} استفاده می‌کند.
\item
  \textbf{محاسبه کنید} حاشیه تحویل مؤثر را برای: آفست \lr{A3 = 4 dB}، \lr{Hys = 2 dB}، \lr{TTT = 320 ms}، سرعت \lr{UE = 120 km/\allowbreak{}h}.
\item
  \textbf{خط زمانی} یک تحویل دیر-هنگام را رسم کنید و شناسایی کنید کدام پارامتر باید تنظیم شود.
\item
  \textbf{مقایسه کنید} نواحی ردیابی \lr{\mbox{(LTE)}} را با نواحی مکانی \lr{\mbox{(2G)}}. مزیت فهرست‌های ناحیه ردیابی چیست؟
\item
  \textbf{طراحی کنید} پارامترهای تحویل را برای سناریوی بزرگراه \lr{(130 km/\allowbreak{}h)}. هر انتخاب را توجیه کنید.
\item
  \textbf{توضیح دهید} چگونه تحویل شرطی در \lr{\mbox{5G}} مشکل تحرک بالا را که \lr{\mbox{LTE}} با آن دست‌وپنجه نرم می‌کند حل می‌کند.
\item
  \textbf{تحلیل کنید} سناریویی که نرخ پینگ-پونگ بین دو سلول 15\% است. سه راهبرد کاهش را به ترتیب ترجیح فهرست کنید.
\item
  \textbf{توصیف کنید} رویه تحویل \lr{\mbox{X2}} را و شناسایی کنید که بدون انتقال وضعیت \lr{\mbox{SN}} کجا از دست رفتن داده می‌تواند رخ دهد.
\end{enumerate}

\end{CNbox}
