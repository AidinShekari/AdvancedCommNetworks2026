\CNsection{5.1}{چرا \lr{\mbox{LTE}} توسعه یافت}

\CNchained\CNsubsection{}{محدودیت‌های \lr{3G (UMTS/\allowbreak{}HSPA)}}

پروژه مشارکت نسل سوم \lr{\mbox{(3GPP)}} پروژه تکامل بلندمدت \lr{\mbox{(LTE)}} را در سال 2004 برای رفع محدودیت‌های بنیادی شبکه‌های \lr{\mbox{3G}} آغاز کرد:

\textbf{پیچیدگی معماری:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{UTRAN}} در \lr{\mbox{3G}} نیازمند ساختار سلسله‌مراتبی بود: \lr{UE \ensuremath{\to} NodeB \ensuremath{\to} RNC \ensuremath{\to} CN}
\item
  کنترلر شبکه رادیویی \lr{\mbox{(RNC)}} یک نقطه شکست واحد و گلوگاه بود
\item
  پشته‌های پروتکل پیچیده بین \lr{\mbox{NodeB}} و \lr{\mbox{RNC}} (واسط \lr{\mbox{Iub}}) سربار پردازشی اضافه می‌کرد
\item
  تحویل نرم نیازمند هماهنگی بین چندین \lr{\mbox{NodeB}} از طریق \lr{\mbox{RNC}} بود
\end{itemize}

\textbf{مشکلات تأخیر:}

\begin{itemize}
\tightlist
\item
  زمان رفت-و-برگشت \lr{\mbox{(RTT)}} در \lr{\mbox{3G}}: معمولاً \lr{50–100 ms}، تحت بار \lr{\mbox{150+ ms}}
\item
  \lr{\mbox{RNC}} در هر دو صفحه کنترل و کاربر گام‌های پردازشی اضافه می‌کرد
\item
  صدای سوییچینگ مداری پیچیدگی و تأخیر بیشتری اضافه می‌کرد
\item
  کاربردهای بلادرنگ (بازی، تماس ویدیویی) به‌طور محسوس آسیب می‌دیدند
\end{itemize}

\textbf{کارایی طیفی:}

\begin{itemize}
\tightlist
\item
  \lr{WCDMA/\allowbreak{}HSPA} به محدودیت‌های نظری نزدیک می‌شد
\item
  تقاضا برای داده موبایل به‌صورت نمایی رشد می‌کرد (آغاز عصر گوشی‌های هوشمند)
\item
  نیاز به پهنای باند انعطاف‌پذیر: عرض کانال \lr{\mbox{1.4~MHz}} تا \lr{\mbox{20~MHz}}
\end{itemize}

\textbf{هدف تمام-\lr{\mbox{IP}}:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{3G}} همچنان دامنه سوییچینگ مداری را برای صدا (\lr{\mbox{MSC}}، \lr{\mbox{MGW}}) حفظ می‌کرد
\item
  پشته دوگانه \lr{(CS + PS) OPEX} و پیچیدگی را افزایش می‌داد
\item
  اجماع صنعت: همه چیز باید به \lr{\mbox{IP}} همگرا شود
\end{itemize}

\textbf{اهداف عملکردی برای \lr{\mbox{LTE}} (نسخه 8 گروه \lr{\mbox{3GPP}}):}\CNbr{}\lr{\textbar{}} پارامتر \lr{\textbar{}} هدف \lr{\textbar{}}\CNbr{}\lr{\mbox{\textbar{}———–\textbar{}——–\textbar{}}}\CNbr{}\lr{\textbar{}} حداکثر نرخ \lr{DL \textbar{} 100~Mbps (20~MHz) \textbar{}}\CNbr{}\lr{\textbar{}} حداکثر نرخ \lr{UL \textbar{} 50~Mbps (20~MHz) \textbar{}}\CNbr{}\lr{\textbar{}} تأخیر صفحه کاربر \lr{\textbar{}} کمتر از \lr{\mbox{5~ms}} \lr{\textbar{}}\CNbr{}\lr{\textbar{}} تأخیر صفحه کنترل \lr{\textbar{}} کمتر از \lr{\mbox{100~ms}} (بیکار\ensuremath{\leftarrow}فعال) \lr{\textbar{}}\CNbr{}\lr{\textbar{}} کارایی طیفی \lr{\textbar{}} بهبود \lr{\mbox{3–4}} برابر نسبت به \lr{\mbox{HSPA}} \lr{\textbar{}}\CNbr{}\lr{\textbar{}} تحرک \lr{\textbar{}} تا \lr{350 km/\allowbreak{}h} \lr{\textbar{}}

\CNsection{5.2}{اصول طراحی \lr{\mbox{LTE}}}

\lr{\mbox{LTE}} از ابتدا با این اصول راهنما طراحی شد:

\CNsubsection{}{معماری مسطح}

\begin{itemize}
\tightlist
\item
  \textbf{حذف \lr{\mbox{RNC}}} \lr{—} توزیع عملکردهای آن به \lr{\mbox{eNodeB}} و شبکه هسته
\item
  گره‌های شبکه کمتر = گام‌های کمتر = تأخیر کمتر
\item
  هر \lr{\mbox{eNodeB}} به‌طور مستقیم به شبکه هسته \lr{\mbox{(EPC)}} متصل می‌شود
\end{itemize}

\CNsubsection{}{تمام-بسته (بدون سوییچینگ مداری)}

\begin{itemize}
\tightlist
\item
  \textbf{شبکه کاملاً سوییچینگ بسته‌ای} \lr{—} بدون دامنه سوییچینگ مداری
\item
  صدا از طریق \lr{\mbox{VoLTE}} (صدای \lr{\mbox{LTE}}) با استفاده از \lr{\mbox{IMS}} مدیریت می‌شود
\item
  پشته پروتکل واحد عملیات را ساده می‌کند
\end{itemize}

\CNsubsection{}{بدون تحویل نرم}

\begin{itemize}
\tightlist
\item
  تحویل نرم در \lr{\mbox{3G}} نیازمند اتصالات همزمان به چندین \lr{\mbox{NodeB}} بود
\item
  \lr{\mbox{LTE}} فقط از \textbf{تحویل سخت} استفاده می‌کند \lr{—} ساده‌تر، کارآمدتر از نظر طیف
\item
  تحویل به‌اندازه کافی سریع است (وقفه حدود \lr{\mbox{20–50~ms}}) که نامحسوس باشد
\end{itemize}

\CNsubsection{}{پشته پروتکل ساده‌شده}

\begin{itemize}
\tightlist
\item
  لایه‌های کمتر، جداسازی تمیزتر دغدغه‌ها
\item
  صفحه کاربر: \lr{PDCP \ensuremath{\to} RLC \ensuremath{\to} MAC \ensuremath{\to} PHY}
\item
  صفحه کنترل: \lr{\mbox{NAS}} (بین \lr{\mbox{UE}} و \lr{\mbox{MME}}) + \lr{\mbox{RRC}} (بین \lr{\mbox{UE}} و \lr{\mbox{eNB}})
\end{itemize}

\CNsubsection{}{دسترسی کانال مشترک}

\begin{itemize}
\tightlist
\item
  بدون کانال‌های اختصاصی \lr{—} تمام منابع به‌صورت پویا به اشتراک گذاشته می‌شوند
\item
  زمان‌بند در \lr{\mbox{eNodeB}} تصمیمات \lr{\mbox{TTI 1~ms}} می‌گیرد
\item
  بهره‌های مالتی‌پلکسینگ آماری را ممکن می‌سازد
\end{itemize}

\CNsubsection{}{ویژگی‌های شبکه خود-سازمان‌ده \lr{\mbox{(SON)}}}

\begin{itemize}
\tightlist
\item
  روابط همسایگی خودکار \lr{\mbox{(ANR)}}
\item
  خود-پیکربندی \lr{\mbox{eNodeB}}های جدید
\item
  قابلیت‌های خود-بهینه‌سازی و خود-ترمیم
\end{itemize}

\CNsection{5.3}{مرور معماری \lr{\mbox{LTE}}}

\begin{enumerate}

\tightlist
\item
  \lr{\mbox{\textbf{\lr{\mbox{E-UTRAN}}}}} (\lr{\mbox{UTRAN}} تکامل‌یافته) \lr{—} شبکه دسترسی رادیویی
\item
  \lr{\mbox{\textbf{\lr{\mbox{EPC}}}}} (هسته بسته تکامل‌یافته) \lr{—} شبکه هسته
\end{enumerate}

\CNsubsection{}{\lr{\mbox{E-UTRAN}}: شبکه دسترسی رادیویی}

\CNsubsection{5.4.2}{\lr{\mbox{eNodeB}} (\lr{\mbox{NodeB}} تکامل‌یافته)}

\textbf{هدف:} گره واحد \lr{\mbox{RAN}} که تمام توابع دسترسی رادیویی را مدیریت می‌کند \lr{—} جایگزین هر دو \lr{\mbox{NodeB}} و \lr{\mbox{RNC}} از \lr{\mbox{3G}}.

\textbf{مشکلی که حل می‌کند:} گلوگاه \lr{\mbox{RNC}} را حذف می‌کند، تأخیر را با یک گام کاهش می‌دهد، تصمیمات زمان‌بندی سریع‌تر را ممکن می‌سازد.

\textbf{توابع جذب‌شده از \lr{\mbox{RNC}} در \lr{\mbox{3G}}:}

\begin{itemize}
\tightlist
\item
  مدیریت منابع رادیویی \lr{\mbox{(RRM)}} \lr{—} زمان‌بندی، کنترل پذیرش
\item
  کنترل حامل رادیویی \lr{—} برقراری، نگهداری، آزادسازی حامل‌ها
\item
  کنترل تحرک اتصال \lr{—} تصمیمات تحویل
\item
  تخصیص پویا منابع (زمان‌بند) \lr{—} دانه‌بندی \lr{\mbox{TTI 1~ms}}
\item
  هماهنگی تداخل بین-سلولی \lr{\mbox{(ICIC)}}
\item
  فشرده‌سازی سرآیند (\lr{\mbox{ROHC}} از طریق \lr{\mbox{PDCP}})
\item
  رمزنگاری و حفاظت یکپارچگی (لایه \lr{\mbox{PDCP}})
\item
  قطعه‌بندی و بازمونتاژ (لایه \lr{\mbox{RLC}})
\end{itemize}

\textbf{توابع منحصر به \lr{\mbox{eNodeB}}:}

\begin{itemize}
\tightlist
\item
  پایان‌دهی واسط \lr{\mbox{S1}} (به \lr{\mbox{MME}} و \lr{\mbox{S-GW}})
\item
  پایان‌دهی واسط \lr{\mbox{X2}} (به \lr{\mbox{eNodeB}}های همسایه)
\item
  توزیع پیام صفحه‌بندی
\item
  اطلاعات پخش (\lr{\mbox{SIB}}ها)
\item
  پیکربندی و گزارش‌دهی اندازه‌گیری
\end{itemize}

\textbf{زمان‌بند \lr{—} مغز \lr{\mbox{eNodeB}}:}

\begin{itemize}
\tightlist
\item
  هر \lr{1~ms (TTI)} تصمیمات تخصیص منابع می‌گیرد
\item
  بلوک‌های منبع \lr{\mbox{(RB)}} را به \lr{\mbox{UE}}ها بر اساس موارد زیر تخصیص می‌دهد:

  \begin{itemize}
  \tightlist
  \item
    کیفیت کانال (گزارش‌های \lr{\mbox{CQI}} از \lr{\mbox{UE}}ها)
  \item
    نیازمندی‌های \lr{\mbox{QoS}} (حامل‌های \lr{\mbox{GBR}} در برابر غیر-\lr{\mbox{GBR}})
  \item
    الگوریتم‌های انصاف (\lr{Proportional Fair}، \lr{Round Robin}، \lr{Max Throughput})
  \item
    وضعیت بافر و اولویت
  \end{itemize}
\end{itemize}

\textbf{رویه‌های \lr{\mbox{NAS}}:}

\begin{itemize}
\tightlist
\item
  \lr{\textbf{\lr{\mbox{Attach:}}} UE} با شبکه ثبت‌نام می‌کند (هویت، احراز هویت، برقراری حامل)
\item
  \lr{\textbf{\lr{\mbox{Detach:}}} UE} لغو ثبت‌نام می‌کند (اختیاری یا با شروع شبکه)
\item
  \textbf{\lr{\mbox{TAU}} (به‌روزرسانی ناحیه ردیابی):} \lr{\mbox{UE}} هنگام عبور از مرز \lr{\mbox{TA}} مکان جدید را گزارش می‌دهد
\item
  \lr{\textbf{\lr{Service Request:}} UE} بیکار درخواست فعال‌شدن می‌کند (برای انتقال داده)
\end{itemize}

\CNsubsection{5.4.4}{\lr{\mbox{S-GW}} (دروازه خدمات‌دهنده)}

\textbf{هدف:} لنگر صفحه کاربر درون \lr{\mbox{E-UTRAN}} \lr{—} تمام داده کاربر از \lr{\mbox{S-GW}} عبور می‌کند.

\textbf{مشکلی که حل می‌کند:} یک نقطه لنگر صفحه کاربر پایدار در حین تحرک درون-\lr{\mbox{LTE}} (تحویل بین \lr{\mbox{eNodeB}}ها) فراهم می‌کند. در حین تحویل فقط یک تغییر تونل در \lr{\mbox{S-GW}} لازم است به‌جای مسیریابی مجدد از طریق شبکه‌های خارجی.

\textbf{توابع کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{لنگر تحرک محلی:} در حین تحویل بین-\lr{\mbox{eNB}}، فقط تونل \lr{\mbox{S1-U}} در \lr{\mbox{S-GW}} تغییر می‌کند
\item
  \textbf{مسیریابی و ارسال بسته:} بسته‌ها را بین \lr{\mbox{eNB}} و \lr{\mbox{P-GW}} مسیریابی می‌کند
\item
  \textbf{بافرینگ داده در حین تحویل:} بسته‌های \lr{\mbox{DL}} را تا تکمیل تحویل بافر می‌کند
\item
  \textbf{اعلان داده پیوند پایین‌سو:} هنگام رسیدن داده \lr{\mbox{DL}} برای \lr{\mbox{UE}} بیکار، صفحه‌بندی را از طریق \lr{\mbox{MME}} فعال می‌کند
\item
  \textbf{رهگیری قانونی:} واسط برای رهگیری قانونی ترافیک کاربر
\item
  \textbf{جمع‌آوری داده صورت‌حساب:} داده مصرف برای صورت‌حساب آفلاین (\lr{\mbox{CDR}}ها)
\item
  \textbf{علامت‌گذاری سطح انتقال:} علامت‌گذاری \lr{\mbox{DSCP}} برای \lr{\mbox{QoS}} در شبکه انتقال
\item
  \textbf{بافرینگ حالت بیکار:} اولین بسته \lr{\mbox{DL}} را بافر می‌کند و \lr{\mbox{MME}} را برای صفحه‌بندی مطلع می‌کند
\end{itemize}

\textbf{یادداشت استقرار:} در بسیاری از شبکه‌ها، \lr{\mbox{S-GW}} و \lr{\mbox{P-GW}} برای سادگی در یک گره واحد \lr{\mbox{(SAE-GW)}} ترکیب می‌شوند.

\CNsubsection{5.4.5}{\lr{\mbox{P-GW}} (دروازه \lr{\mbox{PDN}})}

\textbf{هدف:} نقطه اتصال به شبکه‌های داده بسته‌ای خارجی (اینترنت، \lr{\mbox{IMS}}، \lr{\mbox{VPN}}های سازمانی). آدرس‌های \lr{\mbox{IP}} را تخصیص می‌دهد و سیاست را اجرا می‌کند.

\textbf{مشکلی که حل می‌کند:} یک نقطه لنگر \lr{\mbox{IP}} واحد و پایدار برای \lr{\mbox{UE}} بدون توجه به تحرک فراهم می‌کند. \lr{\mbox{UE}} حتی هنگام جابه‌جایی بین \lr{\mbox{S-GW}}ها همان آدرس \lr{\mbox{IP}} را حفظ می‌کند.

\textbf{توابع کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{تخصیص آدرس \lr{\mbox{IP}}:} آدرس‌های \lr{\mbox{IPv4}} و/\allowbreak{}یا \lr{\mbox{IPv6}} را به \lr{\mbox{UE}}ها تخصیص می‌دهد (از طریق \lr{\mbox{DHCPv4}}، \lr{\mbox{SLAAC}} یا ایستا)
\item
  \textbf{اجرای سیاست:} قواعد \lr{\mbox{QoS}} دریافتی از \lr{\mbox{PCRF}} را اعمال می‌کند (از طریق واسط \lr{\mbox{Gx}})
\item
  \textbf{صورت‌حساب:} صورت‌حساب آنلاین (\lr{\mbox{Gy}} به \lr{\mbox{OCS}}) و آفلاین (\lr{\mbox{Gz}} به \lr{\mbox{OFCS}})
\item
  \textbf{فیلترکردن بسته:} بازرسی عمیق بسته، اجرای \lr{\mbox{TFT}} (قالب جریان ترافیک)
\item
  \textbf{رفتار در هر \lr{\mbox{SDF}} (جریان داده خدمات):} اعمال \lr{\mbox{QoS}}/\allowbreak{}صورت‌حساب متفاوت در هر جریان ترافیک
\item
  \textbf{لنگر تحرک بین-اپراتوری:} در حین رومینگ، \lr{\mbox{P-GW}} در شبکه خانگی (مسیریابی خانگی) یا شبکه بازدیدشده (خروجی محلی) باقی می‌ماند
\item
  \textbf{\lr{\mbox{NAT}} (در صورت نیاز):} ترجمه آدرس شبکه برای \lr{\mbox{IPv4}}
\item
  \textbf{لنگر \lr{GTP/\allowbreak{}PMIP}:} تونل \lr{S5/\allowbreak{}S8} از \lr{\mbox{S-GW}} را پایان می‌دهد
\end{itemize}

\textbf{\lr{\mbox{APN}} (نام نقطه دسترسی):}

\begin{itemize}
\tightlist
\item
  هر اتصال \lr{\mbox{PDN}} با یک \lr{\mbox{APN}} شناسایی می‌شود
\item
  انتخاب \lr{\mbox{P-GW}} بر اساس \lr{\mbox{APN}} (مثلاً «\lr{\mbox{internet}}»، «\lr{\mbox{ims}}»، «\lr{\mbox{enterprise.vpn}}»)
\item
  چندین اتصال \lr{\mbox{PDN}} همزمان ممکن است (مثلاً اینترنت + \lr{\mbox{IMS}} برای \lr{\mbox{VoLTE}})
\end{itemize}

\CNsubsection{5.4.6}{\lr{\mbox{HSS}} (سرور مشترک خانگی)}

\textbf{هدف:} پایگاه داده مرکزی شامل تمام اطلاعات مشترک و اعتبارنامه‌های امنیتی. تحول \lr{HLR/\allowbreak{}AuC} در \lr{\mbox{3G}}.

\textbf{مشکلی که حل می‌کند:} مدیریت مشترک را متمرکز می‌کند، بردارهای احراز هویت را فراهم می‌کند و داده اشتراک را ذخیره می‌کند (\lr{\mbox{APN}}های مجاز، پروفایل‌های \lr{\mbox{QoS}}، مجوزهای رومینگ).

\textbf{توابع کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{ذخیره داده مشترک:}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{IMSI}} (هویت دایمی)
  \item
    \lr{\mbox{MSISDN}} (شماره تلفن)
  \item
    \lr{\mbox{APN}}های مشترک‌شده و پروفایل‌های \lr{\mbox{QoS}} مرتبط
  \item
    محدودیت‌های رومینگ
  \item
    نواحی ردیابی مشترک‌شده
  \end{itemize}
\item
  \textbf{تولید بردار احراز هویت:}

  \begin{itemize}
  \tightlist
  \item
    کلید دایمی \lr{\mbox{K}} را ذخیره می‌کند (مشترک با \lr{\mbox{USIM}})
  \item
    بردارهای احراز هویت (\lr{\mbox{RAND}}، \lr{\mbox{AUTN}}، \lr{\mbox{XRES}}، \lr{\mbox{\ensuremath{K_{\mathrm{ASME}}}}}) را با \lr{\mbox{EPS-AKA}} تولید می‌کند
  \item
    بردارها را به \lr{\mbox{MME}} برای احراز هویت \lr{\mbox{UE}} فراهم می‌کند
  \end{itemize}
\item
  \textbf{مدیریت مکان:}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{MME}} خدمات‌دهنده فعلی را برای هر مشترک ذخیره می‌کند
  \item
    برای تماس‌ها/\allowbreak{}نشست‌های پایانه‌ای استفاده می‌شود (کجا \lr{\mbox{UE}} را صفحه‌بندی کنیم)
  \end{itemize}
\item
  \textbf{مجوزدهی:}

  \begin{itemize}
  \tightlist
  \item
    تعیین می‌کند مشترک به کدام خدمات می‌تواند دسترسی یابد
  \item
    محدودیت‌های اشتراک را اجرا می‌کند (حداکثر نرخ بیت، \lr{\mbox{APN}}های مجاز)
  \end{itemize}
\end{itemize}

\textbf{واسط‌ها:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{S6a}} به \lr{\mbox{MME}} (پروتکل \lr{\mbox{Diameter}})
\item
  \lr{Cx/\allowbreak{}Dx} به \lr{\mbox{IMS}} (برای داده مشترک \lr{\mbox{VoLTE}})
\item
  \lr{\mbox{Sh}} به سرورهای کاربردی
\end{itemize}

\CNsubsection{5.4.7}{\lr{\mbox{PCRF}} (تابع قواعد سیاست و صورت‌حساب)}

\textbf{هدف:} مغز سیاست شبکه \lr{—} تصمیمات بلادرنگ درباره \lr{\mbox{QoS}} و قواعد صورت‌حساب اعمال‌شده بر هر جریان داده می‌گیرد.

\textbf{مشکلی که حل می‌کند:} \lr{\mbox{QoS}} پویا در هر جریان را ممکن می‌سازد (مثلاً تضمین پهنای باند برای تماس \lr{\mbox{VoLTE}} در حالی که وب‌گردی بهترین تلاش است). مدل‌های صورت‌حساب انعطاف‌پذیر را ممکن می‌سازد (تعرفه صفر، طرح‌های لایه‌ای، داده حمایت‌شده).

\textbf{توابع کلیدی:}

\begin{itemize}
\tightlist
\item
  \textbf{تصمیم سیاست:} پارامترهای \lr{\mbox{QoS}} (\lr{\mbox{QCI}}، \lr{\mbox{ARP}}، \lr{\mbox{MBR}}، \lr{\mbox{GBR}}) را برای هر جریان داده خدمات تعیین می‌کند
\item
  \textbf{قواعد صورت‌حساب:} نحوه صورت‌حساب هر جریان را مشخص می‌کند (گروه نرخ‌گذاری، روش اندازه‌گیری)
\item
  \textbf{قواعد \lr{\mbox{PCC}} (کنترل سیاست و صورت‌حساب):} دستورالعمل‌های سیاست + صورت‌حساب ترکیب‌شده که به \lr{\mbox{P-GW}} ارسال می‌شود
\item
  \textbf{به‌روزرسانی پویا سیاست:} می‌تواند قواعد را به‌صورت بلادرنگ اصلاح کند (مثلاً وقتی کاربر تماس ویدیویی را شروع می‌کند)
\item
  \textbf{سیاست مبتنی بر اشتراک:} لایه اشتراک کاربر را در نظر می‌گیرد
\item
  \textbf{سیاست مبتنی بر شرایط شبکه:} می‌تواند بر اساس ازدحام محدودسازی کند
\end{itemize}

\textbf{مثال قاعده \lr{\mbox{PCC}}:}

\tcbset{CNcodemode/.style={unbreakable}}\def\CNcodelang{}

\begin{CNplaincode}
\begin{Verbatim}
Rule: "VoLTE_Voice"
  - Service Data Flow: SIP/RTP traffic to IMS
  - QCI: 1 (Conversational Voice)
  - GBR: 40 kbps UL / 40 kbps DL
  - ARP: Priority 1, pre-emption capable
  - Charging: Included in voice plan (no data deduction)
\end{Verbatim}
\end{CNplaincode}

\textbf{واسط‌ها:}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{Gx}} به \lr{\mbox{P-GW}} (قواعد \lr{\mbox{PCC}} را نصب/\allowbreak{}اصلاح می‌کند)
\item
  \lr{\mbox{Rx}} از \lr{IMS/\allowbreak{}AF} (اطلاعات نشست را دریافت می‌کند، سیاست را فعال می‌کند)
\item
  \lr{\mbox{Sp}} به \lr{\mbox{SPR}} (مخزن پروفایل اشتراک)
\end{itemize}

\item
  اجرای \lr{\mbox{QoS}} در هر حامل
\item
  کپسوله‌سازی ساده: سرآیند \lr{\mbox{GTP-U}} (\lr{\mbox{8}}+ بایت) + بسته \lr{\mbox{IP}} داخلی
\end{itemize}

\CNsubsection{}{\lr{\mbox{CUPS}} به‌عنوان پل به \lr{\mbox{5G}}}

\item

\end{itemize}

  \item

  \end{itemize}
\item
  \textbf{تمام-\lr{\mbox{IP}} /\allowbreak{} بدون سوییچینگ مداری:}

  \begin{itemize}
  \tightlist
  \item
    پشته پروتکل واحد برای مدیریت
  \item
    کاهش \lr{\mbox{OPEX}} (بدون زیرساخت \lr{\mbox{MSC}}، \lr{\mbox{MGW}}، \lr{\mbox{VLR}})
  \item
    مالتی‌پلکسینگ آماری برای صدا (\lr{\mbox{VoLTE}} کارآمدتر)
  \item
    صورت‌حساب و سیاست یکپارچه برای تمام خدمات
  \end{itemize}
\item
  \textbf{مدل حامل \lr{\mbox{EPS}}:}

  \begin{itemize}
  \tightlist
  \item
    حامل یکپارچه از \lr{\mbox{UE}} تا \lr{\mbox{P-GW}} (\lr{\mbox{QoS}} سرتاسری)
  \item
    جایگزین مدل تکه‌تکه \lr{\mbox{3G}} (\lr{\mbox{RAB}} + زمینه \lr{PDP + QoS} جداگانه در هر واسط)
  \item
    حامل پیش‌فرض همیشه فعال (تجربه همیشه متصل)
  \item
    حامل‌های اختصاصی برای نیازهای \lr{\mbox{QoS}} مشخص (\lr{\mbox{VoLTE}}، ویدیو)
  \end{itemize}
\item
  \textbf{جداسازی صفحات کنترل و کاربر:}

  \begin{itemize}
  \tightlist
  \item
    \lr{\mbox{MME}} فقط سیگنالینگ را مدیریت می‌کند (با تعداد \lr{\mbox{UE}}ها مقیاس می‌یابد)
  \item
    \lr{\mbox{S-GW}} فقط داده را مدیریت می‌کند (با حجم ترافیک مقیاس می‌یابد)
  \item
    تحول و مقیاس‌پذیری مستقل
  \end{itemize}
\end{enumerate}

\CNboxsection{نکات کلیدی}

\begin{CNbox}[unbreakable]{sum}{نکات کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{\lr{LTE = RAN} مسطح + هسته تمام-\lr{\mbox{IP}}} \lr{—} سادگی عملکرد را هدایت می‌کند
\item
  \textbf{\lr{\mbox{eNodeB}} تنها گره \lr{\mbox{RAN}} است} \lr{—} تمام توابع \lr{\mbox{RNC}} را جذب کرده است
\item
  \textbf{\lr{\mbox{EPC}} کنترل \lr{\mbox{(MME)}} را از صفحه کاربر \lr{(S-GW/\allowbreak{}P-GW)} جدا می‌کند} \lr{—} مقیاس‌پذیری مستقل
\item
  \textbf{هر واسط هدفی روشن و واحد دارد} \lr{—} ماژولار، قابل جایگزینی
\item
  \textbf{تونل‌های \lr{\mbox{GTP}} شفافیت تحرک را فراهم می‌کنند} \lr{—} آدرس \lr{\mbox{IP}} بدون توجه به جابه‌جایی هرگز تغییر نمی‌کند
\item
  \textbf{\lr{\mbox{PCRF QoS}} پویا را ممکن می‌سازد} \lr{—} رفتار متفاوت در هر جریان کاربرد
\item
  \textbf{\lr{\mbox{IMS}} جایگزین صدای سوییچینگ مداری می‌شود} \lr{—} \lr{VoLTE = SIP} + حامل اختصاصی
\item
  \textbf{\lr{\mbox{CUPS}} پل به \lr{\mbox{5G}} است} \lr{—} همان فلسفه جداسازی، تکامل‌یافته به \lr{\mbox{SBA}} در \lr{\mbox{5G}}
\end{enumerate}

\end{CNbox}
